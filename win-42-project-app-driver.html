Project: the application and the KMDF driver

Layers 1 and 2 in full: the IOCTL contract that never changes, the application that uses it, and the KMDF driver that decides who may sign and then — because it cannot call the enclave itself — hands the work out to user mode using the inverted call model.

The IOCTL contract

This header is seam A, and it outlives every other file in the project: once an application is shipped against it, the layout of these structures is frozen. Three decisions in it are worth dwelling on before the code that uses them.

/* inc/winsign_ioctl.h  -  SEAM A
   Shared by the app (layer 1), the driver (layer 2) and,
   for the two host IOCTLs only, the host service.        */

#pragma once

#define WINSIGN_DEVICE_TYPE     42000
#define WINSIGN_MAX_DATA        (1024 * 1024)
#define WINSIGN_SIG_LEN         64          /* P-256 r||s */
#define WINSIGN_PUBKEY_LEN      64          /* P-256 x||y */

/* --- the application's two calls ------------------------ */

#define IOCTL_WINSIGN_SIGN \
    CTL_CODE(WINSIGN_DEVICE_TYPE, 0x800, METHOD_BUFFERED, \
             FILE_READ_DATA | FILE_WRITE_DATA)

#define IOCTL_WINSIGN_PUBKEY \
    CTL_CODE(WINSIGN_DEVICE_TYPE, 0x801, METHOD_BUFFERED, \
             FILE_READ_DATA)

/* --- the host service's three calls --------------------- */

#define IOCTL_WINSIGN_HOST_REGISTER \
    CTL_CODE(WINSIGN_DEVICE_TYPE, 0x900, METHOD_BUFFERED, \
             FILE_WRITE_DATA)

#define IOCTL_WINSIGN_HOST_WAIT \
    CTL_CODE(WINSIGN_DEVICE_TYPE, 0x901, METHOD_BUFFERED, \
             FILE_WRITE_DATA)

#define IOCTL_WINSIGN_HOST_RESULT \
    CTL_CODE(WINSIGN_DEVICE_TYPE, 0x902, METHOD_BUFFERED, \
             FILE_WRITE_DATA)

/* Work handed OUT to the host by HOST_WAIT.               */
typedef struct _WINSIGN_WORK {
    unsigned __int64 RequestId;
    unsigned __int32 Op;          /* 0 = sign, 1 = pubkey  */
    unsigned __int32 DataLen;
    unsigned char    Data[WINSIGN_MAX_DATA];
} WINSIGN_WORK;

/* The answer posted back by HOST_RESULT.                  */
typedef struct _WINSIGN_RESULT {
    unsigned __int64 RequestId;
    __int32          Status;      /* 0 = ok                */
    unsigned __int32 OutLen;
    unsigned char    Out[WINSIGN_SIG_LEN];
} WINSIGN_RESULT;

/* {8f0a4d31-2c7e-4b19-9f55-6ad3e2b71c40} */
DEFINE_GUID(GUID_DEVINTERFACE_WINSIGN,
    0x8f0a4d31, 0x2c7e, 0x4b19,
    0x9f, 0x55, 0x6a, 0xd3, 0xe2, 0xb7, 0x1c, 0x40);
Three decisions frozen into that header

METHOD_BUFFERED, not METHOD_NEITHER. The I/O manager copies the input into a kernel buffer before the driver ever runs, so there is no user pointer to probe and no window in which the caller can change the data underneath you. That is a whole class of driver bug removed by one word in a macro, and it costs a copy of at most a megabyte. Fixed-size output arrays. Out[64] rather than a pointer means the structure is self-contained and its layout is identical in 32- and 64-bit builds — a variable tail would require a second contract about how to find it. An explicit request id. The driver serves many clients through one host; without an id the driver would have to assume results come back in order, and a buggy or hostile host would then answer one client with another's signature.

Layer 1: the application

Nothing clever here on purpose — it opens the interface, fills the structure and calls. The only part worth reading closely is how it finds the device, because a device interface GUID is not a path and this is the Windows idiom for turning one into the other.

// app/winsign.cpp  -  LAYER 1
#include <windows.h>
#include <setupapi.h>
#include <cfgmgr32.h>
#include <stdio.h>
#include <vector>
#include <string>
#include "../inc/winsign_ioctl.h"

#pragma comment(lib, "setupapi.lib")
#pragma comment(lib, "cfgmgr32.lib")

/* A device interface GUID is not a path. CM_Get_Device_
   Interface_List turns it into one - and returns a LIST,
   because in general there can be several instances.     */
static HANDLE OpenWinsign(void)
{
    ULONG len = 0;
    CONFIGRET cr = CM_Get_Device_Interface_List_SizeW(
        &len, (LPGUID)&GUID_DEVINTERFACE_WINSIGN, NULL,
        CM_GET_DEVICE_INTERFACE_LIST_PRESENT);
    if (cr != CR_SUCCESS || len <= 1) {
        fprintf(stderr, "winsign: device not present\n");
        return INVALID_HANDLE_VALUE;
    }

    std::vector<wchar_t> buf(len);
    cr = CM_Get_Device_Interface_ListW(
        (LPGUID)&GUID_DEVINTERFACE_WINSIGN, NULL,
        buf.data(), len,
        CM_GET_DEVICE_INTERFACE_LIST_PRESENT);
    if (cr != CR_SUCCESS)
        return INVALID_HANDLE_VALUE;

    /* The list is a multi-sz; the first entry is enough.  */
    HANDLE h = CreateFileW(buf.data(),
                           GENERIC_READ | GENERIC_WRITE,
                           0, NULL, OPEN_EXISTING, 0, NULL);
    if (h == INVALID_HANDLE_VALUE)
        fprintf(stderr, "winsign: CreateFile %lu\n",
                GetLastError());
    return h;
}

static int DoSign(HANDLE h, const wchar_t *path)
{
    HANDLE f = CreateFileW(path, GENERIC_READ, FILE_SHARE_READ,
                           NULL, OPEN_EXISTING, 0, NULL);
    if (f == INVALID_HANDLE_VALUE) {
        fprintf(stderr, "open input: %lu\n", GetLastError());
        return 1;
    }

    LARGE_INTEGER size;
    if (!GetFileSizeEx(f, &size) || size.QuadPart == 0 ||
        size.QuadPart > WINSIGN_MAX_DATA) {
        fprintf(stderr, "input must be 1..%d bytes\n",
                WINSIGN_MAX_DATA);
        CloseHandle(f);
        return 1;
    }

    std::vector<unsigned char> data((size_t)size.QuadPart);
    DWORD got = 0, total = 0;
    /* ReadFile can return short. Loop. (C track, chapter 16) */
    while (total < data.size()) {
        if (!ReadFile(f, data.data() + total,
                      (DWORD)(data.size() - total),
                      &got, NULL) || got == 0) {
            fprintf(stderr, "read: %lu\n", GetLastError());
            CloseHandle(f);
            return 1;
        }
        total += got;
    }
    CloseHandle(f);

    unsigned char sig[WINSIGN_SIG_LEN];
    DWORD out = 0;
    if (!DeviceIoControl(h, IOCTL_WINSIGN_SIGN,
                         data.data(), (DWORD)data.size(),
                         sig, sizeof(sig), &out, NULL)) {
        fprintf(stderr, "sign: %lu\n", GetLastError());
        return 1;
    }
    if (out != sizeof(sig)) {
        fprintf(stderr, "sign: short result %lu\n", out);
        return 1;
    }

    std::wstring outPath(path);
    outPath += L".sig";
    HANDLE o = CreateFileW(outPath.c_str(), GENERIC_WRITE, 0,
                           NULL, CREATE_ALWAYS, 0, NULL);
    if (o == INVALID_HANDLE_VALUE) {
        fprintf(stderr, "create sig: %lu\n", GetLastError());
        return 1;
    }
    DWORD wrote = 0;
    WriteFile(o, sig, sizeof(sig), &wrote, NULL);
    CloseHandle(o);
    wprintf(L"wrote %s (%u bytes)\n", outPath.c_str(), wrote);
    return 0;
}

static int DoPubkey(HANDLE h)
{
    unsigned char pub[WINSIGN_PUBKEY_LEN];
    DWORD out = 0;
    if (!DeviceIoControl(h, IOCTL_WINSIGN_PUBKEY, NULL, 0,
                         pub, sizeof(pub), &out, NULL)) {
        fprintf(stderr, "pubkey: %lu\n", GetLastError());
        return 1;
    }
    for (DWORD i = 0; i < out; i++)
        printf("%02x", pub[i]);
    printf("\n");
    return 0;
}

int wmain(int argc, wchar_t **argv)
{
    if (argc < 2) {
        fprintf(stderr,
                "usage: winsign sign <file> | winsign pubkey\n");
        return 2;
    }

    HANDLE h = OpenWinsign();
    if (h == INVALID_HANDLE_VALUE)
        return 1;

    int rc;
    if (!wcscmp(argv[1], L"sign") && argc == 3)
        rc = DoSign(h, argv[2]);
    else if (!wcscmp(argv[1], L"pubkey"))
        rc = DoPubkey(h);
    else {
        fprintf(stderr, "unknown command\n");
        rc = 2;
    }

    CloseHandle(h);
    return rc;
}

Layer 2: the KMDF driver

The driver holds three things: a policy about who may sign, a queue of client requests waiting for an answer, and a queue of host requests waiting for work. Almost all of its length is the machinery for matching those two queues up safely.

// sys/winsign.c  -  LAYER 2
#include <ntddk.h>
#include <wdf.h>
#include <initguid.h>
#include "../inc/winsign_ioctl.h"

/* Per-device state.                                       */
typedef struct _DEVICE_CONTEXT {
    WDFQUEUE  ClientQueue;   /* client reqs awaiting answer */
    WDFQUEUE  HostQueue;     /* host reqs awaiting work     */
    WDFFILEOBJECT HostFile;  /* the registered host, or NULL*/
    WDFSPINLOCK Lock;        /* guards HostFile             */
    LONG64    NextId;
} DEVICE_CONTEXT, *PDEVICE_CONTEXT;

WDF_DECLARE_CONTEXT_TYPE_WITH_NAME(DEVICE_CONTEXT, DeviceGetCtx)

/* Per-request state: the id we assigned it.               */
typedef struct _REQUEST_CONTEXT {
    unsigned __int64 RequestId;
} REQUEST_CONTEXT, *PREQUEST_CONTEXT;

WDF_DECLARE_CONTEXT_TYPE_WITH_NAME(REQUEST_CONTEXT, ReqGetCtx)

DRIVER_INITIALIZE                      DriverEntry;
EVT_WDF_DRIVER_DEVICE_ADD              WsEvtDeviceAdd;
EVT_WDF_IO_QUEUE_IO_DEVICE_CONTROL     WsEvtIoDeviceControl;
EVT_WDF_FILE_CLOSE                     WsEvtFileClose;

/* ---------------------------------------------------------
   Matching a client request to a waiting host request
   --------------------------------------------------------- */

/* Hand one client request out to a waiting host IOCTL, if
   there is one. Called with a client request already parked
   on ClientQueue.                                          */
static VOID WsTryDispatch(PDEVICE_CONTEXT ctx)
{
    WDFREQUEST hostReq, clientReq;
    PREQUEST_CONTEXT rc;
    WINSIGN_WORK *work;
    void *inBuf;
    size_t inLen, workLen;
    NTSTATUS status;

    for (;;) {
        /* A host IOCTL waiting for work?                  */
        status = WdfIoQueueRetrieveNextRequest(ctx->HostQueue,
                                               &hostReq);
        if (!NT_SUCCESS(status))
            return;                 /* nobody is waiting   */

        /* A client request waiting for an answer?         */
        status = WdfIoQueueRetrieveNextRequest(ctx->ClientQueue,
                                               &clientReq);
        if (!NT_SUCCESS(status)) {
            /* No work. Put the host request back so it
               keeps waiting rather than returning empty. */
            WdfRequestForwardToIoQueue(hostReq, ctx->HostQueue);
            return;
        }

        rc = ReqGetCtx(clientReq);

        status = WdfRequestRetrieveOutputBuffer(hostReq,
                     sizeof(WINSIGN_WORK), (PVOID *)&work,
                     &workLen);
        if (!NT_SUCCESS(status)) {
            WdfRequestComplete(hostReq, status);
            WdfRequestForwardToIoQueue(clientReq,
                                       ctx->ClientQueue);
            continue;
        }

        RtlZeroMemory(work, sizeof(*work));
        work->RequestId = rc->RequestId;

        /* Input length zero means this was a PUBKEY call. */
        status = WdfRequestRetrieveInputBuffer(clientReq, 1,
                                               &inBuf, &inLen);
        if (NT_SUCCESS(status) && inLen > 0) {
            work->Op = 0;
            work->DataLen = (unsigned __int32)inLen;
            RtlCopyMemory(work->Data, inBuf, inLen);
        } else {
            work->Op = 1;
            work->DataLen = 0;
        }

        /* Park the client request again - it is answered by
           HOST_RESULT, not here.                          */
        status = WdfRequestForwardToIoQueue(clientReq,
                                            ctx->ClientQueue);
        if (!NT_SUCCESS(status)) {
            WdfRequestComplete(clientReq, status);
            WdfRequestComplete(hostReq, STATUS_CANCELLED);
            continue;
        }

        WdfRequestCompleteWithInformation(hostReq,
            STATUS_SUCCESS, sizeof(WINSIGN_WORK));
        return;
    }
}

/* Find the parked client request with this id and answer it. */
static NTSTATUS WsDeliverResult(PDEVICE_CONTEXT ctx,
                                const WINSIGN_RESULT *res)
{
    WDFREQUEST req = NULL;
    WDFREQUEST found = NULL;
    PREQUEST_CONTEXT rc;
    NTSTATUS status;
    void *out;
    size_t outLen;

    /* Walk the parked requests looking for our id. KMDF's
       find/retrieve pair exists exactly for this.         */
    while (NT_SUCCESS(WdfIoQueueFindRequest(ctx->ClientQueue,
               req, NULL, NULL, &found))) {
        rc = ReqGetCtx(found);
        if (rc->RequestId == res->RequestId) {
            status = WdfIoQueueRetrieveFoundRequest(
                         ctx->ClientQueue, found, &req);
            if (!NT_SUCCESS(status))
                return status;   /* it was cancelled       */
            goto answer;
        }
        if (req != NULL)
            WdfObjectDereference(req);
        req = found;
    }
    if (req != NULL)
        WdfObjectDereference(req);

    /* No such id. A correct host never does this; a hostile
       one does, which is why we simply drop it.           */
    return STATUS_NOT_FOUND;

answer:
    if (res->Status != 0) {
        WdfRequestComplete(req, STATUS_UNSUCCESSFUL);
        return STATUS_SUCCESS;
    }

    if (res->OutLen == 0 || res->OutLen > sizeof(res->Out)) {
        WdfRequestComplete(req, STATUS_INVALID_PARAMETER);
        return STATUS_SUCCESS;
    }

    status = WdfRequestRetrieveOutputBuffer(req, res->OutLen,
                                            &out, &outLen);
    if (!NT_SUCCESS(status)) {
        WdfRequestComplete(req, status);
        return STATUS_SUCCESS;
    }

    RtlCopyMemory(out, res->Out, res->OutLen);
    WdfRequestCompleteWithInformation(req, STATUS_SUCCESS,
                                      res->OutLen);
    return STATUS_SUCCESS;
}

/* ---------------------------------------------------------
   The IOCTL entry point
   --------------------------------------------------------- */

VOID WsEvtIoDeviceControl(_In_ WDFQUEUE Queue,
                          _In_ WDFREQUEST Request,
                          _In_ size_t OutputBufferLength,
                          _In_ size_t InputBufferLength,
                          _In_ ULONG IoControlCode)
{
    WDFDEVICE device = WdfIoQueueGetDevice(Queue);
    PDEVICE_CONTEXT ctx = DeviceGetCtx(device);
    PREQUEST_CONTEXT rc;
    WDF_OBJECT_ATTRIBUTES attrs;
    WINSIGN_RESULT *res;
    size_t len;
    NTSTATUS status;
    BOOLEAN isHost;

    /* Is this caller the registered host? One comparison,
       under the lock, and never trusted for anything more
       than routing.                                       */
    WdfSpinLockAcquire(ctx->Lock);
    isHost = (ctx->HostFile != NULL &&
              ctx->HostFile == WdfRequestGetFileObject(Request));
    WdfSpinLockRelease(ctx->Lock);

    switch (IoControlCode) {

    case IOCTL_WINSIGN_HOST_REGISTER:
        WdfSpinLockAcquire(ctx->Lock);
        if (ctx->HostFile != NULL) {
            WdfSpinLockRelease(ctx->Lock);
            /* One host. A second would be able to answer
               requests it did not receive.                */
            WdfRequestComplete(Request, STATUS_DEVICE_BUSY);
            return;
        }
        ctx->HostFile = WdfRequestGetFileObject(Request);
        WdfSpinLockRelease(ctx->Lock);
        WdfRequestComplete(Request, STATUS_SUCCESS);
        return;

    case IOCTL_WINSIGN_HOST_WAIT:
        if (!isHost) {
            WdfRequestComplete(Request, STATUS_ACCESS_DENIED);
            return;
        }
        if (OutputBufferLength < sizeof(WINSIGN_WORK)) {
            WdfRequestComplete(Request,
                               STATUS_BUFFER_TOO_SMALL);
            return;
        }
        status = WdfRequestForwardToIoQueue(Request,
                                            ctx->HostQueue);
        if (!NT_SUCCESS(status)) {
            WdfRequestComplete(Request, status);
            return;
        }
        WsTryDispatch(ctx);
        return;

    case IOCTL_WINSIGN_HOST_RESULT:
        if (!isHost) {
            WdfRequestComplete(Request, STATUS_ACCESS_DENIED);
            return;
        }
        status = WdfRequestRetrieveInputBuffer(Request,
                     sizeof(WINSIGN_RESULT), (PVOID *)&res,
                     &len);
        if (!NT_SUCCESS(status)) {
            WdfRequestComplete(Request, status);
            return;
        }
        WsDeliverResult(ctx, res);
        WdfRequestComplete(Request, STATUS_SUCCESS);
        return;

    case IOCTL_WINSIGN_SIGN:
        if (InputBufferLength == 0 ||
            InputBufferLength > WINSIGN_MAX_DATA) {
            WdfRequestComplete(Request,
                               STATUS_INVALID_PARAMETER);
            return;
        }
        if (OutputBufferLength < WINSIGN_SIG_LEN) {
            WdfRequestComplete(Request,
                               STATUS_BUFFER_TOO_SMALL);
            return;
        }
        break;

    case IOCTL_WINSIGN_PUBKEY:
        if (OutputBufferLength < WINSIGN_PUBKEY_LEN) {
            WdfRequestComplete(Request,
                               STATUS_BUFFER_TOO_SMALL);
            return;
        }
        break;

    default:
        WdfRequestComplete(Request,
                           STATUS_INVALID_DEVICE_REQUEST);
        return;
    }

    /* A client request. Refuse early if no host is running -
       far better than hanging until the user gives up.    */
    WdfSpinLockAcquire(ctx->Lock);
    isHost = (ctx->HostFile != NULL);
    WdfSpinLockRelease(ctx->Lock);
    if (!isHost) {
        WdfRequestComplete(Request, STATUS_DEVICE_NOT_READY);
        return;
    }

    WDF_OBJECT_ATTRIBUTES_INIT_CONTEXT_TYPE(&attrs,
                                            REQUEST_CONTEXT);
    status = WdfObjectAllocateContext(Request, &attrs,
                                      (PVOID *)&rc);
    if (!NT_SUCCESS(status)) {
        WdfRequestComplete(Request, status);
        return;
    }
    rc->RequestId = (unsigned __int64)
        InterlockedIncrement64(&ctx->NextId);

    status = WdfRequestForwardToIoQueue(Request,
                                        ctx->ClientQueue);
    if (!NT_SUCCESS(status)) {
        WdfRequestComplete(Request, status);
        return;
    }

    WsTryDispatch(ctx);
}

/* If the host dies, every parked client request would wait
   forever. Purge them instead.                            */
VOID WsEvtFileClose(_In_ WDFFILEOBJECT FileObject)
{
    WDFDEVICE device = WdfFileObjectGetDevice(FileObject);
    PDEVICE_CONTEXT ctx = DeviceGetCtx(device);
    BOOLEAN wasHost;

    WdfSpinLockAcquire(ctx->Lock);
    wasHost = (ctx->HostFile == FileObject);
    if (wasHost)
        ctx->HostFile = NULL;
    WdfSpinLockRelease(ctx->Lock);

    if (wasHost) {
        WdfIoQueuePurgeSynchronously(ctx->ClientQueue);
        WdfIoQueueStart(ctx->ClientQueue);
    }
}

/* ---------------------------------------------------------
   Setup
   --------------------------------------------------------- */

NTSTATUS WsEvtDeviceAdd(_In_ WDFDRIVER Driver,
                        _Inout_ PWDFDEVICE_INIT DeviceInit)
{
    WDF_OBJECT_ATTRIBUTES attrs;
    WDF_FILEOBJECT_CONFIG fileConfig;
    WDF_IO_QUEUE_CONFIG queueConfig;
    WDFDEVICE device;
    PDEVICE_CONTEXT ctx;
    NTSTATUS status;

    UNREFERENCED_PARAMETER(Driver);

    WDF_FILEOBJECT_CONFIG_INIT(&fileConfig, NULL,
                               WDF_NO_EVENT_CALLBACK,
                               WsEvtFileClose);
    WdfDeviceInitSetFileObjectConfig(DeviceInit, &fileConfig,
                                     WDF_NO_OBJECT_ATTRIBUTES);

    WDF_OBJECT_ATTRIBUTES_INIT_CONTEXT_TYPE(&attrs,
                                            DEVICE_CONTEXT);
    status = WdfDeviceCreate(&DeviceInit, &attrs, &device);
    if (!NT_SUCCESS(status))
        return status;

    ctx = DeviceGetCtx(device);
    RtlZeroMemory(ctx, sizeof(*ctx));

    WDF_OBJECT_ATTRIBUTES_INIT(&attrs);
    attrs.ParentObject = device;
    status = WdfSpinLockCreate(&attrs, &ctx->Lock);
    if (!NT_SUCCESS(status))
        return status;

    /* The default queue takes the IOCTLs.                 */
    WDF_IO_QUEUE_CONFIG_INIT_DEFAULT_QUEUE(&queueConfig,
                                       WdfIoQueueDispatchParallel);
    queueConfig.EvtIoDeviceControl = WsEvtIoDeviceControl;
    status = WdfIoQueueCreate(device, &queueConfig,
                              WDF_NO_OBJECT_ATTRIBUTES, NULL);
    if (!NT_SUCCESS(status))
        return status;

    /* Two MANUAL queues: KMDF never dispatches from these,
       which is exactly what "parked" means.               */
    WDF_IO_QUEUE_CONFIG_INIT(&queueConfig,
                             WdfIoQueueDispatchManual);
    status = WdfIoQueueCreate(device, &queueConfig,
                              WDF_NO_OBJECT_ATTRIBUTES,
                              &ctx->ClientQueue);
    if (!NT_SUCCESS(status))
        return status;

    WDF_IO_QUEUE_CONFIG_INIT(&queueConfig,
                             WdfIoQueueDispatchManual);
    status = WdfIoQueueCreate(device, &queueConfig,
                              WDF_NO_OBJECT_ATTRIBUTES,
                              &ctx->HostQueue);
    if (!NT_SUCCESS(status))
        return status;

    return WdfDeviceCreateDeviceInterface(device,
               &GUID_DEVINTERFACE_WINSIGN, NULL);
}

NTSTATUS DriverEntry(_In_ PDRIVER_OBJECT DriverObject,
                     _In_ PUNICODE_STRING RegistryPath)
{
    WDF_DRIVER_CONFIG config;

    WDF_DRIVER_CONFIG_INIT(&config, WsEvtDeviceAdd);
    return WdfDriverCreate(DriverObject, RegistryPath,
                           WDF_NO_OBJECT_ATTRIBUTES,
                           &config, WDF_NO_HANDLE);
}

Where the access decision actually lives

There is no token check in the code above, and that is deliberate rather than an omission. The instinct is to call something like a privilege check inside EvtIoDeviceControl — and on Windows that instinct is usually wrong.

   WHY NOT CHECK THE TOKEN IN EvtIoDeviceControl:

     · the callback is not guaranteed to run in the calling
       thread's context, so "the current token" may not be
       the caller's token at all
     · by IOCTL time the handle already exists; a check here
       is later than it needed to be
     · you would be reimplementing, imperfectly, an access
       check the I/O manager already performs

   WHERE IT BELONGS: a security descriptor on the device
   interface, evaluated by the I/O manager at CreateFile
   time, declared in the INF:

     [winsign_Device.NT.HW]
     AddReg = winsign_SecurityAddReg

     [winsign_SecurityAddReg]
     HKR,,Security,,"D:P(A;;GA;;;SY)(A;;GA;;;BA)"

   READ THE SDDL: D:P is a protected DACL that does not
   inherit. (A;;GA;;;SY) grants all access to SYSTEM,
   (A;;GA;;;BA) to the Builtin Administrators group. Nobody
   else is mentioned, so nobody else gets in.

   THE HOST runs as SYSTEM and matches the first ACE. To let
   ordinary users sign, add an ACE for them - and note that
   is a POLICY change in one string, not a code change.
This is the weakest of the four checks, and knowing that is the point

An SDDL string stops an unprivileged user. It does not stop an administrator, and it certainly does not stop someone who already has kernel code execution — they can rewrite the descriptor, or simply call the driver from kernel mode. That is not a flaw to fix here; it is the reason layers 3 and 4 exist. Every check at layer 2 is a convenience against ordinary misuse, and the design deliberately does not depend on any of them for the property it actually claims.

Building, signing and installing

Two build systems again — the driver needs the WDK and a test-signing certificate, the application needs neither. The certificate steps are once per machine, not once per build.

# --- once per test machine, from an elevated prompt -------
bcdedit /set testsigning on
shutdown /r /t 0

# --- once per project -------------------------------------
# Visual Studio's "Kernel Mode Driver, Empty (KMDF)" template
# generates a test certificate and signs on build. To do it
# by hand:
makecert -r -pe -ss PrivateCertStore -n "CN=WinsignTest" ^
         winsign.cer
signtool sign /v /s PrivateCertStore /n WinsignTest ^
         /fd sha256 /t http://timestamp.digicert.com ^
         winsign.sys
certmgr -add winsign.cer -s -r localMachine root
certmgr -add winsign.cer -s -r localMachine trustedpublisher

# --- install ----------------------------------------------
# winsign.sys is a software-only device, so it needs a root-
# enumerated device to attach to. devcon creates one:
devcon install winsign.inf ROOT\WINSIGN

# --- confirm it came up -----------------------------------
pnputil /enum-devices /class System /connected
# and in WinDbg:
!wdfkd.wdfldr
Check VBS before going any further

Everything in chapters 42 and 43 fails in confusing ways if virtualisation-based security is not actually running, and nothing tells you so directly. msinfo32 reports it under "Virtualization-based security", or from PowerShell: Get-CimInstance -ClassName Win32_DeviceGuard -Namespace root\Microsoft\Windows\DeviceGuard — you want VirtualizationBasedSecurityStatus to be 2, meaning running. If it is 0 or 1, fix that first; no amount of debugging the enclave will help.

Proving layers 1 and 2 before the enclave exists

The whole point of the build order in chapter 40 is that this half can be finished and tested on its own. Twenty lines of fake host give you a working end-to-end system with a stub signature.

/* tools/fakehost.cpp - a host that signs nothing.
   It proves seams A and B and nothing else.            */
HANDLE h = OpenWinsign();
DeviceIoControl(h, IOCTL_WINSIGN_HOST_REGISTER,
                NULL, 0, NULL, 0, &out, NULL);

for (;;) {
    WINSIGN_WORK work;
    if (!DeviceIoControl(h, IOCTL_WINSIGN_HOST_WAIT,
                         NULL, 0, &work, sizeof(work),
                         &out, NULL))
        break;

    WINSIGN_RESULT res = {};
    res.RequestId = work.RequestId;
    res.Status    = 0;
    res.OutLen    = WINSIGN_SIG_LEN;
    memset(res.Out, 0xAB, sizeof(res.Out));   /* a stub */

    DeviceIoControl(h, IOCTL_WINSIGN_HOST_RESULT,
                    &res, sizeof(res), NULL, 0, &out, NULL);
}

/* THE TEST:
     winsign sign C:\test.txt
     certutil -dump C:\test.txt.sig
   should show 64 bytes of 0xAB. If it does, seams A and B
