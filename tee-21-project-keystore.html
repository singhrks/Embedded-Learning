Project: a secure key store, client and TA

A complete, working pair: a trusted application that generates ECDSA keys which can never leave the secure world, and a command-line client that drives it. Everything from the previous twenty chapters appears here at once, and the result is a genuinely useful thing — a miniature Keymint.

What we are building

The first full project, and it deliberately exercises most of the track at once. Read the specification before writing anything — knowing what the finished thing guarantees makes the code far easier to follow.

   ks create <name>        generate a P-256 keypair inside the
                           TEE. The private half never exists
                           outside it.
   ks pubkey <name>        print the public key
   ks sign   <name> <file> sign a file's SHA-256 digest
   ks verify <name> <file> <sig>
   ks list                 list stored key names
   ks delete <name>

   PROPERTIES WE ARE ENFORCING:

     * the private key is generated by TEE_GenerateKey and
       stored as a persistent OBJECT (chapter 15). It is never
       serialised into a buffer.
     * asking for the private attribute returns
       TEE_ERROR_ACCESS_DENIED - structurally, not by policy.
     * keys are scoped to the calling user (LOGIN_USER), so two
       Linux users cannot see each other's keys.
     * the client passes a NAME, never a pointer or an index
       into TA state.
     * every rule from chapter 20 is applied.

   Verification is done with OpenSSL on the host, to prove the
   signatures are real ECDSA and not something only this TA
   understands.

/* Copy a client-supplied name into TA memory and validate it.
 * ONE read of shared memory (chapter 20, rule 2).
 */
static TEE_Result copy_name(const TEE_Param *p, char *out,
                            size_t outsz, size_t *outlen)
{
        size_t len = p->memref.size;

        if (!p->memref.buffer || len == 0 || len >= outsz)
                return TEE_ERROR_BAD_PARAMETERS;

        if (TEE_CheckMemoryAccessRights(
                    TEE_MEMORY_ACCESS_READ |
                    TEE_MEMORY_ACCESS_ANY_OWNER,
                    p->memref.buffer, len) != TEE_SUCCESS)
                return TEE_ERROR_ACCESS_DENIED;

        TEE_MemMove(out, p->memref.buffer, len);
        out[len] = '\0';

        /* Validate the COPY. Only printable, no separators -
         * the name becomes an object ID and appears in the
         * NUL-separated list output.
         */
        for (size_t i = 0; i < len; i++) {
                char c = out[i];
                bool ok = (c >= 'a' && c <= 'z') ||
                          (c >= 'A' && c <= 'Z') ||
                          (c >= '0' && c <= '9') ||
                          c == '-' || c == '_' || c == '.';
                if (!ok)
                        return TEE_ERROR_BAD_PARAMETERS;
        }

        *outlen = len;
        return TEE_SUCCESS;
}

static TEE_Result open_key(const char *name, size_t namelen,
                           uint32_t flags, TEE_ObjectHandle *obj)
{
        return TEE_OpenPersistentObject(TEE_STORAGE_PRIVATE,
                                        name, namelen,
                                        flags, obj);
}

/* ------------------------------------------------------------
 * KS_CMD_CREATE
 * ---------------------------------------------------------- */
static TEE_Result cmd_create(uint32_t ptypes, TEE_Param params[4])
{
        const uint32_t exp =
                TEE_PARAM_TYPES(TEE_PARAM_TYPE_MEMREF_INPUT,
                                TEE_PARAM_TYPE_NONE,
                                TEE_PARAM_TYPE_NONE,
                                TEE_PARAM_TYPE_NONE);
        char             name[KS_MAX_NAME_LEN];
        size_t           namelen;
        TEE_ObjectHandle key  = TEE_HANDLE_NULL;
        TEE_ObjectHandle pobj = TEE_HANDLE_NULL;
        TEE_Attribute    attr;
        TEE_Result       res;

        if (ptypes != exp) {
                EMSG("ptypes 0x%08x expected 0x%08x", ptypes, exp);
                return TEE_ERROR_BAD_PARAMETERS;
        }

        res = copy_name(&params[0], name, sizeof name, &namelen);
        if (res != TEE_SUCCESS)
                return res;

        /* Refuse to overwrite silently - a client that wants a
         * new key must delete the old one deliberately.
         */
        res = open_key(name, namelen, TEE_DATA_FLAG_ACCESS_READ,
                       &pobj);
        if (res == TEE_SUCCESS) {
                TEE_CloseObject(pobj);
                return TEE_ERROR_ACCESS_CONFLICT;
        }
        pobj = TEE_HANDLE_NULL;

        res = TEE_AllocateTransientObject(TEE_TYPE_ECDSA_KEYPAIR,
                                          256, &key);
        if (res != TEE_SUCCESS)
                goto out;

        TEE_InitValueAttribute(&attr, TEE_ATTR_ECC_CURVE,
                               TEE_ECC_CURVE_NIST_P256, 0);

        /* The private key is generated HERE, inside the TEE,
         * and never exists anywhere else.
         */
        res = TEE_GenerateKey(key, 256, &attr, 1);
        if (res != TEE_SUCCESS)
                goto out;

        /* Persist the OBJECT, not bytes (chapter 15). */
        res = TEE_CreatePersistentObject(
                TEE_STORAGE_PRIVATE, name, namelen,
                TEE_DATA_FLAG_ACCESS_READ |
                TEE_DATA_FLAG_ACCESS_WRITE_META,
                key, NULL, 0, &pobj);
        if (res == TEE_SUCCESS)
                IMSG("created key '%s'", name);
out:
        TEE_FreeTransientObject(key);
        if (pobj != TEE_HANDLE_NULL)
                TEE_CloseObject(pobj);
        return res;
}

/* ------------------------------------------------------------
 * KS_CMD_GET_PUBKEY
 * ---------------------------------------------------------- */
static TEE_Result cmd_get_pubkey(uint32_t ptypes, TEE_Param params[4])
{
        const uint32_t exp =
                TEE_PARAM_TYPES(TEE_PARAM_TYPE_MEMREF_INPUT,
                                TEE_PARAM_TYPE_MEMREF_OUTPUT,
                                TEE_PARAM_TYPE_NONE,
                                TEE_PARAM_TYPE_NONE);
        char             name[KS_MAX_NAME_LEN];
        size_t           namelen;
        TEE_ObjectHandle pobj = TEE_HANDLE_NULL;
        uint8_t          pub[KS_PUBKEY_LEN];
        uint32_t         xlen = 32, ylen = 32;
        TEE_Result       res;

        if (ptypes != exp)
                return TEE_ERROR_BAD_PARAMETERS;

        res = copy_name(&params[0], name, sizeof name, &namelen);
        if (res != TEE_SUCCESS)
                return res;

        res = open_key(name, namelen, TEE_DATA_FLAG_ACCESS_READ,
                       &pobj);
        if (res != TEE_SUCCESS)
                return res;      /* ITEM_NOT_FOUND if absent    */

        /* Both coordinates are 32 bytes for P-256, but
         * TEE_GetObjectBufferAttribute may return fewer if the
         * value has leading zeros. Zero-pad on the LEFT so the
         * output is always fixed width.
         */
        TEE_MemFill(pub, 0, sizeof pub);

        res = TEE_GetObjectBufferAttribute(pobj,
                        TEE_ATTR_ECC_PUBLIC_VALUE_X,
                        pub + (32 - xlen), &xlen);
        if (res != TEE_SUCCESS)
                goto out;

        res = TEE_GetObjectBufferAttribute(pobj,
                        TEE_ATTR_ECC_PUBLIC_VALUE_Y,
                        pub + 32 + (32 - ylen), &ylen);
        if (res != TEE_SUCCESS)
                goto out;

        /* Short-buffer protocol (chapter 7). */
        if (params[1].memref.size < sizeof pub) {
                params[1].memref.size = sizeof pub;
                res = TEE_ERROR_SHORT_BUFFER;
                goto out;
        }

        /* Write output ONCE, on success (chapter 20, rule 5). */
        TEE_MemMove(params[1].memref.buffer, pub, sizeof pub);
        params[1].memref.size = sizeof pub;
out:
        TEE_CloseObject(pobj);
        return res;
}

/* ------------------------------------------------------------
 * KS_CMD_SIGN
 * ---------------------------------------------------------- */
static TEE_Result cmd_sign(uint32_t ptypes, TEE_Param params[4])
{
        const uint32_t exp =
                TEE_PARAM_TYPES(TEE_PARAM_TYPE_MEMREF_INPUT,
                                TEE_PARAM_TYPE_MEMREF_INPUT,
                                TEE_PARAM_TYPE_MEMREF_OUTPUT,
                                TEE_PARAM_TYPE_NONE);
        char                name[KS_MAX_NAME_LEN];
        size_t              namelen;
        uint8_t             digest[KS_DIGEST_LEN];
        uint8_t             sig[KS_MAX_SIG_LEN];
        uint32_t            siglen = sizeof sig;
        TEE_ObjectHandle    pobj = TEE_HANDLE_NULL;
        TEE_OperationHandle op   = TEE_HANDLE_NULL;
        TEE_Result          res;

        if (ptypes != exp)
                return TEE_ERROR_BAD_PARAMETERS;

        res = copy_name(&params[0], name, sizeof name, &namelen);
        if (res != TEE_SUCCESS)
                return res;

        /* Fixed-size digest. Exactly 32 bytes or nothing -
         * no length arithmetic to get wrong.
         */
        if (params[1].memref.size != KS_DIGEST_LEN)
                return TEE_ERROR_BAD_PARAMETERS;
        if (TEE_CheckMemoryAccessRights(
                    TEE_MEMORY_ACCESS_READ |
                    TEE_MEMORY_ACCESS_ANY_OWNER,
                    params[1].memref.buffer,
                    KS_DIGEST_LEN) != TEE_SUCCESS)
                return TEE_ERROR_ACCESS_DENIED;

        TEE_MemMove(digest, params[1].memref.buffer,
                    KS_DIGEST_LEN);

        res = open_key(name, namelen, TEE_DATA_FLAG_ACCESS_READ,
                       &pobj);
        if (res != TEE_SUCCESS)
                return res;

        res = TEE_AllocateOperation(&op, TEE_ALG_ECDSA_P256,
                                    TEE_MODE_SIGN, 256);
        if (res != TEE_SUCCESS)
                goto out;

        res = TEE_SetOperationKey(op, pobj);
        if (res != TEE_SUCCESS)
                goto out;

        res = TEE_AsymmetricSignDigest(op, NULL, 0,
                                       digest, KS_DIGEST_LEN,
                                       sig, &siglen);
        if (res != TEE_SUCCESS)
                goto out;

        if (params[2].memref.size < siglen) {
                params[2].memref.size = siglen;
                res = TEE_ERROR_SHORT_BUFFER;
                goto out;
        }

        TEE_MemMove(params[2].memref.buffer, sig, siglen);
        params[2].memref.size = siglen;
out:
        TEE_FreeOperation(op);
        TEE_CloseObject(pobj);
        TEE_MemFill(digest, 0, sizeof digest);
        return res;
}

/* ------------------------------------------------------------
 * KS_CMD_VERIFY
 * ---------------------------------------------------------- */
static TEE_Result cmd_verify(uint32_t ptypes, TEE_Param params[4])
{
        const uint32_t exp =
                TEE_PARAM_TYPES(TEE_PARAM_TYPE_MEMREF_INPUT,
                                TEE_PARAM_TYPE_MEMREF_INPUT,
                                TEE_PARAM_TYPE_MEMREF_INPUT,
                                TEE_PARAM_TYPE_NONE);
        char                name[KS_MAX_NAME_LEN];
        size_t              namelen;
        uint8_t             digest[KS_DIGEST_LEN];
        uint8_t             sig[KS_MAX_SIG_LEN];
        size_t              siglen;
        TEE_ObjectHandle    pobj = TEE_HANDLE_NULL;
        TEE_OperationHandle op   = TEE_HANDLE_NULL;
        TEE_Result          res;

        if (ptypes != exp)
                return TEE_ERROR_BAD_PARAMETERS;

        res = copy_name(&params[0], name, sizeof name, &namelen);
        if (res != TEE_SUCCESS)
                return res;

        if (params[1].memref.size != KS_DIGEST_LEN)
                return TEE_ERROR_BAD_PARAMETERS;

        siglen = params[2].memref.size;
        if (siglen == 0 || siglen > sizeof sig)
                return TEE_ERROR_BAD_PARAMETERS;

        TEE_MemMove(digest, params[1].memref.buffer, KS_DIGEST_LEN);
        TEE_MemMove(sig, params[2].memref.buffer, siglen);

        res = open_key(name, namelen, TEE_DATA_FLAG_ACCESS_READ,
                       &pobj);
        if (res != TEE_SUCCESS)
                return res;

        res = TEE_AllocateOperation(&op, TEE_ALG_ECDSA_P256,
                                    TEE_MODE_VERIFY, 256);
        if (res != TEE_SUCCESS)
                goto out;

        res = TEE_SetOperationKey(op, pobj);
        if (res != TEE_SUCCESS)
                goto out;

        /* Returns TEE_ERROR_SIGNATURE_INVALID on mismatch -
         * pass it straight back; there is nothing secret about
         * "this signature is wrong".
         */
        res = TEE_AsymmetricVerifyDigest(op, NULL, 0,
                                         digest, KS_DIGEST_LEN,
                                         sig, siglen);
out:
        TEE_FreeOperation(op);
        TEE_CloseObject(pobj);
        return res;
}

/* ------------------------------------------------------------
 * KS_CMD_DELETE
 * ---------------------------------------------------------- */
static TEE_Result cmd_delete(uint32_t ptypes, TEE_Param params[4])
{
        const uint32_t exp =
                TEE_PARAM_TYPES(TEE_PARAM_TYPE_MEMREF_INPUT,
                                TEE_PARAM_TYPE_NONE,
                                TEE_PARAM_TYPE_NONE,
                                TEE_PARAM_TYPE_NONE);
        char             name[KS_MAX_NAME_LEN];
        size_t           namelen;
        TEE_ObjectHandle pobj = TEE_HANDLE_NULL;
        TEE_Result       res;

        if (ptypes != exp)
                return TEE_ERROR_BAD_PARAMETERS;

        res = copy_name(&params[0], name, sizeof name, &namelen);
        if (res != TEE_SUCCESS)
                return res;

        res = open_key(name, namelen,
                       TEE_DATA_FLAG_ACCESS_READ |
                       TEE_DATA_FLAG_ACCESS_WRITE_META, &pobj);
        if (res != TEE_SUCCESS)
                return res;

        /* Closes the handle as well - do not close it again. */
        return TEE_CloseAndDeletePersistentObject1(pobj);
}

/* ------------------------------------------------------------
 * KS_CMD_LIST
 * ---------------------------------------------------------- */
static TEE_Result cmd_list(uint32_t ptypes, TEE_Param params[4])
{
        const uint32_t exp =
                TEE_PARAM_TYPES(TEE_PARAM_TYPE_MEMREF_OUTPUT,
                                TEE_PARAM_TYPE_NONE,
                                TEE_PARAM_TYPE_NONE,
                                TEE_PARAM_TYPE_NONE);
        TEE_ObjectEnumHandle e = TEE_HANDLE_NULL;
        uint8_t  out[KS_MAX_KEYS * (KS_MAX_NAME_LEN + 1)];
        size_t   used = 0;
        TEE_Result res;

        if (ptypes != exp)
                return TEE_ERROR_BAD_PARAMETERS;

        res = TEE_AllocatePersistentObjectEnumerator(&e);
        if (res != TEE_SUCCESS)
                return res;

        res = TEE_StartPersistentObjectEnumerator(
                        e, TEE_STORAGE_PRIVATE);
        if (res == TEE_ERROR_ITEM_NOT_FOUND) {
                res = TEE_SUCCESS;      /* no keys yet - not an
                                           error                */
                goto emit;
        }
        if (res != TEE_SUCCESS)
                goto out;

        for (;;) {
                TEE_ObjectInfo info;
                uint8_t        id[KS_MAX_NAME_LEN];
                uint32_t       idlen = sizeof id;

                res = TEE_GetNextPersistentObject(e, &info,
                                                  id, &idlen);
                if (res == TEE_ERROR_ITEM_NOT_FOUND) {
                        res = TEE_SUCCESS;
                        break;
                }
                if (res != TEE_SUCCESS)
                        goto out;

                if (used + idlen + 1 > sizeof out) {
                        res = TEE_ERROR_SHORT_BUFFER;
                        goto out;
                }
                TEE_MemMove(out + used, id, idlen);
                used += idlen;
                out[used++] = '\0';
        }

emit:
        if (params[0].memref.size < used) {
                params[0].memref.size = used;
                res = TEE_ERROR_SHORT_BUFFER;
                goto out;
        }
        TEE_MemMove(params[0].memref.buffer, out, used);
        params[0].memref.size = used;
out:
        if (e != TEE_HANDLE_NULL)
                TEE_FreePersistentObjectEnumerator(e);
        return res;
}

TEE_Result TA_OpenSessionEntryPoint(uint32_t ptypes,
                                    TEE_Param params[4] __unused,
                                    void **sess_ctx)
{
        TEE_Identity id;
        TEE_Result   res;

        if (ptypes != TEE_PARAM_TYPES(TEE_PARAM_TYPE_NONE,
                                      TEE_PARAM_TYPE_NONE,
                                      TEE_PARAM_TYPE_NONE,
                                      TEE_PARAM_TYPE_NONE))
                return TEE_ERROR_BAD_PARAMETERS;

        /* Refuse anonymous callers. With LOGIN_USER, OP-TEE
         * scopes secure storage per uid, so each user gets a
         * separate key namespace (chapter 11).
         */
        res = TEE_GetPropertyAsIdentity(TEE_PROPSET_CURRENT_CLIENT,
                                        "gpd.client.identity", &id);
        if (res != TEE_SUCCESS)
                return res;

        if (id.login != TEE_LOGIN_USER &&
            id.login != TEE_LOGIN_REE_KERNEL) {
                EMSG("refusing login type 0x%08x", id.login);
                return TEE_ERROR_ACCESS_DENIED;
        }

        *sess_ctx = NULL;       /* no per-session state needed  */
        return TEE_SUCCESS;
}

void TA_CloseSessionEntryPoint(void *sess_ctx __unused)
{
}

TEE_Result TA_InvokeCommandEntryPoint(void *sess_ctx __unused,
                                      uint32_t cmd,
                                      uint32_t ptypes,
                                      TEE_Param params[4])
{
        switch (cmd) {
        case KS_CMD_CREATE:     return cmd_create(ptypes, params);
        case KS_CMD_GET_PUBKEY: return cmd_get_pubkey(ptypes, params);
        case KS_CMD_SIGN:       return cmd_sign(ptypes, params);
        case KS_CMD_VERIFY:     return cmd_verify(ptypes, params);
        case KS_CMD_DELETE:     return cmd_delete(ptypes, params);
        case KS_CMD_LIST:       return cmd_list(ptypes, params);
        default:                return TEE_ERROR_NOT_SUPPORTED;
        }
}

The client

The normal-world side, which drives the operations and never sees key material — which is the property the whole project exists to demonstrate.

/* host/main.c */

#include <err.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <openssl/sha.h>
#include <tee_client_api.h>
#include "keystore_ta.h"

static TEEC_Context ctx;
static TEEC_Session sess;

static const char *origin_str(uint32_t o)
{
        switch (o) {
        case TEEC_ORIGIN_API:         return "API";
        case TEEC_ORIGIN_COMMS:       return "COMMS";
        case TEEC_ORIGIN_TEE:         return "TEE";
        case TEEC_ORIGIN_TRUSTED_APP: return "TA";
        default:                      return "?";
        }
}

static void ks_open(void)
{
        TEEC_UUID uuid = KEYSTORE_TA_UUID;
        uint32_t  origin;
        TEEC_Result res;

        res = TEEC_InitializeContext(NULL, &ctx);
        if (res != TEEC_SUCCESS)
                errx(1, "InitializeContext: 0x%08x", res);

        /* LOGIN_USER: the kernel supplies our uid; the TA uses
         * it to scope storage. We cannot forge it.
         */
        res = TEEC_OpenSession(&ctx, &sess, &uuid,
                               TEEC_LOGIN_USER, NULL, NULL,
                               &origin);
        if (res != TEEC_SUCCESS)
                errx(1, "OpenSession: 0x%08x origin %s",
                     res, origin_str(origin));
}

static void ks_close(void)
{
        TEEC_CloseSession(&sess);
        TEEC_FinalizeContext(&ctx);
}

static TEEC_Result invoke(uint32_t cmd, TEEC_Operation *op)
{
        uint32_t origin;
        TEEC_Result res = TEEC_InvokeCommand(&sess, cmd, op,
                                             &origin);
        if (res != TEEC_SUCCESS && res != TEEC_ERROR_SHORT_BUFFER)
                warnx("cmd %u: 0x%08x origin %s", cmd, res,
                      origin_str(origin));
        return res;
}

/* ---- SHA-256 of a file, computed in the NORMAL world.
 *      The TA signs a digest, not a file: hashing is not
 *      security-sensitive here and keeping bulk data out of
 *      the TEE is the whole point (chapter 1).            */
static int sha256_file(const char *path, uint8_t out[32])
{
        SHA256_CTX c;
        uint8_t    buf[65536];
        size_t     n;
        FILE      *f = fopen(path, "rb");

        if (!f) { warn("%s", path); return -1; }
        SHA256_Init(&c);
        while ((n = fread(buf, 1, sizeof buf, f)) > 0)
                SHA256_Update(&c, buf, n);
        fclose(f);
        SHA256_Final(out, &c);
        return 0;
}

static void hexdump(const uint8_t *p, size_t n)
{
        for (size_t i = 0; i < n; i++)
                printf("%02x", p[i]);
        printf("\n");
}

/* ---- commands ------------------------------------------- */

static int do_create(const char *name)
{
        TEEC_Operation op;

        memset(&op, 0, sizeof op);
        op.paramTypes = TEEC_PARAM_TYPES(TEEC_MEMREF_TEMP_INPUT,
                                         TEEC_NONE, TEEC_NONE,
                                         TEEC_NONE);
        op.params[0].tmpref.buffer = (void *)name;
        op.params[0].tmpref.size   = strlen(name);

        if (invoke(KS_CMD_CREATE, &op) != TEEC_SUCCESS)
                return 1;
        printf("created '%s'\n", name);
        return 0;
}

static int do_pubkey(const char *name)
{
        TEEC_Operation op;
        uint8_t pub[KS_PUBKEY_LEN];

        memset(&op, 0, sizeof op);
        op.paramTypes = TEEC_PARAM_TYPES(TEEC_MEMREF_TEMP_INPUT,
                                         TEEC_MEMREF_TEMP_OUTPUT,
                                         TEEC_NONE, TEEC_NONE);
        op.params[0].tmpref.buffer = (void *)name;
        op.params[0].tmpref.size   = strlen(name);
        op.params[1].tmpref.buffer = pub;
        op.params[1].tmpref.size   = sizeof pub;

        if (invoke(KS_CMD_GET_PUBKEY, &op) != TEEC_SUCCESS)
                return 1;

        /* Uncompressed SEC1 point, so OpenSSL can read it. */
        printf("04");
        hexdump(pub, op.params[1].tmpref.size);
        return 0;
}

static int do_sign(const char *name, const char *path)
{
        TEEC_Operation op;
        uint8_t digest[KS_DIGEST_LEN], sig[KS_MAX_SIG_LEN];

        if (sha256_file(path, digest))
                return 1;

        memset(&op, 0, sizeof op);
        op.paramTypes = TEEC_PARAM_TYPES(TEEC_MEMREF_TEMP_INPUT,
                                         TEEC_MEMREF_TEMP_INPUT,
                                         TEEC_MEMREF_TEMP_OUTPUT,
                                         TEEC_NONE);
        op.params[0].tmpref.buffer = (void *)name;
        op.params[0].tmpref.size   = strlen(name);
        op.params[1].tmpref.buffer = digest;
        op.params[1].tmpref.size   = sizeof digest;
        op.params[2].tmpref.buffer = sig;
        op.params[2].tmpref.size   = sizeof sig;

        if (invoke(KS_CMD_SIGN, &op) != TEEC_SUCCESS)
                return 1;

        hexdump(sig, op.params[2].tmpref.size);
        return 0;
}

static int hex2bin(const char *hex, uint8_t *out, size_t outsz,
                   size_t *outlen)
{
        size_t n = strlen(hex);

        if (n % 2 || n / 2 > outsz)
                return -1;
        for (size_t i = 0; i < n / 2; i++)
                if (sscanf(hex + 2 * i, "%2hhx", &out[i]) != 1)
                        return -1;
        *outlen = n / 2;
        return 0;
}

static int do_verify(const char *name, const char *path,
                     const char *sighex)
{
        TEEC_Operation op;
        uint8_t digest[KS_DIGEST_LEN], sig[KS_MAX_SIG_LEN];
        size_t  siglen;
        TEEC_Result res;

        if (sha256_file(path, digest))
                return 1;
        if (hex2bin(sighex, sig, sizeof sig, &siglen)) {
                warnx("bad signature hex");
                return 1;
        }

        memset(&op, 0, sizeof op);
        op.paramTypes = TEEC_PARAM_TYPES(TEEC_MEMREF_TEMP_INPUT,
                                         TEEC_MEMREF_TEMP_INPUT,
                                         TEEC_MEMREF_TEMP_INPUT,
                                         TEEC_NONE);
        op.params[0].tmpref.buffer = (void *)name;
        op.params[0].tmpref.size   = strlen(name);
        op.params[1].tmpref.buffer = digest;
        op.params[1].tmpref.size   = sizeof digest;
        op.params[2].tmpref.buffer = sig;
        op.params[2].tmpref.size   = siglen;

        res = invoke(KS_CMD_VERIFY, &op);
        printf("%s\n", res == TEEC_SUCCESS ? "VALID" : "INVALID");
        return res == TEEC_SUCCESS ? 0 : 1;
}

static int do_list(void)
{
        TEEC_Operation op;
        char buf[KS_MAX_KEYS * (KS_MAX_NAME_LEN + 1)];
        size_t off = 0, n;

        memset(&op, 0, sizeof op);
        op.paramTypes = TEEC_PARAM_TYPES(TEEC_MEMREF_TEMP_OUTPUT,
                                         TEEC_NONE, TEEC_NONE,
                                         TEEC_NONE);
        op.params[0].tmpref.buffer = buf;
        op.params[0].tmpref.size   = sizeof buf;

        if (invoke(KS_CMD_LIST, &op) != TEEC_SUCCESS)
                return 1;

        n = op.params[0].tmpref.size;
        while (off < n) {
                printf("%s\n", buf + off);
                off += strlen(buf + off) + 1;
        }
        return 0;
}

static int do_delete(const char *name)
{
        TEEC_Operation op;

        memset(&op, 0, sizeof op);
        op.paramTypes = TEEC_PARAM_TYPES(TEEC_MEMREF_TEMP_INPUT,
                                         TEEC_NONE, TEEC_NONE,
                                         TEEC_NONE);
        op.params[0].tmpref.buffer = (void *)name;
        op.params[0].tmpref.size   = strlen(name);

        if (invoke(KS_CMD_DELETE, &op) != TEEC_SUCCESS)
                return 1;
        printf("deleted '%s'\n", name);
        return 0;
}

int main(int argc, char **argv)
{
        int rc;

        if (argc < 2) {
                fprintf(stderr,
                        "usage: ks create|pubkey|delete <name>\n"
                        "       ks sign   <name> <file>\n"
                        "       ks verify <name> <file> <sighex>\n"
                        "       ks list\n");
                return 2;
        }

        ks_open();

        if      (!strcmp(argv[1], "create") && argc == 3)
                rc = do_create(argv[2]);
        else if (!strcmp(argv[1], "pubkey") && argc == 3)
                rc = do_pubkey(argv[2]);
        else if (!strcmp(argv[1], "delete") && argc == 3)
                rc = do_delete(argv[2]);
        else if (!strcmp(argv[1], "sign")   && argc == 4)
                rc = do_sign(argv[2], argv[3]);
        else if (!strcmp(argv[1], "verify") && argc == 5)
                rc = do_verify(argv[2], argv[3], argv[4]);
        else if (!strcmp(argv[1], "list")   && argc == 2)
                rc = do_list();
        else {
                warnx("bad usage");
                rc = 2;
        }

        ks_close();
        return rc;
}

Proving the signature is real

A security property you cannot demonstrate from the outside is a hope rather than a property. These are the checks that the key really is unreachable — and if any of them fails, the project is decoration.

# The most satisfying part: verify with OpenSSL, on the host,
# with no OP-TEE involved at all. If this works, the TA is
# producing genuine ECDSA P-256 signatures.

# 1. Get the public key and turn it into a PEM.
$ PUB=$(ssh guest ks pubkey device-key)
$ echo $PUB | xxd -r -p > pub.raw          # 65 bytes: 04||X||Y

# Wrap it in the SubjectPublicKeyInfo DER prefix for
# prime256v1:
$ printf '3059301306072a8648ce3d020106082a8648ce3d030107034200' \
    | xxd -r -p > spki.der
$ cat spki.der pub.raw > pubkey.der
$ openssl ec -pubin -inform DER -in pubkey.der -out pubkey.pem
read EC key
writing EC key

$ openssl ec -pubin -in pubkey.pem -text -noout
Public-Key: (256 bit)
pub: 04:a1:e5:...
ASN1 OID: prime256v1
NIST CURVE: P-256

# 2. The TA emits raw r||s; OpenSSL wants DER. Convert.
$ SIG=$(ssh guest ks sign device-key /tmp/msg)
$ python3 - "$SIG" <<'EOF' > sig.der
import sys, binascii
raw = binascii.unhexlify(sys.argv[1])
r, s = raw[:32], raw[32:]
def der_int(b):
    b = b.lstrip(b'\x00') or b'\x00'
    if b[0] & 0x80:
        b = b'\x00' + b
    return b'\x02' + bytes([len(b)]) + b
body = der_int(r) + der_int(s)
sys.stdout.buffer.write(b'\x30' + bytes([len(body)]) + body)
EOF

# 3. Verify.
$ printf 'hello world\n' > msg
$ openssl dgst -sha256 -verify pubkey.pem -signature sig.der msg
Verified OK

# THAT is the point of the project. A key that was generated
# inside the secure world, that has never existed in normal-
# world memory, that survives reboots, and whose signatures any
# standard tool can verify.
Raw r||s versus DER

GlobalPlatform's TEE_AsymmetricSignDigest returns the two ECDSA components concatenated and fixed-width. Almost every other ecosystem — OpenSSL, X.509, JWS with ES256 uses raw, but TLS and CMS use DER — expects an ASN.1 SEQUENCE of two INTEGERs. Converting is trivial and forgetting to is the commonest reason a TA's signatures "do not verify" when they are perfectly correct.
