Project: a DX12 triangle, every line explained

The smallest complete DX12 program. Roughly 400 lines, and by the end of this chapter every one of them should be a decision you understand rather than a line you copied — which is the entire difference between using DX12 and fighting it.

The order of operations

   STARTUP
     1.  enable the debug layer      BEFORE the device
     2.  create the DXGI factory
     3.  pick an adapter             not adapter 0
     4.  create the device
     5.  configure the info queue and DRED
     6.  create the command queue
     7.  create the swap chain       needs the QUEUE, not the
                                     device
     8.  create descriptor heaps     RTV heap for backbuffers
     9.  create RTVs for each backbuffer
     10. create command allocators   one per frame in flight
     11. create the command list
     12. create the fence and event
     13. compile shaders
     14. create the root signature
     15. create the PSO
     16. create the vertex buffer + upload it
     17. WAIT for the upload to finish

   PER FRAME
     18. wait for this frame slot's fence
     19. reset allocator, then list
     20. transition backbuffer PRESENT -> RENDER_TARGET
     21. set state, clear, draw
     22. transition RENDER_TARGET -> PRESENT
     23. close, execute, present, signal

   SHUTDOWN
     24. flush - wait for the GPU to be completely idle
     25. release, in any order (ComPtr does it)

   Step 24 is not optional: destroying resources the GPU is
   still using is a page fault or a hang, and it happens on
   exit where nobody looks.

The shader

// triangle.hlsl - compile both entry points:
//   dxc -T vs_6_0 -E VSMain -Fo vs.cso triangle.hlsl
//   dxc -T ps_6_0 -E PSMain -Fo ps.cso triangle.hlsl
//
// dxc, not fxc: fxc only emits DXBC (shader model 5.1 and
// below). Anything using wave intrinsics or SM 6 features
// needs dxc.

struct VSIn  { float3 pos : POSITION; float4 col : COLOR; };
struct VSOut { float4 pos : SV_Position; float4 col : COLOR; };

VSOut VSMain(VSIn i)
{
    VSOut o;
    o.pos = float4(i.pos, 1.0);   // already in clip space -
                                  // no matrices in a first
                                  // triangle
    o.col = i.col;
    return o;
}

float4 PSMain(VSOut i) : SV_Target
{
    return i.col;                 // interpolated across the
                                  // triangle by the
                                  // rasteriser, for free
}

Initialisation

#include <d3d12.h>
#include <dxgi1_6.h>
#include <d3dcompiler.h>
#include <wrl/client.h>
#include <stdexcept>
#include <vector>
using Microsoft::WRL::ComPtr;

#pragma comment(lib, "d3d12.lib")
#pragma comment(lib, "dxgi.lib")

static const UINT kFrames = 2;

struct Vertex { float pos[3]; float col[4]; };

class TriangleApp {
    ComPtr<ID3D12Device>              device_;
    ComPtr<ID3D12CommandQueue>        queue_;
    ComPtr<IDXGISwapChain3>           swap_;
    ComPtr<ID3D12DescriptorHeap>      rtvHeap_;
    ComPtr<ID3D12Resource>            backBuf_[kFrames];
    ComPtr<ID3D12CommandAllocator>    alloc_[kFrames];
    ComPtr<ID3D12GraphicsCommandList> cmd_;
    ComPtr<ID3D12RootSignature>       rootSig_;
    ComPtr<ID3D12PipelineState>       pso_;
    ComPtr<ID3D12Resource>            vb_;
    D3D12_VERTEX_BUFFER_VIEW          vbv_{};
    ComPtr<ID3D12Fence>               fence_;
    HANDLE   fenceEvent_ = nullptr;
    UINT64   fenceVal_[kFrames] = {};
    UINT64   nextFence_ = 1;
    UINT     frame_ = 0, rtvSize_ = 0;
    UINT     width_ = 1280, height_ = 720;

public:
void Init(HWND hwnd)
{
    /* 1. DEBUG LAYER FIRST. After device creation it silently
          does nothing. */
#if defined(_DEBUG)
    ComPtr<ID3D12Debug> dbg;
    if (SUCCEEDED(D3D12GetDebugInterface(IID_PPV_ARGS(&dbg))))
        dbg->EnableDebugLayer();
    UINT factoryFlags = DXGI_CREATE_FACTORY_DEBUG;
#else
    UINT factoryFlags = 0;
#endif

    /* 2-4. factory, adapter, device (chapter 12). */
    ComPtr<IDXGIFactory6> factory;
    CreateDXGIFactory2(factoryFlags, IID_PPV_ARGS(&factory));

    ComPtr<IDXGIAdapter1> adapter;
    factory->EnumAdapterByGpuPreference(
        0, DXGI_GPU_PREFERENCE_HIGH_PERFORMANCE,
        IID_PPV_ARGS(&adapter));

    if (FAILED(D3D12CreateDevice(adapter.Get(),
                                 D3D_FEATURE_LEVEL_11_0,
                                 IID_PPV_ARGS(&device_))))
        throw std::runtime_error("D3D12CreateDevice");

    /* 6. The queue. The swap chain needs it, not the device -
          because presentation is itself queued work. */
    D3D12_COMMAND_QUEUE_DESC qd = {};
    qd.Type = D3D12_COMMAND_LIST_TYPE_DIRECT;
    device_->CreateCommandQueue(&qd, IID_PPV_ARGS(&queue_));

    /* 7. Swap chain. FLIP_DISCARD is mandatory for DX12 and
          is also the only mode that allows tearing control
          and true independent flip. */
    DXGI_SWAP_CHAIN_DESC1 sd = {};
    sd.BufferCount = kFrames;
    sd.Width       = width_;
    sd.Height      = height_;
    sd.Format      = DXGI_FORMAT_R8G8B8A8_UNORM;
    sd.BufferUsage = DXGI_USAGE_RENDER_TARGET_OUTPUT;
    sd.SwapEffect  = DXGI_SWAP_EFFECT_FLIP_DISCARD;
    sd.SampleDesc.Count = 1;          /* MSAA is NOT allowed on
                                         a flip-model swap
                                         chain - render to your
                                         own MSAA target and
                                         resolve into this   */
    ComPtr<IDXGISwapChain1> sc1;
    factory->CreateSwapChainForHwnd(queue_.Get(), hwnd, &sd,
                                    nullptr, nullptr, &sc1);
    factory->MakeWindowAssociation(hwnd, DXGI_MWA_NO_ALT_ENTER);
    sc1.As(&swap_);
    frame_ = swap_->GetCurrentBackBufferIndex();

    /* 8-9. RTV heap and one view per backbuffer. */
    D3D12_DESCRIPTOR_HEAP_DESC hd = {};
    hd.NumDescriptors = kFrames;
    hd.Type  = D3D12_DESCRIPTOR_HEAP_TYPE_RTV;
    hd.Flags = D3D12_DESCRIPTOR_HEAP_FLAG_NONE;   /* CPU-only */
    device_->CreateDescriptorHeap(&hd, IID_PPV_ARGS(&rtvHeap_));

    /* QUERY the increment - do not assume (chapter 14). */
    rtvSize_ = device_->GetDescriptorHandleIncrementSize(
                   D3D12_DESCRIPTOR_HEAP_TYPE_RTV);

    auto rtv = rtvHeap_->GetCPUDescriptorHandleForHeapStart();
    for (UINT i = 0; i < kFrames; i++) {
        swap_->GetBuffer(i, IID_PPV_ARGS(&backBuf_[i]));
        device_->CreateRenderTargetView(backBuf_[i].Get(),
                                        nullptr, rtv);
        rtv.ptr += rtvSize_;
        device_->CreateCommandAllocator(
            D3D12_COMMAND_LIST_TYPE_DIRECT,
            IID_PPV_ARGS(&alloc_[i]));       /* 10 */
    }

    /* 12. Fence. */
    device_->CreateFence(0, D3D12_FENCE_FLAG_NONE,
                         IID_PPV_ARGS(&fence_));
    fenceEvent_ = CreateEvent(nullptr, FALSE, FALSE, nullptr);

    CreatePipeline();
    CreateVertexBuffer();
}
void CreatePipeline()
{
    /* 14. An EMPTY root signature - this shader takes no
           parameters at all. Still required: a PSO must have
           one. */
    D3D12_ROOT_SIGNATURE_DESC rs = {};
    rs.Flags = D3D12_ROOT_SIGNATURE_FLAG_ALLOW_INPUT_ASSEMBLER_INPUT_LAYOUT;

    ComPtr<ID3DBlob> sig, err;
    if (FAILED(D3D12SerializeRootSignature(
            &rs, D3D_ROOT_SIGNATURE_VERSION_1, &sig, &err)))
        throw std::runtime_error((char *)err->GetBufferPointer());

    device_->CreateRootSignature(0, sig->GetBufferPointer(),
                                 sig->GetBufferSize(),
                                 IID_PPV_ARGS(&rootSig_));

    ComPtr<ID3DBlob> vs = LoadBlob(L"vs.cso");
    ComPtr<ID3DBlob> ps = LoadBlob(L"ps.cso");

    /* The semantic NAMES must match the HLSL exactly, and the
       OFFSETS must match the C++ struct exactly. Both are
       silent when wrong (chapter 7). */
    D3D12_INPUT_ELEMENT_DESC layout[] = {
      { "POSITION", 0, DXGI_FORMAT_R32G32B32_FLOAT,    0,  0,
        D3D12_INPUT_CLASSIFICATION_PER_VERTEX_DATA, 0 },
      { "COLOR",    0, DXGI_FORMAT_R32G32B32A32_FLOAT, 0, 12,
        D3D12_INPUT_CLASSIFICATION_PER_VERTEX_DATA, 0 },
    };

    D3D12_GRAPHICS_PIPELINE_STATE_DESC p = {};
    p.pRootSignature = rootSig_.Get();
    p.VS = { vs->GetBufferPointer(), vs->GetBufferSize() };
    p.PS = { ps->GetBufferPointer(), ps->GetBufferSize() };
    p.InputLayout = { layout, _countof(layout) };

    p.RasterizerState.FillMode = D3D12_FILL_MODE_SOLID;
    p.RasterizerState.CullMode = D3D12_CULL_MODE_NONE;
        /* NONE for a first triangle: with BACK, a winding
           order you did not think about gives you an empty
           screen and no error. Turn culling on once you can
           see something. */
    p.RasterizerState.DepthClipEnable = TRUE;

    p.BlendState.RenderTarget[0].RenderTargetWriteMask =
        D3D12_COLOR_WRITE_ENABLE_ALL;
        /* zero-initialised means BlendEnable = FALSE, which is
           what we want - but the WRITE MASK also defaults to
           0, which writes nothing. Set it. */

    p.DepthStencilState.DepthEnable   = FALSE;
    p.DepthStencilState.StencilEnable = FALSE;

    p.PrimitiveTopologyType = D3D12_PRIMITIVE_TOPOLOGY_TYPE_TRIANGLE;
    p.NumRenderTargets = 1;
    p.RTVFormats[0]    = DXGI_FORMAT_R8G8B8A8_UNORM;
        /* MUST match the swap chain format exactly, or PSO
           creation fails. */
    p.SampleDesc.Count = 1;
    p.SampleMask       = UINT_MAX;
        /* THE CLASSIC BLACK SCREEN. Defaults to 0, which
           masks out every sample. */

    if (FAILED(device_->CreateGraphicsPipelineState(
            &p, IID_PPV_ARGS(&pso_))))
        throw std::runtime_error("CreateGraphicsPipelineState");

    /* 11. The list is created OPEN, on allocator 0. Close it
           so the frame loop can Reset symmetrically. */
    device_->CreateCommandList(0, D3D12_COMMAND_LIST_TYPE_DIRECT,
                               alloc_[0].Get(), pso_.Get(),
                               IID_PPV_ARGS(&cmd_));
    cmd_->Close();
}
void CreateVertexBuffer()
{
    Vertex verts[] = {
        { {  0.0f,  0.75f, 0.0f }, { 1, 0, 0, 1 } },
        { {  0.75f,-0.75f, 0.0f }, { 0, 1, 0, 1 } },
        { { -0.75f,-0.75f, 0.0f }, { 0, 0, 1, 1 } },
    };
    const UINT size = sizeof verts;

    /* SHORTCUT, and it is worth naming as one: this puts the
       vertex buffer in an UPLOAD heap so we can memcpy
       directly, with no staging buffer and no copy command.
       For three vertices that is fine. For real geometry it
       is wrong - upload memory is slower for the GPU to read,
       and you want DEFAULT plus a copy (chapter 14). */
    D3D12_HEAP_PROPERTIES hp = { D3D12_HEAP_TYPE_UPLOAD };
    D3D12_RESOURCE_DESC rd = {};
    rd.Dimension        = D3D12_RESOURCE_DIMENSION_BUFFER;
    rd.Width            = size;
    rd.Height           = 1;
    rd.DepthOrArraySize = 1;
    rd.MipLevels        = 1;
    rd.Format           = DXGI_FORMAT_UNKNOWN;
    rd.SampleDesc.Count = 1;
    rd.Layout           = D3D12_TEXTURE_LAYOUT_ROW_MAJOR;

    device_->CreateCommittedResource(
        &hp, D3D12_HEAP_FLAG_NONE, &rd,
        D3D12_RESOURCE_STATE_GENERIC_READ,
            /* UPLOAD heap resources must be created in
               GENERIC_READ and can never leave it. */
        nullptr, IID_PPV_ARGS(&vb_));

    void *p = nullptr;
    D3D12_RANGE readNothing = { 0, 0 };
    vb_->Map(0, &readNothing, &p);
    memcpy(p, verts, size);          /* WRITE ONLY. Never read
                                        from upload memory. */
    vb_->Unmap(0, nullptr);

    vbv_.BufferLocation = vb_->GetGPUVirtualAddress();
    vbv_.StrideInBytes  = sizeof(Vertex);
    vbv_.SizeInBytes    = size;
        /* Stride must equal the C++ struct size and match the
           input layout offsets. A mismatch renders garbage
           silently. */
}

The frame

void Render()
{
    /* 18. Do not touch this allocator until the GPU has
           finished the last frame that used it. */
    WaitForFence(fenceVal_[frame_]);

    /* 19. Allocator first (frees its memory), then the list
           (begins recording into it). */
    alloc_[frame_]->Reset();
    cmd_->Reset(alloc_[frame_].Get(), pso_.Get());

    cmd_->SetGraphicsRootSignature(rootSig_.Get());

    D3D12_VIEWPORT vp = { 0, 0, (float)width_, (float)height_,
                          0.0f, 1.0f };
    D3D12_RECT     sc = { 0, 0, (LONG)width_, (LONG)height_ };
    cmd_->RSSetViewports(1, &vp);
    cmd_->RSSetScissorRects(1, &sc);
        /* A scissor of {0,0,0,0} clips everything away. It is
           the second most common blank-screen cause after
           SampleMask, and neither produces an error. */

    /* 20. The backbuffer arrives in PRESENT state. */
    Transition(backBuf_[frame_].Get(),
               D3D12_RESOURCE_STATE_PRESENT,
               D3D12_RESOURCE_STATE_RENDER_TARGET);

    D3D12_CPU_DESCRIPTOR_HANDLE rtv =
        rtvHeap_->GetCPUDescriptorHandleForHeapStart();
    rtv.ptr += SIZE_T(frame_) * rtvSize_;

    cmd_->OMSetRenderTargets(1, &rtv, FALSE, nullptr);

    const float clear[4] = { 0.05f, 0.06f, 0.09f, 1.0f };
    cmd_->ClearRenderTargetView(rtv, clear, 0, nullptr);

    /* 21. Three lines to draw a triangle. */
    cmd_->IASetPrimitiveTopology(D3D_PRIMITIVE_TOPOLOGY_TRIANGLELIST);
        /* The PSO holds the TYPE (triangle); the list holds
           LIST vs STRIP. Both are required. */
    cmd_->IASetVertexBuffers(0, 1, &vbv_);
    cmd_->DrawInstanced(3, 1, 0, 0);

    /* 22. Back to PRESENT, or Present() fails validation. */
    Transition(backBuf_[frame_].Get(),
               D3D12_RESOURCE_STATE_RENDER_TARGET,
               D3D12_RESOURCE_STATE_PRESENT);

    /* 23. Close, submit, present, signal. */
    cmd_->Close();
    ID3D12CommandList *lists[] = { cmd_.Get() };
    queue_->ExecuteCommandLists(1, lists);

    swap_->Present(1, 0);        /* 1 = vsync                  */

    fenceVal_[frame_] = nextFence_;
    queue_->Signal(fence_.Get(), nextFence_++);

    frame_ = swap_->GetCurrentBackBufferIndex();
        /* ASK the swap chain, do not assume (frame+1)%N.
           With FLIP_DISCARD the order is not guaranteed. */
}

void WaitForFence(UINT64 v)
{
    if (v == 0 || fence_->GetCompletedValue() >= v) return;
    fence_->SetEventOnCompletion(v, fenceEvent_);
    WaitForSingleObject(fenceEvent_, INFINITE);
}

void Transition(ID3D12Resource *r, D3D12_RESOURCE_STATES from,
                D3D12_RESOURCE_STATES to)
{
    D3D12_RESOURCE_BARRIER b = {};
    b.Type = D3D12_RESOURCE_BARRIER_TYPE_TRANSITION;
    b.Transition.pResource   = r;
    b.Transition.StateBefore = from;
    b.Transition.StateAfter  = to;
    b.Transition.Subresource = D3D12_RESOURCE_BARRIER_ALL_SUBRESOURCES;
    cmd_->ResourceBarrier(1, &b);
}

/* 24. SHUTDOWN. Wait for everything, then let ComPtr free. */
void Shutdown()
{
    queue_->Signal(fence_.Get(), nextFence_);
    WaitForFence(nextFence_);
    CloseHandle(fenceEvent_);
}
};   /* class TriangleApp */
The four blank-screen causes, in order of frequency

1. SampleMask left at 0. 2. Scissor rect not set, so it is {0,0,0,0}. 3. RenderTargetWriteMask left at 0. 4. Backface culling with the winding order you did not think about. None of the four produces an error message. If you get a clear colour but no triangle, work down that list before anything else.
