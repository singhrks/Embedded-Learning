Project: layers 1 and 2 — the application and the kernel driver

The normal-world half of the stack: a shared header that is a permanent promise, an unprivileged application, and a Linux driver that owns the TEE session and is the only thing in the system permitted to talk to the trusted application. Every line of both, plus the build.

Seam A: the header that can never change

Start here, because this is the part you can never take back. Once a program in the field includes this header, its layout and its numbers are fixed forever — which is the Linux track's ABI rule applied to your own interface.

/* uapi/secsign_uapi.h  —  SEAM A
 *
 * Shared by the application (layer 1) and the driver
 * (layer 2). PERMANENT: additions only, at the end, never
 * a change to anything already here.
 */
#ifndef _UAPI_SECSIGN_H
#define _UAPI_SECSIGN_H

#include <linux/types.h>
#include <linux/ioctl.h>

#define SECSIGN_MAX_DATA   (1u << 20)   /* 1 MiB  */
#define SECSIGN_SIG_LEN    64           /* P-256 r||s */
#define SECSIGN_PUBKEY_LEN 64           /* P-256 x||y */

struct secsign_sign {
        __u64 data;          /* IN  - user pointer to the message */
        __u32 data_len;      /* IN  - bytes, <= SECSIGN_MAX_DATA  */
        __u32 _pad;          /* MUST be zero - reserved so the
                                struct can grow without changing
                                its size (chapter 21 of the Linux
                                track)                            */
        __u8  sig[SECSIGN_SIG_LEN];   /* OUT */
};

struct secsign_pubkey {
        __u8  key[SECSIGN_PUBKEY_LEN]; /* OUT */
};

#define SECSIGN_IOC_MAGIC  'S'
#define SECSIGN_IOC_SIGN   _IOWR(SECSIGN_IOC_MAGIC, 1, struct secsign_sign)
#define SECSIGN_IOC_PUBKEY _IOR (SECSIGN_IOC_MAGIC, 2, struct secsign_pubkey)

#endif /* _UAPI_SECSIGN_H */
Three decisions in that header worth naming

__u64 data rather than a pointer, so the structure has the same layout for 32-bit and 64-bit callers and no compatibility handler is ever needed. An explicit _pad that must be zero, so a later version can use those bytes for a flag while old binaries — which leave it zero — keep working. And fixed-size output arrays rather than a second pointer, so the signature comes back in the same copy that carried the request, removing a whole second round of pointer validation.

Layer 1: the application

Deliberately ordinary. It has no special privileges, knows nothing about the TEE, and would work unchanged if the signing were done by a completely different mechanism — which is what a good seam looks like from above.

/* cli/main.c  —  LAYER 1 */
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <fcntl.h>
#include <unistd.h>
#include <errno.h>
#include <sys/ioctl.h>
#include <sys/stat.h>

#include "secsign_uapi.h"

#define DEV "/dev/secsign"

static int do_pubkey(int fd)
{
        struct secsign_pubkey pk;

        memset(&pk, 0, sizeof(pk));
        if (ioctl(fd, SECSIGN_IOC_PUBKEY, &pk) < 0) {
                perror("SECSIGN_IOC_PUBKEY");
                return 1;
        }

        for (unsigned i = 0; i < SECSIGN_PUBKEY_LEN; i++)
                printf("%02x", pk.key[i]);
        putchar('\n');
        return 0;
}

static int do_sign(int fd, const char *path)
{
        struct secsign_sign req;
        struct stat st;
        unsigned char *buf = NULL;
        char *outpath = NULL;
        int in = -1, out = -1, ret = 1;
        ssize_t n;

        in = open(path, O_RDONLY);
        if (in < 0) { perror(path); goto done; }

        if (fstat(in, &st) < 0) { perror("fstat"); goto done; }

        if (st.st_size <= 0 || (size_t)st.st_size > SECSIGN_MAX_DATA) {
                fprintf(stderr, "%s: must be 1..%u bytes\n",
                        path, SECSIGN_MAX_DATA);
                goto done;
        }

        buf = malloc((size_t)st.st_size);
        if (!buf) { perror("malloc"); goto done; }

        /* read() can return short - loop. (C track, chapter 16) */
        for (size_t got = 0; got < (size_t)st.st_size; got += (size_t)n) {
                n = read(in, buf + got, (size_t)st.st_size - got);
                if (n < 0) { perror("read"); goto done; }
                if (n == 0) { fprintf(stderr, "short read\n"); goto done; }
        }

        memset(&req, 0, sizeof(req));      /* zeroes _pad too */
        req.data     = (__u64)(uintptr_t)buf;
        req.data_len = (__u32)st.st_size;

        if (ioctl(fd, SECSIGN_IOC_SIGN, &req) < 0) {
                perror("SECSIGN_IOC_SIGN");
                goto done;
        }

        if (asprintf(&outpath, "%s.sig", path) < 0) {
                outpath = NULL; goto done;
        }
        out = open(outpath, O_WRONLY | O_CREAT | O_TRUNC, 0644);
        if (out < 0) { perror(outpath); goto done; }

        if (write(out, req.sig, SECSIGN_SIG_LEN) != SECSIGN_SIG_LEN) {
                perror("write"); goto done;
        }

        printf("wrote %s (%d bytes)\n", outpath, SECSIGN_SIG_LEN);
        ret = 0;

done:
        if (out >= 0) close(out);
        if (in  >= 0) close(in);
        free(outpath);
        free(buf);
        return ret;
}

int main(int argc, char **argv)
{
        int fd, ret;

        if (argc < 2) {
                fprintf(stderr,
                        "usage: %s sign <file>\n"
                        "       %s pubkey\n", argv[0], argv[0]);
                return 2;
        }

        fd = open(DEV, O_RDWR);
        if (fd < 0) {
                fprintf(stderr, "%s: %s\n", DEV, strerror(errno));
                return 1;
        }

        if (!strcmp(argv[1], "pubkey"))
                ret = do_pubkey(fd);
        else if (!strcmp(argv[1], "sign") && argc == 3)
                ret = do_sign(fd, argv[2]);
        else { fprintf(stderr, "unknown command\n"); ret = 2; }

        close(fd);
        return ret;
}

Seam B: the header shared with the trusted application

The second contract, between layers 2 and 3. It is separate from the UAPI on purpose — the application has no business knowing the TA's UUID or command numbering, and keeping them apart means the TA interface can be revised without touching anything user space depends on.

/* ta/include/secsign_ta.h  —  SEAM B
 * Included by the kernel driver AND by the TA.
 */
#ifndef SECSIGN_TA_H
#define SECSIGN_TA_H

/* Generate your own with uuidgen; this one is an example. */
#define SECSIGN_TA_UUID \
        { 0x7a1b3c4d, 0x5e6f, 0x4718, \
          { 0x9a, 0x2b, 0xc3, 0xd4, 0xe5, 0xf6, 0x07, 0x18 } }

#define SECSIGN_CMD_SIGN    0
#define SECSIGN_CMD_PUBKEY  1

#define SECSIGN_TA_SIG_LEN     64
#define SECSIGN_TA_PUBKEY_LEN  64
#define SECSIGN_TA_MAX_DATA    (1u << 20)

#endif /* SECSIGN_TA_H */

Layer 2: the Linux kernel driver

The driver does three jobs: expose a device node, enforce policy, and own exactly one TEE session that it serialises every caller through. It is the first place the request is validated and the last place it exists in a form the normal world fully controls.

// SPDX-License-Identifier: GPL-2.0
/* kmod/secsign_drv.c  —  LAYER 2
 *
 * A misc device that forwards signing requests to a trusted
 * application over the in-kernel TEE client interface.
 */
#include <linux/module.h>
#include <linux/miscdevice.h>
#include <linux/fs.h>
#include <linux/slab.h>
#include <linux/uaccess.h>
#include <linux/mutex.h>
#include <linux/tee_drv.h>
#include <linux/uuid.h>

#include "secsign_uapi.h"
#include "../ta/include/secsign_ta.h"

struct secsign_dev {
        struct tee_context *ctx;
        u32                 session;
        struct mutex        lock;   /* serialises the single
                                       session across callers */
        struct miscdevice   misc;
};

static struct secsign_dev *sdev;

/* ---- session management -------------------------------- */

static int secsign_match(struct tee_ioctl_version_data *ver,
                         const void *data)
{
        /* Only OP-TEE, and only the kernel-side ABI. */
        return ver->impl_id == TEE_IMPL_ID_OPTEE &&
               ver->gen_caps & TEE_GEN_CAP_GP;
}

static int secsign_open_session(struct secsign_dev *d)
{
        struct tee_ioctl_open_session_arg arg = { };
        const uuid_t uuid = UUID_INIT(0x7a1b3c4d, 0x5e6f, 0x4718,
                                      0x9a, 0x2b, 0xc3, 0xd4,
                                      0xe5, 0xf6, 0x07, 0x18);
        int ret;

        d->ctx = tee_client_open_context(NULL, secsign_match,
                                         NULL, NULL);
        if (IS_ERR(d->ctx))
                return PTR_ERR(d->ctx);

        export_uuid(arg.uuid, &uuid);
        arg.clnt_login = TEE_IOCTL_LOGIN_REE_KERNEL;
        arg.num_params = 0;

        ret = tee_client_open_session(d->ctx, &arg, NULL);
        if (ret < 0 || arg.ret) {
                tee_client_close_context(d->ctx);
                d->ctx = NULL;
                return ret ? ret : -EINVAL;
        }

        d->session = arg.session;
        return 0;
}

/* TEE_IOCTL_LOGIN_REE_KERNEL is what lets the TA tell a
   request from the KERNEL apart from one from an arbitrary
   user process (chapter 11). The TA can, and in this project
   does, refuse anything else.                              */

/* ---- the one real operation ---------------------------- */

static int secsign_do_sign(struct secsign_dev *d,
                           const void *data, u32 len,
                           u8 *sig_out)
{
        struct tee_ioctl_invoke_arg arg = { };
        struct tee_param param[2] = { };
        struct tee_shm *shm_in, *shm_out;
        void *in_va, *out_va;
        int ret;

        /* Shared memory is NORMAL-world memory the secure
           world can see. Allocate, fill, invoke, free.     */
        shm_in = tee_shm_alloc_kernel_buf(d->ctx, len);
        if (IS_ERR(shm_in))
                return PTR_ERR(shm_in);

        shm_out = tee_shm_alloc_kernel_buf(d->ctx,
                                           SECSIGN_TA_SIG_LEN);
        if (IS_ERR(shm_out)) {
                ret = PTR_ERR(shm_out);
                goto out_in;
        }

        in_va = tee_shm_get_va(shm_in, 0);
        if (IS_ERR(in_va)) { ret = PTR_ERR(in_va); goto out_both; }
        memcpy(in_va, data, len);

        arg.func       = SECSIGN_CMD_SIGN;
        arg.session    = d->session;
        arg.num_params = 2;

        param[0].attr = TEE_IOCTL_PARAM_ATTR_TYPE_MEMREF_INPUT;
        param[0].u.memref.shm      = shm_in;
        param[0].u.memref.size     = len;
        param[0].u.memref.shm_offs = 0;

        param[1].attr = TEE_IOCTL_PARAM_ATTR_TYPE_MEMREF_OUTPUT;
        param[1].u.memref.shm      = shm_out;
        param[1].u.memref.size     = SECSIGN_TA_SIG_LEN;
        param[1].u.memref.shm_offs = 0;

        ret = tee_client_invoke_func(d->ctx, &arg, param);
        if (ret < 0)
                goto out_both;
        if (arg.ret) {
                /* arg.ret_origin says WHICH LAYER failed -
                   far more useful than the code (ch. 08). */
                pr_err("secsign: TA failed 0x%x origin %u\n",
                       arg.ret, arg.ret_origin);
                ret = -EIO;
                goto out_both;
        }
        if (param[1].u.memref.size != SECSIGN_TA_SIG_LEN) {
                ret = -EPROTO;      /* TA returned a bad length */
                goto out_both;
        }

        out_va = tee_shm_get_va(shm_out, 0);
        if (IS_ERR(out_va)) { ret = PTR_ERR(out_va); goto out_both; }
        memcpy(sig_out, out_va, SECSIGN_TA_SIG_LEN);
        ret = 0;

out_both:
        tee_shm_free(shm_out);
out_in:
        tee_shm_free(shm_in);
        return ret;
}

/* ---- ioctl --------------------------------------------- */

static long secsign_ioctl(struct file *f, unsigned int cmd,
                          unsigned long uarg)
{
        void __user *uptr = (void __user *)uarg;
        long ret;

        switch (cmd) {
        case SECSIGN_IOC_SIGN: {
                struct secsign_sign req;
                void *kbuf;

                if (copy_from_user(&req, uptr, sizeof(req)))
                        return -EFAULT;

                /* VALIDATE EVERYTHING. req came from user
                   space and every field is attacker-chosen. */
                if (req._pad)                       /* reserved */
                        return -EINVAL;
                if (req.data_len == 0 ||
                    req.data_len > SECSIGN_MAX_DATA)
                        return -EINVAL;

                kbuf = kvmalloc(req.data_len, GFP_KERNEL);
                if (!kbuf)
                        return -ENOMEM;

                /* COPY, then use. The user buffer can change
                   after this point and it no longer matters. */
                if (copy_from_user(kbuf,
                                   (void __user *)(uintptr_t)req.data,
                                   req.data_len)) {
                        ret = -EFAULT;
                        goto sign_out;
                }

                mutex_lock(&sdev->lock);
                ret = secsign_do_sign(sdev, kbuf, req.data_len,
                                      req.sig);
                mutex_unlock(&sdev->lock);
                if (ret)
                        goto sign_out;

                /* Copy the whole struct back: the signature
                   rides home in the same object.            */
                if (copy_to_user(uptr, &req, sizeof(req)))
                        ret = -EFAULT;

sign_out:
                kvfree(kbuf);
                return ret;
        }

        case SECSIGN_IOC_PUBKEY: {
                struct secsign_pubkey pk;
                struct tee_ioctl_invoke_arg arg = { };
                struct tee_param param[1] = { };
                struct tee_shm *shm;
                void *va;

                memset(&pk, 0, sizeof(pk));

                shm = tee_shm_alloc_kernel_buf(sdev->ctx,
                                SECSIGN_TA_PUBKEY_LEN);
                if (IS_ERR(shm))
                        return PTR_ERR(shm);

                arg.func       = SECSIGN_CMD_PUBKEY;
                arg.session    = sdev->session;
                arg.num_params = 1;

                param[0].attr =
                        TEE_IOCTL_PARAM_ATTR_TYPE_MEMREF_OUTPUT;
                param[0].u.memref.shm      = shm;
                param[0].u.memref.size     = SECSIGN_TA_PUBKEY_LEN;
                param[0].u.memref.shm_offs = 0;

                mutex_lock(&sdev->lock);
                ret = tee_client_invoke_func(sdev->ctx, &arg, param);
                mutex_unlock(&sdev->lock);

                if (ret < 0 || arg.ret) {
                        tee_shm_free(shm);
                        return ret ? ret : -EIO;
                }

                va = tee_shm_get_va(shm, 0);
                if (!IS_ERR(va))
                        memcpy(pk.key, va, SECSIGN_TA_PUBKEY_LEN);
                tee_shm_free(shm);

                if (IS_ERR(va))
                        return PTR_ERR(va);

                return copy_to_user(uptr, &pk, sizeof(pk))
                        ? -EFAULT : 0;
        }

        default:
                return -ENOTTY;   /* the correct errno for an
                                     unknown ioctl (ch. 23 of
                                     the Linux track)        */
        }
}

static const struct file_operations secsign_fops = {
        .owner          = THIS_MODULE,
        .unlocked_ioctl = secsign_ioctl,
        .compat_ioctl   = compat_ptr_ioctl,  /* the struct is
                                                layout-identical,
                                                so this is enough */
        .llseek         = no_llseek,
};

/* ---- module init/exit ---------------------------------- */

static int __init secsign_init(void)
{
        int ret;

        sdev = kzalloc(sizeof(*sdev), GFP_KERNEL);
        if (!sdev)
                return -ENOMEM;

        mutex_init(&sdev->lock);

        ret = secsign_open_session(sdev);
        if (ret) {
                pr_err("secsign: no TEE session (%d)\n", ret);
                goto err_free;
        }

        sdev->misc.minor = MISC_DYNAMIC_MINOR;
        sdev->misc.name  = "secsign";
        sdev->misc.fops  = &secsign_fops;
        sdev->misc.mode  = 0600;      /* POLICY: root only.
                                         Loosen deliberately,
                                         with a udev rule.  */

        ret = misc_register(&sdev->misc);
        if (ret)
                goto err_session;

        pr_info("secsign: ready\n");
        return 0;

err_session:
        tee_client_close_session(sdev->ctx, sdev->session);
        tee_client_close_context(sdev->ctx);
err_free:
        kfree(sdev);
        sdev = NULL;
        return ret;
}

static void __exit secsign_exit(void)
{
        misc_deregister(&sdev->misc);
        tee_client_close_session(sdev->ctx, sdev->session);
        tee_client_close_context(sdev->ctx);
        kfree(sdev);
        sdev = NULL;
}

module_init(secsign_init);
module_exit(secsign_exit);

MODULE_LICENSE("GPL");
MODULE_DESCRIPTION("TEE-backed signing service");
Opening the session at module init is a deliberate simplification

It makes the driver fail loudly and early if the TA is missing, which is exactly what you want while developing. It is the wrong choice for production: the TEE driver may not have probed yet when your module loads, so a real driver either defers (returning -EPROBE_DEFER from a platform driver probe, Linux chapter 19) or opens the session lazily on first use and retries. Opening it lazily also means a TA crash can be recovered from by reopening, rather than requiring a module reload.

Building and loading

Two build systems, because the module is built against the running kernel's headers and the CLI is not. The only thing they share is the UAPI header, which is why it sits outside both trees.

# kmod/Makefile
obj-m += secsign.o
secsign-objs := secsign_drv.o
ccflags-y += -I$(src)/../uapi

KDIR ?= /lib/modules/$(shell uname -r)/build

all:
	$(MAKE) -C $(KDIR) M=$(PWD) modules
clean:
	$(MAKE) -C $(KDIR) M=$(PWD) clean

# Cross-compiling for the arm64 target you will actually run:
#   make -C $(KDIR) M=$(PWD) ARCH=arm64 \
#        CROSS_COMPILE=aarch64-linux-gnu- modules

# cli/Makefile
CFLAGS  += -O2 -Wall -Wextra -I../uapi -D_GNU_SOURCE
secsign-cli: main.c
	$(CC) $(CFLAGS) -o $@ $<
# On the target, in the order that tells you where a failure is:

# 1. Is there a TEE at all? (chapter 04)
$ ls /dev/tee*
/dev/tee0  /dev/teepriv0

# 2. Is tee-supplicant running? Storage will fail without it.
$ pgrep tee-supplicant || echo "NOT RUNNING"

# 3. Load the driver. This is where a missing or unsigned TA
#    shows up, loudly.
$ sudo insmod secsign.ko
$ dmesg | tail -3
secsign: ready

# If instead you see:
#   secsign: no TEE session (-22)
# the TA was not found or failed to load - go back to
# chapter 28 and check the UUID and the signature.

$ ls -l /dev/secsign
crw------- 1 root root 10, 123 ... /dev/secsign

# 4. Exercise it.
$ echo -n "hello world" > /tmp/msg
$ sudo ./secsign-cli sign /tmp/msg
wrote /tmp/msg.sig (64 bytes)
$ sudo ./secsign-cli pubkey
04a1b2c3...

How it shows up

SymptomCause
insmod fails, "no TEE session"The TA is missing, unsigned, or its UUID does not match the one in the driver
ioctl returns -ENOTTYWrong command number — usually a header mismatch between the app and the driver
ioctl returns -EFAULTA user pointer used directly rather than via copy_from_user
ioctl returns -EINVAL on a valid file_pad not zeroed — the application skipped the memset
Works for root, EACCES otherwiseThe 0600 mode; change it deliberately rather than by reflex
Concurrent callers serialise and feel slowExpected — one session, one mutex. Scaling that is a design change, not a tuning one
32-bit application, garbage resultsWould happen if the struct used a pointer instead of __u64 — which is why it does not
-EPROTO from the driverThe TA returned a signature of the wrong length; trust the check, not the TA
