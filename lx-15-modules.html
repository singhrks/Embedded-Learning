Kernel modules

A module is kernel code you can load and unload while the system runs. That single property changes development from "reboot to test" into "reload in a second", which is worth more to your productivity than any other kernel skill.

The smallest complete module

A module is just an object file the kernel can load at runtime, with two functions marked as its entry and exit points. Here is the whole thing — it is genuinely this small, and everything else is built on this shape.

/* hello.c */
#include <linux/module.h>
#include <linux/kernel.h>
#include <linux/init.h>

static int __init hello_init(void)
{
        pr_info("hello: loaded\n");
        return 0;              /* non-zero aborts the load */
}

static void __exit hello_exit(void)
{
        pr_info("hello: unloaded\n");
}

module_init(hello_init);
module_exit(hello_exit);

MODULE_LICENSE("GPL");         /* REQUIRED. See below.      */
MODULE_AUTHOR("Your Name");
MODULE_DESCRIPTION("A minimal example module");
MODULE_VERSION("1.0");
# Makefile
obj-m += hello.o

KDIR ?= /lib/modules/$(shell uname -r)/build

all:
	$(MAKE) -C $(KDIR) M=$(PWD) modules

clean:
	$(MAKE) -C $(KDIR) M=$(PWD) clean

# CROSS-COMPILING for an ARM64 target - what you will
# actually do:
#   make -C /path/to/kernel/source M=$(PWD) \
#        ARCH=arm64 CROSS_COMPILE=aarch64-linux-gnu- modules
#
# KDIR must point at a CONFIGURED kernel tree that matches the
# target EXACTLY - same version, same .config. A module built
# against a different configuration will refuse to load, or
# load and corrupt memory.
# Using it:
    sudo insmod hello.ko              # load this exact file
    lsmod | grep hello                # is it there?
    dmesg | tail                      # what did it print?
    sudo rmmod hello                  # unload
    modinfo hello.ko                  # metadata, parameters,
                                      # dependencies, vermagic

# modprobe, the better tool - resolves DEPENDENCIES:
    sudo cp hello.ko /lib/modules/$(uname -r)/extra/
    sudo depmod -a                    # rebuild the dep map
    sudo modprobe hello               # loads what it needs
    sudo modprobe -r hello            # unload, with deps

MODULE_LICENSE is not paperwork

The license string is not a formality or a legal disclaimer. The kernel reads it and enforces it: declaring a non-GPL license removes your access to a large part of the kernel's own API, and the failure it produces at build time confuses people who assume the line is decorative.

/* It has a mechanical effect: symbols exported with
   EXPORT_SYMBOL_GPL are only linkable from a module whose
   license string is GPL-compatible.                        */

   "GPL"                  GPL v2 or later
   "GPL v2"
   "GPL and additional rights"
   "Dual BSD/GPL"
   "Dual MIT/GPL"
   "Proprietary"          <-- TAINTS the kernel, and cannot
                              use GPL-only symbols

/* Omit it entirely and you get:

     hello: module license 'unspecified' taints kernel.
     Disabling lock debugging due to kernel taint

   A tainted kernel is reported in every oops, and many kernel
   developers will decline to debug one. More practically:
   most of the API you need for a real driver is
   EXPORT_SYMBOL_GPL, so a "Proprietary" module cannot call
   it. This is a deliberate design decision, not an
   oversight.                                               */

/* CHECK the taint state: */
        cat /proc/sys/kernel/tainted
        # or read the flags in an oops header:
        #   "CPU: 0 PID: 431 Tainted: G           O"
        #    O = an out-of-tree module is loaded
        #    P = a proprietary module
        #    E = an unsigned module

Module parameters

Module parameters let you pass values in at load time, and they show up as files under /sys so they can be inspected afterwards. They are useful for debugging switches — and a poor choice for describing hardware, for reasons that matter more than they first appear.

static int buffer_size = 4096;
module_param(buffer_size, int, 0644);
MODULE_PARM_DESC(buffer_size, "Ring buffer size in bytes");

static char *device_name = "mydev";
module_param(device_name, charp, 0444);

static bool verbose;
module_param(verbose, bool, 0644);

static int channels[4];
static int nr_channels;
module_param_array(channels, int, &nr_channels, 0444);

/* SET THEM: */
        insmod mydrv.ko buffer_size=8192 verbose=1 \
               channels=1,2,3
        modprobe mydrv buffer_size=8192
        # built into the kernel, on the command line:
        mydrv.buffer_size=8192
        # persistently:
        echo "options mydrv buffer_size=8192" \
            > /etc/modprobe.d/mydrv.conf

/* AND READ OR CHANGE AT RUNTIME, if the mode allows: */
        cat /sys/module/mydrv/parameters/buffer_size
        echo 1 > /sys/module/mydrv/parameters/verbose

/* THE PERMISSION ARGUMENT is the sysfs mode:
        0        not visible in sysfs at all
        0444     read-only
        0644     writable by root

   Only use 0644 if changing the value at runtime is actually
   SAFE. A buffer size that is read once at probe time should
   be 0444 - otherwise you invite someone to change it and
   wonder why nothing happened.

/* AND THE BIGGER POINT: module parameters are a poor
   configuration mechanism for a real driver. They are global,
   not per-device. For anything describing HARDWARE, use the
   device tree (chapter 20). Parameters are for debug knobs
   and for behaviour that genuinely applies to the whole
   driver.                                                  */

Exporting and using symbols

By default a module's functions are private to it. To let another module call into yours, you have to export the symbol explicitly — and which of the two export macros you use is another decision the kernel enforces rather than merely records.

/* In one module: */
int my_shared_function(int arg)
{
        ...
}
EXPORT_SYMBOL_GPL(my_shared_function);

/* In another - just call it, but the FIRST module must be
   loaded first. modprobe works this out from the symbol
   table; insmod does not.                                  */

        depmod -a
        modprobe consumer          # loads provider first

/* SEEING what a module needs and provides: */
        modinfo mydrv.ko | grep depends
        cat /proc/kallsyms | grep my_shared_function
        nm -u mydrv.ko             # undefined = required
        nm -g --defined-only mydrv.ko

/* WHEN A SYMBOL IS MISSING: */
        insmod: ERROR: could not insert module: Unknown symbol
                in module
        dmesg:
          mydrv: Unknown symbol my_shared_function (err -2)
          mydrv: Unknown symbol dma_alloc_attrs (err 0)
                                                    ^^^^^^
        # err -2  = the symbol does not exist at all
        # err 0   = it exists but is GPL-only and your license
        #           string does not qualify

/* VERSION MAGIC - the other common load failure: */
        insmod: ERROR: could not insert module: Invalid module
                format
        dmesg:
          mydrv: version magic '6.6.0 SMP mod_unload aarch64'
          should be '6.6.0-vendor SMP preempt mod_unload aarch64'

   Every difference matters: version, SMP, PREEMPT, and the
   ABI-relevant config options. Rebuild against the target's
   exact tree. With CONFIG_MODVERSIONS the CRC of each
   exported function's signature is checked too, so a changed
   struct definition is caught rather than silently
   miscompiled.                                             */

Making a module safely unloadable

Unload is where modules crash

Loading is easy. Unloading requires that nothing can still call into your code after module_exit returns, because the memory holding your functions is freed. Everything you registered must be unregistered, and everything asynchronous must be waited for.

static void __exit my_exit(void)
{
        /* IN REVERSE ORDER OF INITIALISATION, and stopping
           sources BEFORE cancelling consumers:            */

        /* 1. stop new work arriving */
        platform_driver_unregister(&my_driver);
        misc_deregister(&my_misc);
        unregister_chrdev_region(devno, 1);

        /* 2. stop the hardware */
        writel(0, base + REG_INT_ENABLE);
        free_irq(irq, d);

        /* 3. WAIT for anything already in flight */
        cancel_delayed_work_sync(&d->poll_work);
        cancel_work_sync(&d->work);
        del_timer_sync(&d->timer);
        hrtimer_cancel(&d->hrt);
        kthread_stop(d->task);
        synchronize_rcu();              /* if you used RCU */

        /* 4. only now, free */
        kfree(d->buffer);
        iounmap(base);
}

/* THE _sync SUFFIX IS NOT OPTIONAL. cancel_work() returns
   immediately; cancel_work_sync() waits for a running
   instance to finish. The first one gives you a crash a few
   microseconds after rmmod, in code that no longer exists.

/* REFERENCE COUNTING - preventing unload while in use:

   For file operations, .owner = THIS_MODULE in the
   file_operations struct does it automatically: the module
   count is incremented while any fd is open, and rmmod
   returns -EBUSY.

        static const struct file_operations my_fops = {
                .owner = THIS_MODULE,     <-- do not omit
                .open  = my_open,
                ...
        };

   For anything else:
        try_module_get(THIS_MODULE);
        module_put(THIS_MODULE);

   CHECK who is using you:
        lsmod
        Module      Size  Used by
        mydrv      16384  2 othermod,thirdmod
                          ^ refcount   ^ by whom

   "Used by 0" and rmmod still fails -> something took a
   reference without registering a name, or a device is still
   bound.                                                   */

Module versus built-in

The Kconfig tristate choice between m and y is not merely a packaging detail — it decides whether your driver can even be present when the root filesystem is first mounted, and whether the unload path you wrote is ever actually exercised before it matters in production.

=m (module)=y (built in)
Development speedFast — reload in a secondSlow — rebuild and reboot
Available atWhenever loadedBoot, at its initcall level
Needed for the root filesystemRequires an initramfsWorks directly
UnloadableYes — exercises your teardown pathNo
MemoryOnly when loadedAlways
OrderingBy load order and dependenciesBy initcall level, then link order
 
/* THE ADVICE: develop as a module, ship however you need.

   Building as a module during development is not just about
   iteration speed. It forces you to write a correct remove()
   path - and a driver whose teardown is correct is a driver
   that also survives unbind, suspend, and probe failure.

   TEST THE CYCLE, repeatedly:
        for i in $(seq 100); do
                insmod mydrv.ko && rmmod mydrv || break
        done
   with KASAN enabled. Anything that leaks or races shows up
   within a few iterations, and this is by far the cheapest
   bug-finding exercise available to you.

   ALSO TEST UNBIND/BIND, which exercises probe and remove
   without touching the module:
        echo 1c000000.my-device > \
             /sys/bus/platform/drivers/my-driver/unbind
        echo 1c000000.my-device > \
             /sys/bus/platform/drivers/my-driver/bind      */

Module signing

A loadable module runs with full kernel privileges, so being able to load one is equivalent to owning the machine. On a system with secure boot the kernel therefore refuses to load anything that is not signed by a key it trusts.

# A hardened kernel refuses unsigned modules (isolation
# ch. 16 - the chain of trust does not stop at the kernel):

    CONFIG_MODULE_SIG=y
    CONFIG_MODULE_SIG_FORCE=y       # REFUSE unsigned
    CONFIG_MODULE_SIG_ALL=y         # sign during the build
    CONFIG_MODULE_SIG_HASH="sha256"

# The build signs with certs/signing_key.pem. For a product,
# supply your own key and keep the private half in an HSM:
    CONFIG_MODULE_SIG_KEY="/path/to/key.pem"

# Sign an out-of-tree module by hand:
    scripts/sign-file sha256 signing_key.pem \
                      signing_key.x509 mydrv.ko

# SYMPTOM of an unsigned module on an enforcing kernel:
    insmod: ERROR: could not insert module: Key was rejected
            by service
    dmesg: Loading of unsigned module is rejected

# Without SIG_FORCE it loads but taints with 'E'.
