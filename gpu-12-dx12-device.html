DX12: adapters, devices and the debug layer

Everything starts with picking a GPU and creating a device. It is thirty lines, and every one of them is a decision that will bite you later if you copy it from a tutorial without reading it — starting with which adapter you got and whether validation is on.

The object graph

   IDXGIFactory                 enumerates adapters, creates
        |                       swap chains
        +-- IDXGIAdapter        one GPU (or the software one)
             |
             +-- ID3D12Device   THE ROOT. Everything is created
                  |             from it. Not tied to a window.
                  |
                  +-- ID3D12CommandQueue      where work is
                  |    |                      submitted
                  |    +-- IDXGISwapChain     presentation
                  |
                  +-- ID3D12CommandAllocator  backing memory
                  |    +-- ID3D12GraphicsCommandList
                  |
                  +-- ID3D12DescriptorHeap    (chapter 14)
                  +-- ID3D12Resource          buffers, textures
                  +-- ID3D12RootSignature     (chapter 15)
                  +-- ID3D12PipelineState     (chapter 15)
                  +-- ID3D12Fence             (chapter 16)

   NOTE what the device is NOT: it has no state. There is no
   "current shader" or "current texture" on it. All state
   lives in a command list, and is reset every time you reset
   the list. That statelessness is what makes multi-threaded
   recording possible.

Choosing an adapter

#include <d3d12.h>
#include <dxgi1_6.h>
#include <wrl/client.h>
using Microsoft::WRL::ComPtr;

#pragma comment(lib, "d3d12.lib")
#pragma comment(lib, "dxgi.lib")

static void ThrowIfFailed(HRESULT hr, const char *what)
{
    if (FAILED(hr))
        throw std::runtime_error(std::string(what) + " failed: 0x"
                                 + std::to_string((unsigned)hr));
}

ComPtr<IDXGIAdapter1> PickAdapter(IDXGIFactory6 *factory)
{
    ComPtr<IDXGIAdapter1> adapter;

    /* EnumAdapterByGpuPreference sorts for you. Without it you
       get adapter 0, which on a laptop is very often the
       INTEGRATED GPU - and then you spend an afternoon
       wondering why performance is a third of expected. */
    for (UINT i = 0;
         factory->EnumAdapterByGpuPreference(
             i, DXGI_GPU_PREFERENCE_HIGH_PERFORMANCE,
             IID_PPV_ARGS(&adapter)) != DXGI_ERROR_NOT_FOUND;
         i++)
    {
        DXGI_ADAPTER_DESC1 desc;
        adapter->GetDesc1(&desc);

        if (desc.Flags & DXGI_ADAPTER_FLAG_SOFTWARE)
            continue;              /* WARP - useful, but not
                                      what you want by default */

        /* Probe WITHOUT creating: pass nullptr for the device
           and it just tells you whether it would succeed. */
        if (SUCCEEDED(D3D12CreateDevice(adapter.Get(),
                                        D3D_FEATURE_LEVEL_12_0,
                                        _uuidof(ID3D12Device),
                                        nullptr)))
        {
            wprintf(L"using %s (%zu MB)\n", desc.Description,
                    desc.DedicatedVideoMemory / (1024 * 1024));
            return adapter;
        }
    }
    throw std::runtime_error("no D3D12-capable adapter");
}
Feature levels are not the whole story

D3D_FEATURE_LEVEL_12_0 is a coarse baseline. The things you actually care about — resource binding tier, shader model, raytracing, mesh shaders, wave intrinsics — are queried separately with CheckFeatureSupport. A device can report feature level 12_1 and still not support the shader model your HLSL was compiled for, which fails at PSO creation with a message that does not mention shader models.

/* The capabilities worth checking at startup, before you
   depend on them. */

D3D12_FEATURE_DATA_D3D12_OPTIONS opts = {};
device->CheckFeatureSupport(D3D12_FEATURE_D3D12_OPTIONS,
                            &opts, sizeof opts);
/* opts.ResourceBindingTier decides how many descriptors you
   may use per table - Tier 1 is genuinely restrictive, Tier 3
   is effectively unlimited (chapter 14). */

D3D12_FEATURE_DATA_SHADER_MODEL sm = { D3D_SHADER_MODEL_6_6 };
if (FAILED(device->CheckFeatureSupport(D3D12_FEATURE_SHADER_MODEL,
                                       &sm, sizeof sm)))
        sm.HighestShaderModel = D3D_SHADER_MODEL_5_1;
/* NOTE the in/out convention: you pass the HIGHEST you want,
   and it writes back the highest supported. */

D3D12_FEATURE_DATA_D3D12_OPTIONS5 o5 = {};
device->CheckFeatureSupport(D3D12_FEATURE_D3D12_OPTIONS5,
                            &o5, sizeof o5);
bool raytracing = o5.RaytracingTier >=
                  D3D12_RAYTRACING_TIER_1_0;

D3D12_FEATURE_DATA_D3D12_OPTIONS7 o7 = {};
device->CheckFeatureSupport(D3D12_FEATURE_D3D12_OPTIONS7,
                            &o7, sizeof o7);
bool meshShaders = o7.MeshShaderTier != D3D12_MESH_SHADER_TIER_NOT_SUPPORTED;

/* Descriptor sizes are DEVICE-DEPENDENT and must be queried.
   Hardcoding 32 works on your machine and corrupts on
   someone else's (chapter 14). */
UINT rtvSize = device->GetDescriptorHandleIncrementSize(
                   D3D12_DESCRIPTOR_HEAP_TYPE_RTV);
UINT cbvSize = device->GetDescriptorHandleIncrementSize(
                   D3D12_DESCRIPTOR_HEAP_TYPE_CBV_SRV_UAV);

The debug layer, properly configured

/* MUST be enabled BEFORE the device is created. Enabling it
   afterwards silently does nothing, which is a common and
   frustrating waste of an afternoon. */

void EnableDebugLayer()
{
#if defined(_DEBUG)
    ComPtr<ID3D12Debug1> debug;
    if (SUCCEEDED(D3D12GetDebugInterface(IID_PPV_ARGS(&debug))))
    {
        debug->EnableDebugLayer();

        /* GPU-based validation instruments your SHADERS to
           check descriptor and resource-state correctness at
           execution time. It is 10-100x slower and it finds
           an entire class of bug the CPU-side layer cannot -
           notably reading a descriptor that was never
           written. Run it periodically, not always. */
        debug->SetEnableGPUBasedValidation(TRUE);
    }
#endif
}

/* After device creation: break in the debugger AT the error,
   rather than discovering it later. */
void ConfigureInfoQueue(ID3D12Device *device)
{
#if defined(_DEBUG)
    ComPtr<ID3D12InfoQueue> iq;
    if (FAILED(device->QueryInterface(IID_PPV_ARGS(&iq))))
        return;

    iq->SetBreakOnSeverity(D3D12_MESSAGE_SEVERITY_CORRUPTION, TRUE);
    iq->SetBreakOnSeverity(D3D12_MESSAGE_SEVERITY_ERROR,      TRUE);
    iq->SetBreakOnSeverity(D3D12_MESSAGE_SEVERITY_WARNING,    TRUE);

    /* Silence the handful of warnings that are genuinely
       noise, and NOTHING ELSE. Every suppression you add is a
       bug you have chosen not to see. */
    D3D12_MESSAGE_ID deny[] = {
        D3D12_MESSAGE_ID_CLEARRENDERTARGETVIEW_MISMATCHINGCLEARVALUE,
        D3D12_MESSAGE_ID_CLEARDEPTHSTENCILVIEW_MISMATCHINGCLEARVALUE,
    };
    D3D12_INFO_QUEUE_FILTER filter = {};
    filter.DenyList.NumIDs  = _countof(deny);
    filter.DenyList.pIDList = deny;
    iq->PushStorageFilter(&filter);
#endif
}
Device removal, and how to find out why

DXGI_ERROR_DEVICE_REMOVED is DX12's segfault: the GPU hit something it could not recover from — a page fault, a timeout, a bad command. By itself it tells you nothing. Enable ID3D12DeviceRemovedExtendedData (DRED) before device creation and you get the last executed command and the auto-breadcrumbs leading to it. Without DRED you are guessing; with it you usually know within a minute.

/* DRED - set this up at startup, always, including in
   release builds. It costs almost nothing and it is the
   difference between a diagnosable crash and a mystery. */

ComPtr<ID3D12DeviceRemovedExtendedDataSettings> dred;
if (SUCCEEDED(D3D12GetDebugInterface(IID_PPV_ARGS(&dred)))) {
    dred->SetAutoBreadcrumbsEnablement(D3D12_DRED_ENABLEMENT_FORCED_ON);
    dred->SetPageFaultEnablement(D3D12_DRED_ENABLEMENT_FORCED_ON);
}

/* Then, when Present or a fence wait returns DEVICE_REMOVED: */
void ReportDeviceRemoval(ID3D12Device *device)
{
    HRESULT reason = device->GetDeviceRemovedReason();
    /* DXGI_ERROR_DEVICE_HUNG    - your commands hung the GPU
       DXGI_ERROR_DEVICE_RESET   - TDR: took longer than ~2 s
       DXGI_ERROR_DRIVER_INTERNAL_ERROR - driver bug, or yours
       DXGI_ERROR_INVALID_CALL   - you did something illegal */

    ComPtr<ID3D12DeviceRemovedExtendedData> data;
    if (FAILED(device->QueryInterface(IID_PPV_ARGS(&data))))
        return;

    D3D12_DRED_AUTO_BREADCRUMBS_OUTPUT crumbs = {};
    if (SUCCEEDED(data->GetAutoBreadcrumbsOutput(&crumbs)))
        for (auto *n = crumbs.pHeadAutoBreadcrumbNode; n;
             n = n->pNext)
            wprintf(L"cmdlist %s: %u of %u ops completed\n",
                    n->pCommandListDebugNameW,
                    *n->pLastBreadcrumbValue,
                    n->BreadcrumbCount);
            /* The op AFTER the last completed one is the one
               that killed you. */

    D3D12_DRED_PAGE_FAULT_OUTPUT pf = {};
    if (SUCCEEDED(data->GetPageFaultAllocationOutput(&pf)))
        wprintf(L"page fault at GPU VA 0x%llx\n",
                pf.PageFaultVA);
        /* and the existing/freed allocation lists tell you
           WHICH resource - usually one you freed too early */
}

Name everything

/* Every D3D12 object takes a debug name. Without them, PIX,
   RenderDoc, the debug layer and DRED all report
   "ID3D12Resource" and you have no idea which one. */

resource->SetName(L"GBuffer.Albedo");
pso->SetName(L"PSO.OpaqueLit");
cmdList->SetName(L"CmdList.Shadow.Thread2");

/* A macro so it is never skipped: */
#if defined(_DEBUG)
  #define NAME_OBJ(obj) (obj)->SetName(L#obj)
#else
  #define NAME_OBJ(obj) ((void)0)
#endif

/* And mark up your command lists so a capture is readable: */
PIXBeginEvent(cmdList, PIX_COLOR(0,255,0), "Shadow Pass");
    /* ... draws ... */
PIXEndEvent(cmdList);

/* This takes ten minutes to add and saves hours on every
   single debugging session afterwards. Do it on day one, not
   when you first need it.                                  */
