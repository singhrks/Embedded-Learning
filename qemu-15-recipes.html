Recipes: one command line per track

A reference card. Each recipe is a complete, working command line for one kind of work on this site, with the reason for every non-obvious option. Copy, adjust the paths, run.

Assumptions

All recipes assume qemu-system-aarch64 8.x or newer, a cross toolchain, and an Image plus initramfs.cpio.gz built as in chapter 5 and chapter 6. Add -accel kvm -cpu host -M virt,gic-version=host on an arm64 host to any of them.

ARM architecture track: exception levels and system registers

#!/bin/sh
# Bare-metal payload, all exception levels available, stopped
# for the debugger.
qemu-system-aarch64 \
  -M virt,secure=on,virtualization=on,gic-version=3 \
  -cpu max \
  -smp 1 -m 512M -nographic \
  -kernel payload.elf \
  -d int,mmu,guest_errors,unimp -D qemu.log \
  -s -S

#   secure=on         EL3 exists -> you can write monitor code
#   virtualization=on EL2 exists -> you can write hypervisor code
#   -cpu max          every architectural feature: SVE, MTE,
#                     pointer auth, FEAT_NV2, S-EL2
#   -d int            EVERY exception, with ESR/ELR/FAR. This is
#                     the flag that makes this track teachable.
#   -s -S             stopped, GDB on :1234

# then:
$ aarch64-none-elf-gdb payload.elf \
    -ex 'target remote :1234' -ex 'layout regs' -ex 'break _start'

Linux kernel and driver track

#!/bin/sh
# Kernel plus initramfs, module loading, GDB, and a shared
# directory so you can rebuild a module without repacking.
KDIR=${KDIR:-$HOME/work/linux}

qemu-system-aarch64 \
  -M virt -cpu cortex-a57 -smp 2 -m 1G -nographic \
  -kernel $KDIR/arch/arm64/boot/Image \
  -initrd initramfs.cpio.gz \
  -append "console=ttyAMA0 earlycon nokaslr root=/dev/ram0 rw" \
  -fsdev local,id=fs0,path=$PWD/share,security_model=none \
  -device virtio-9p-device,fsdev=fs0,mount_tag=share \
  -netdev user,id=n0,hostfwd=tcp::2222-:22 \
  -device virtio-net-device,netdev=n0 \
  -object rng-random,filename=/dev/urandom,id=rng0 \
  -device virtio-rng-device,rng=rng0 \
  -d guest_errors,unimp -D qemu.log \
  -s

#   nokaslr           so GDB symbols match (chapter 11)
#   9p share          rebuild a module on the host, insmod it in
#                     the guest, no repacking
#   virtio-rng        without it, early boot blocks on entropy

# In the guest:
#   mount -t 9p -o trans=virtio,version=9p2000.L share /mnt
#   insmod /mnt/mydriver.ko
# On the host:
#   gdb-multiarch $KDIR/vmlinux -ex 'target remote :1234' \
#     -ex "source $KDIR/vmlinux-gdb.py" -ex 'lx-symbols'

Virtualisation track: nested, with everything visible

#!/bin/sh
# A guest that can itself run KVM, under TCG so that every
# instruction is traceable.
qemu-system-aarch64 \
  -M virt,virtualization=on,gic-version=3,its=on \
  -cpu max \
  -smp 4 -m 4G -nographic \
  -kernel Image \
  -append "console=ttyAMA0 earlycon kvm-arm.mode=nested" \
  -drive file=rootfs.ext4,if=none,format=raw,id=hd0 \
  -device virtio-blk-device,drive=hd0 \
  -netdev user,id=n0,hostfwd=tcp::2222-:22 \
  -device virtio-net-device,netdev=n0 \
  --trace 'kvm_*' --trace 'virtio_*' -D trace.log

#   -cpu max provides FEAT_NV2 even though no laptop has it
#   its=on   needed for PCIe MSI-X inside the guest

# On an arm64 host, the fast version:
#   -M virt,virtualization=on,gic-version=host \
#   -cpu host,el2=on -accel kvm

OP-TEE and TrustZone track

#!/bin/sh
# TF-A at EL3, OP-TEE at S-EL1, Linux at EL1, with the secure
# world's console on a second UART.
qemu-system-aarch64 \
  -M virt,secure=on,virtualization=on,gic-version=3 \
  -cpu max -smp 2 -m 1G -nographic \
  -bios bl1.bin \
  -device loader,file=fip.bin,addr=0x40100000 \
  -semihosting-config enable=on,target=native \
  -serial mon:stdio \
  -serial file:secure-world.log \
  -netdev user,id=n0 -device virtio-net-device,netdev=n0

#   -bios bl1.bin     TF-A's first stage. NOT -kernel: we are
#                     the firmware now (chapter 5)
#   fip.bin           the Firmware Image Package: BL2, BL31,
#                     BL32 (OP-TEE) and BL33 (U-Boot/Linux)
#   semihosting       TF-A on QEMU uses it to load the FIP
#   two -serial       first = ttyAMA0 = normal world
#                     second = the SECURE world's console. Every
#                     TA panic and trace line lands there.

# The OP-TEE build system generates this for you:
#   cd optee/build && make run
# and the two consoles appear in separate windows. See the
# OP-TEE track's chapter 4 for the full build.

Device-model development

#!/bin/sh
# Your own QEMU build, your own device, full tracing on both
# sides of the boundary.
QEMU=$HOME/opt/qemu/bin/qemu-system-aarch64

$QEMU \
  -M virt -cpu cortex-a57 -m 512M -nographic \
  -kernel Image -initrd initramfs.cpio.gz \
  -append "console=ttyAMA0 earlycon" \
  -device acme-widget \
  --trace 'acme_widget_*' \
  --trace 'memory_region_ops_read' \
  --trace 'memory_region_ops_write' \
  -d guest_errors,unimp -D dev.log \
  -s

#   memory_region_ops_* traces EVERY MMIO access QEMU
#   dispatched, for every device - so "is my device being
#   touched at all" is answered without any device-specific
#   instrumentation (chapter 12).

Fast iteration: overlay plus snapshot

#!/bin/sh
# Golden image untouched; every run disposable; snapshot before
# the expensive setup.
[ -f work.qcow2 ] || \
  qemu-img create -f qcow2 -b golden.qcow2 -F qcow2 work.qcow2

qemu-system-aarch64 \
  -M virt -cpu cortex-a57 -smp 4 -m 2G -nographic \
  -kernel Image \
  -drive file=work.qcow2,if=none,id=hd0,cache=unsafe \
  -device virtio-blk-device,drive=hd0 \
  -append "console=ttyAMA0 root=/dev/vda rw" \
  -monitor unix:/tmp/hmp.sock,server=on,wait=off \
  -qmp unix:/tmp/qmp.sock,server=on,wait=off

#   cache=unsafe   10x faster and safe HERE because the overlay
#                  is disposable. Never on anything you keep.

# Reset to pristine:  rm work.qcow2 && rerun
# Snapshot:  echo savevm ready | socat - UNIX-CONNECT:/tmp/hmp.sock
# Restore:   echo loadvm ready | socat - UNIX-CONNECT:/tmp/hmp.sock

CI: boot, run a test, exit with a status

#!/bin/sh
# Headless, bounded, detects panics, returns a real exit code.
set -e
timeout 300 qemu-system-aarch64 \
  -M virt -cpu cortex-a57 -smp 2 -m 1G \
  -nographic -no-reboot \
  -kernel Image -initrd test-initramfs.cpio.gz \
  -append "console=ttyAMA0 panic=-1 rdinit=/run-tests.sh" \
  -device pvpanic-pci \
  -serial mon:stdio \
  | tee test.log

grep -q "ALL TESTS PASSED" test.log || { echo "FAIL"; exit 1; }
grep -q "Kernel panic"     test.log && { echo "PANIC"; exit 1; }
echo OK

#   -no-reboot      a reset ends the run instead of looping
#   panic=-1        panic reboots immediately -> with -no-reboot,
#                   QEMU exits. A panic becomes a fast failure
#                   rather than a 300-second timeout.
#   pvpanic-pci     QEMU emits a GUEST_PANICKED QMP event too
#   timeout 300     the outer bound. Always have one.

# The test script inside the initramfs ends with:
#   echo "ALL TESTS PASSED"
#   poweroff -f

User-mode: run one cross-compiled binary

# No VM, no kernel, no rootfs. The fastest way to run an
# aarch64 test suite on an x86 machine.

$ qemu-aarch64 -L /usr/aarch64-linux-gnu ./my-test

# Transparently, via binfmt:
$ sudo apt install qemu-user-static binfmt-support
$ ./my-test              # the kernel routes it to qemu-aarch64

# Under GDB:
$ qemu-aarch64 -g 1234 -L /usr/aarch64-linux-gnu ./my-test &
$ gdb-multiarch ./my-test -ex 'target remote :1234'

# With instruction counting, to compare two implementations
# deterministically:
$ qemu-aarch64 -plugin libinsn.so -d plugin ./my-test

# Emulate a CPU with features your host lacks:
$ qemu-aarch64 -cpu max ./sve-test

The flag cheat sheet

FlagReach for it when
-nographicAlways, for headless work. Ctrl-A c for the monitor, Ctrl-A x to quit
-s -SYou want to debug from instruction zero
-d intThe guest hangs, resets, or dies silently. Start here
-d guest_errors,unimpAlways, during bring-up
-D fileEvery time you use -d
--trace 'memory_region_ops_*'"Is my device being accessed?"
-M virt,dumpdtb=x.dtb"Why is my device not probing?"
-monitor stdio / Ctrl-A cAnything is wrong and you want info registers
-snapshotYou are about to break the disk image
-icount shift=5A race that will not reproduce
-no-reboot -no-shutdownYou need to inspect the machine after it dies
-nodefaultsYou want exactly the machine you specified and nothing more
-accel kvm -cpu hostSame architecture and you want speed
-cpu maxYou need a feature no hardware you own has
-semihosting-config enable=onBare-metal code that wants host file or console access
