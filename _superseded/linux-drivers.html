Linux driver development

A driver is the translator between a hardware register block and a kernel subsystem. This chapter covers the device model that binds them, the four driver types you will meet, and every mechanism a driver uses: MMIO, interrupts, DMA, sysfs, and the user-space interface.

The device model

Linux separates what hardware exists from what code drives it, and matches them at runtime. Three objects:

ObjectRepresentsComes from
struct bus_typeA kind of connection: platform, PCI, I2C, SPI, USBThe subsystem; you rarely create one
struct deviceOne piece of hardware that existsDevice tree, ACPI, or bus enumeration
struct device_driverCode that can drive some devicesYour module, when it registers
Probe is the whole job

Almost all driver work happens in probe(): get resources, map registers, request the IRQ, initialise the hardware, register with a subsystem. remove() undoes it in reverse. Everything else is the operational callbacks.

The four driver shapes

Most real drivers combine them: a platform driver that also registers a character device, or a PCI driver that registers with the network subsystem.

Device tree

SoC peripherals cannot be probed — there is no way to ask an address whether a UART lives there. The device tree is a data structure, passed by the bootloader, that describes the hardware so that one kernel binary can support many boards.

/ {
    compatible = "acme,eval-board", "acme,soc9000";
    #address-cells = <2>;      /* addresses are 2 cells (64-bit) */
    #size-cells = <2>;

    soc {
        compatible = "simple-bus";
        ranges;                /* child addresses map 1:1 to parent */

        uart0: serial@40000000 {
            compatible = "acme,uart-v2", "ns16550a";
            reg = <0x0 0x40000000 0x0 0x1000>;      /* base, size */
            interrupts = <GIC_SPI 33 IRQ_TYPE_LEVEL_HIGH>;
            clocks = <&clk_uart>;
            clock-names = "baud";
            status = "okay";
        };

        widget0: widget@40010000 {
            compatible = "acme,widget-v1";
            reg = <0x0 0x40010000 0x0 0x100>;
            interrupts = <GIC_SPI 45 IRQ_TYPE_EDGE_RISING>;
            clocks = <&clk_widget>;
            resets = <&rst 12>;
            dmas = <&dma0 5>;
            dma-names = "rx";
            acme,fifo-depth = <64>;                 /* custom property */
            acme,enable-turbo;                      /* boolean property */
            reset-gpios = <&gpio1 7 GPIO_ACTIVE_LOW>;
        };
    };
};

Bindings are documentation with teeth

Every compatible string needs a binding in Documentation/devicetree/bindings/, written as a YAML schema. make dt_binding_check and make dtbs_check validate your tree against it. Upstream will not accept a driver without one.

Anatomy of a platform driver

the complete skeleton — this is the shape of thousands of real drivers
#include <linux/module.h>
#include <linux/platform_device.h>
#include <linux/of.h>
#include <linux/io.h>
#include <linux/clk.h>
#include <linux/interrupt.h>

#define REG_ID        0x00
#define REG_CTRL      0x04
#define REG_STATUS    0x08
#define   STATUS_IRQ  BIT(0)

struct widget {
        struct device   *dev;
        void __iomem    *regs;      /* mapped registers */
        struct clk      *clk;
        int              irq;
        u32              fifo_depth;
        spinlock_t       lock;
};

static irqreturn_t widget_isr(int irq, void *data)
{
        struct widget *w = data;
        u32 status = readl(w->regs + REG_STATUS);

        if (!(status & STATUS_IRQ))
                return IRQ_NONE;               /* shared IRQ: not ours */

        writel(STATUS_IRQ, w->regs + REG_STATUS);   /* W1C: clear at source */
        /* ... handle ... */
        return IRQ_HANDLED;
}

static int widget_probe(struct platform_device *pdev)
{
        struct device *dev = &pdev->dev;
        struct widget *w;
        int ret;

        w = devm_kzalloc(dev, sizeof(*w), GFP_KERNEL);
        if (!w)
                return -ENOMEM;
        w->dev = dev;
        spin_lock_init(&w->lock);

        /* 1. map the register block described by "reg" */
        w->regs = devm_platform_ioremap_resource(pdev, 0);
        if (IS_ERR(w->regs))
                return PTR_ERR(w->regs);

        /* 2. clocks: the block is dead until it is clocked */
        w->clk = devm_clk_get(dev, NULL);
        if (IS_ERR(w->clk))
                return dev_err_probe(dev, PTR_ERR(w->clk), "no clock\n");
        ret = clk_prepare_enable(w->clk);
        if (ret)
                return ret;

        /* 3. optional properties with sane defaults */
        if (device_property_read_u32(dev, "acme,fifo-depth", &w->fifo_depth))
                w->fifo_depth = 32;

        /* 4. interrupt */
        w->irq = platform_get_irq(pdev, 0);
        if (w->irq < 0) {
                ret = w->irq;
                goto err_clk;
        }
        ret = devm_request_irq(dev, w->irq, widget_isr, 0, dev_name(dev), w);
        if (ret)
                goto err_clk;

        /* 5. sanity-check the hardware is really there */
        dev_info(dev, "widget id 0x%08x, fifo %u\n",
                 readl(w->regs + REG_ID), w->fifo_depth);

        platform_set_drvdata(pdev, w);
        return 0;

err_clk:
        clk_disable_unprepare(w->clk);
        return ret;
}

static void widget_remove(struct platform_device *pdev)
{
        struct widget *w = platform_get_drvdata(pdev);

        writel(0, w->regs + REG_CTRL);       /* quiesce the hardware FIRST */
        clk_disable_unprepare(w->clk);
        /* devm_ resources are released automatically after this returns */
}

static const struct of_device_id widget_of_match[] = {
        { .compatible = "acme,widget-v1" },
        { }
};
MODULE_DEVICE_TABLE(of, widget_of_match);

static struct platform_driver widget_driver = {
        .probe  = widget_probe,
        .remove = widget_remove,
        .driver = {
                .name           = "acme-widget",
                .of_match_table = widget_of_match,
        },
};
module_platform_driver(widget_driver);

MODULE_LICENSE("GPL");
Two idioms worth copying forever
  • devm_* — resources are freed automatically when the device unbinds, even on an error path. This eliminates the most common class of driver leak.
  • dev_err_probe() — returns the error, logs it, and stays quiet if the error is -EPROBE_DEFER. Use it for every resource acquisition in probe.

Deferred probe

Your driver may probe before the clock controller it depends on exists. Returning -EPROBE_DEFER tells the core "not yet, try me again later", and it will re-run probe once more drivers have loaded. This is how Linux resolves arbitrary dependency graphs without any explicit ordering.

Register access

void __iomem *base = devm_platform_ioremap_resource(pdev, 0);

u32 v = readl(base + REG_STATUS);     /* 32-bit read,  with a barrier  */
writel(v | ENABLE, base + REG_CTRL);  /* 32-bit write, with a barrier  */
readb/readw/readq   writeb/writew/writeq

/* relaxed variants: no barrier, faster, use only when ordering does not matter */
readl_relaxed()  writel_relaxed()

/* poll a register until a condition holds, with a timeout */
ret = readl_poll_timeout(base + REG_STATUS, val,
                         val & STATUS_READY,   /* condition */
                         10,                   /* sleep 10 us between polls */
                         1000);                /* give up after 1000 us */
if (ret)
        return -ETIMEDOUT;
Why not just dereference the pointer?

void __iomem * is deliberately a distinct type that sparse checks. readl()/ writel() add the memory barriers required by the ARM memory model (chapter 03) and handle endianness. A direct dereference gives you neither, and the resulting bug appears only under load on multi-core systems.

Regmap

When a device is behind I2C or SPI, or has many registers, regmap gives a uniform API plus register caching and debugfs register dumps for free:

static const struct regmap_config widget_regmap = {
        .reg_bits = 32, .val_bits = 32, .reg_stride = 4,
        .max_register = 0xFF,
        .cache_type = REGCACHE_RBTREE,
};
w->map = devm_regmap_init_mmio(dev, w->regs, &widget_regmap);
regmap_read(w->map, REG_STATUS, &val);
regmap_update_bits(w->map, REG_CTRL, ENABLE_MASK, ENABLE_MASK);

Character devices — the user-space interface

A character device gives your driver a node in /dev that user space can open, read, write, ioctl and poll.

   user: open("/dev/widget0", O_RDWR)
        |
        v
   VFS looks up the inode -> sees a char device with major:minor
        |
        v
   finds the registered cdev for that major
        |
        v
   file->f_op = your struct file_operations
        |
        v
   calls your .open()   — typically stashes your device struct in
                          file->private_data so every later call can find it
        |
   read()  -> .read
   write() -> .write
   ioctl() -> .unlocked_ioctl
   poll()  -> .poll
   close() -> .release
registering a character device the modern way
static const struct file_operations widget_fops = {
        .owner          = THIS_MODULE,
        .open           = widget_open,
        .release        = widget_release,
        .read           = widget_read,
        .write          = widget_write,
        .unlocked_ioctl = widget_ioctl,
        .poll           = widget_poll,
        .llseek         = no_llseek,
};

/* in probe() */
        ret = alloc_chrdev_region(&w->devt, 0, 1, "widget");
        if (ret)
                return ret;

        cdev_init(&w->cdev, &widget_fops);
        w->cdev.owner = THIS_MODULE;
        ret = cdev_add(&w->cdev, w->devt, 1);
        if (ret)
                goto err_region;

        /* the class makes udev create /dev/widget0 automatically */
        w->class = class_create("widget");
        device_create(w->class, dev, w->devt, w, "widget%d", 0);

ioctl — defining a command interface

/* put this in a header shared with user space */
#define WIDGET_IOC_MAGIC  'W'
#define WIDGET_RESET      _IO(WIDGET_IOC_MAGIC,  0)
#define WIDGET_GET_STATS  _IOR(WIDGET_IOC_MAGIC, 1, struct widget_stats)
#define WIDGET_SET_MODE   _IOW(WIDGET_IOC_MAGIC, 2, __u32)
#define WIDGET_XCHG       _IOWR(WIDGET_IOC_MAGIC, 3, struct widget_xfer)

static long widget_ioctl(struct file *f, unsigned int cmd, unsigned long arg)
{
        struct widget *w = f->private_data;
        void __user *up = (void __user *)arg;
        struct widget_stats st;

        if (_IOC_TYPE(cmd) != WIDGET_IOC_MAGIC)
                return -ENOTTY;

        switch (cmd) {
        case WIDGET_RESET:
                return widget_do_reset(w);

        case WIDGET_GET_STATS:
                widget_fill_stats(w, &st);
                if (copy_to_user(up, &st, sizeof(st)))
                        return -EFAULT;
                return 0;

        default:
                return -ENOTTY;      /* the correct errno for an unknown ioctl */
        }
}
ioctl struct design rules

The structure is a permanent ABI. Use fixed-width types (__u32, not int), avoid pointers inside structs (or use __u64 and cast, so 32-bit user space on a 64-bit kernel works), add explicit padding so the layout is identical on both, and include a version or size field so you can extend it later. Getting this wrong means a compat layer forever.

sysfs and debugfs

Not everything needs an ioctl. For simple values, an attribute file is far nicer — scriptable, greppable, self-documenting.

static ssize_t fifo_depth_show(struct device *dev,

Interrupt handling in practice

/* simple handler */
ret = devm_request_irq(dev, irq, my_isr,
                       IRQF_SHARED,        /* if the line is shared */
                       dev_name(dev), w);

/* threaded — preferred for anything non-trivial */
ret = devm_request_threaded_irq(dev, irq, my_hard_isr, my_thread_fn,
                                IRQF_ONESHOT, dev_name(dev), w);
Return valueMeaning
IRQ_HANDLEDThis was my device and I dealt with it
IRQ_NONENot mine — essential on shared lines, or the core cannot detect a stuck interrupt
IRQ_WAKE_THREADWake the threaded handler

Rules for handlers

  • Clear the interrupt at the device, not just at the GIC, or a level-triggered line re-fires immediately (see chapter 03).
  • Cannot sleep. No mutex, no GFP_KERNEL, no copy_to_user, no msleep.
  • Keep it short. Acknowledge and defer. Long handlers destroy system latency.
  • Use spin_lock_irqsave in process context for data shared with the handler.
  • Free the IRQ before the device can still raise one. In remove(), disable interrupts at the device first, then release. devm_ gets the order right for you.

# if the count never increases: wrong IRQ number, wrong trigger type in DT,

DMA in a driver

The full model is in chapter 05. The driver-side summary:

/* 1. declare what your device can address */
ret = dma_set_mask_and_coherent(dev, DMA_BIT_MASK(48));

/* 2a. coherent: long-lived structures like descriptor rings */
w->ring = dma_alloc_coherent(dev, RING_SZ, &w->ring_dma, GFP_KERNEL);

/* 2b. streaming: one transfer, then hand the buffer back */
dma_addr_t da = dma_map_single(dev, buf, len, DMA_FROM_DEVICE);
if (dma_mapping_error(dev, da))
        return -ENOMEM;

writel(lower_32_bits(da), w->regs + REG_DMA_LO);
writel(upper_32_bits(da), w->regs + REG_DMA_HI);
writel(len,               w->regs + REG_DMA_LEN);
writel(DMA_GO,            w->regs + REG_CTRL);
/* ... wait for the completion interrupt ... */

dma_unmap_single(dev, da, len, DMA_FROM_DEVICE);   /* now the CPU may read buf */

/* scatter-gather for multi-page transfers */
nents = dma_map_sg(dev, sgl, orig_nents, DMA_TO_DEVICE);
for_each_sg(sgl, sg, nents, i)
        program_descriptor(sg_dma_address(sg), sg_dma_len(sg));
The four DMA rules
  1. Program the device with the dma_addr_t, never a virtual or physical address.
  2. Never DMA to stack memory or to vmalloc memory.
  3. Between map and unmap, the buffer belongs to the device. Use dma_sync_single_for_cpu() if you must peek.
  4. Stop the device before you unmap, on every path including error paths and remove().

Power management

static int widget_runtime_suspend(struct device *dev)
{
        struct widget *w = dev_get_drvdata(dev);
        clk_disable_unprepare(w->clk);
        return 0;
}
static int widget_runtime_resume(struct device *dev)
{
        struct widget *w = dev_get_drvdata(dev);
        return clk_prepare_enable(w->clk);
}

static const struct dev_pm_ops widget_pm_ops = {
        SET_RUNTIME_PM_OPS(widget_runtime_suspend, widget_runtime_resume, NULL)
        SET_SYSTEM_SLEEP_PM_OPS(widget_suspend, widget_resume)
};
/* .driver.pm = &widget_pm_ops */

/* in your I/O paths */
pm_runtime_get_sync(dev);      /* wake it up, refcount++ */
   ... do the work ...
pm_runtime_put_autosuspend(dev);   /* refcount--, idle after a delay */

Runtime PM is refcounted: the device sleeps when the count reaches zero. Getting the get/put pairing wrong gives you a device that either never sleeps (wasting power) or sleeps mid-transaction (aborts).

Error handling patterns

/* the classic goto ladder — idiomatic in the kernel */
static int probe(struct platform_device *pdev)
{
        ret = step_one();
        if (ret)
                return ret;

        ret = step_two();
        if (ret)
                goto err_one;

        ret = step_three();
        if (ret)
                goto err_two;

        return 0;

err_two:
        undo_two();
err_one:
        undo_one();
        return ret;
}

/* ...or avoid the ladder entirely with devm_ */
static int probe(struct platform_device *pdev)
{
        w->regs = devm_platform_ioremap_resource(pdev, 0);
        if (IS_ERR(w->regs))
                return PTR_ERR(w->regs);
        ret = devm_request_irq(dev, irq, isr, 0, name, w);
        if (ret)
                return ret;          /* the ioremap is released automatically */
        return 0;
}

Driver review checklist

  • Every probe() failure path releases what it acquired (or uses devm_).
  • remove() quiesces the hardware before freeing anything.
  • No sleeping calls in interrupt context or under a spinlock.
  • All user pointers go through copy_to_user/copy_from_user.
  • All ioctl inputs are validated — lengths, indices, enum ranges.
  • Register access uses readl/writel, and MMIO is not mapped cacheable.
  • DMA uses dma_addr_t and the mapping API; buffers are never on the stack.
  • Shared IRQ handlers return IRQ_NONE when the interrupt was not theirs.
  • checkpatch.pl --strict is clean, and make C=1 (sparse) reports nothing.
  • Tested with lockdep, KASAN and CONFIG_DEBUG_ATOMIC_SLEEP enabled.
