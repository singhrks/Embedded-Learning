Project — a complete Windows KMDF driver

The same ring buffer as the Linux project, built the Windows way: a KMDF software driver with read, write and IOCTL support, a manual queue for blocking reads, a timer, and a Win32 client application. Deliberately parallel to chapter 10, so you can compare the two ecosystems line by line.

What you need
  • Visual Studio 2022 with "Desktop development with C++"
  • Windows SDK and the matching WDK (SDK first, then WDK)
  • A disposable VM with bcdedit /set testsigning on and a snapshot taken
  • WinDbg on the host, kernel debugging attached (see chapter 11)

Specification

A software-only device — no hardware needed — installed as Root\KRingBuf and reachable from user mode as \\.\KRingBuf:

  • WriteFile appends bytes to a kernel ring buffer.
  • ReadFile consumes bytes; if empty, the request is parked and completed when data arrives — the Windows equivalent of a blocking read.
  • DeviceIoControl exposes reset, statistics, and a watermark setting.
  • A periodic timer injects a heartbeat byte, so you can watch a parked read complete asynchronously and exercise the DPC path without hardware.
  • Multiple concurrent readers and writers are safe.

Step 1 — create the project

  1. Visual Studio → Create a new project → search "Kernel Mode Driver, Empty (KMDF)".
  2. Name it KRingBuf.
  3. Right-click the project → Add → New Item and create the four files below.
  4. Project properties → Driver Signing → Sign Mode = Test Sign, and pick or create a test certificate.
  5. Set the configuration to Debug / x64 (or ARM64 if your VM is ARM).

#include <ntddk.h>
#include <wdf.h>
#include "public.h"

#define KRB_POOL_TAG      'fBRK'      // "KRBf" reversed - shows in !poolused

typedef struct _DEVICE_CONTEXT {

Step 4 — the driver

driver.c — part 1: entry, device creation, ring helpers
#include "driver.h"

//
// DriverEntry: called once when the driver is loaded.
// Its only real job is to register EvtDeviceAdd.
//
NTSTATUS
DriverEntry(_In_ PDRIVER_OBJECT DriverObject, _In_ PUNICODE_STRING RegistryPath)
{
    WDF_DRIVER_CONFIG config;
    NTSTATUS status;

    WDF_DRIVER_CONFIG_INIT(&config, KrbEvtDeviceAdd);

    status = WdfDriverCreate(DriverObject, RegistryPath,
                             WDF_NO_OBJECT_ATTRIBUTES, &config, WDF_NO_HANDLE);
    if (!NT_SUCCESS(status)) {
        KdPrint(("KRingBuf: WdfDriverCreate failed 0x%08X\n", status));
        return status;
    }

    KdPrint(("KRingBuf: loaded\n"));
    return STATUS_SUCCESS;
}

//
// Called once per device instance. Create every framework object here.
// NOTE: no hardware access - resources are not assigned yet.
//
NTSTATUS
KrbEvtDeviceAdd(_In_ WDFDRIVER Driver, _Inout_ PWDFDEVICE_INIT DeviceInit)
{
    NTSTATUS               status;
    WDFDEVICE              device;
    WDF_OBJECT_ATTRIBUTES  attrs;
    WDF_IO_QUEUE_CONFIG    queueConfig;
    WDF_TIMER_CONFIG       timerConfig;
    PDEVICE_CONTEXT        ctx;
    WDFQUEUE               defaultQueue;

    UNREFERENCED_PARAMETER(Driver);

    // ---- 1. the device object, with our context attached ----
    WDF_OBJECT_ATTRIBUTES_INIT_CONTEXT_TYPE(&attrs, DEVICE_CONTEXT);
    attrs.EvtCleanupCallback = KrbEvtDeviceContextCleanup;

    status = WdfDeviceCreate(&DeviceInit, &attrs, &device);
    if (!NT_SUCCESS(status))
        return status;

    ctx = DeviceGetContext(device);
    RtlZeroMemory(ctx, sizeof(*ctx));
    ctx->Capacity  = KRB_CAPACITY;
    ctx->Watermark = 1;

    // ---- 2. the ring buffer itself. NonPagedPoolNx: the timer DPC
    //         touches it at DISPATCH_LEVEL, where paging is illegal. ----
    ctx->Buffer = (PUCHAR)ExAllocatePool2(POOL_FLAG_NON_PAGED,
                                          ctx->Capacity, KRB_POOL_TAG);
    if (ctx->Buffer == NULL)
        return STATUS_INSUFFICIENT_RESOURCES;

    // ---- 3. the lock ----
    WDF_OBJECT_ATTRIBUTES_INIT(&attrs);
    attrs.ParentObject = device;
    status = WdfSpinLockCreate(&attrs, &ctx->Lock);
    if (!NT_SUCCESS(status))
        return status;

    // ---- 4. the default queue: sequential, so only one request is
    //         in flight at a time and our handlers need no extra locking
    //         against each other. ----
    WDF_IO_QUEUE_CONFIG_INIT_DEFAULT_QUEUE(&queueConfig,
                                           WdfIoQueueDispatchSequential);
    queueConfig.EvtIoRead          = KrbEvtIoRead;
    queueConfig.EvtIoWrite         = KrbEvtIoWrite;
    queueConfig.EvtIoDeviceControl = KrbEvtIoDeviceControl;

    status = WdfIoQueueCreate(device, &queueConfig,
                              WDF_NO_OBJECT_ATTRIBUTES, &defaultQueue);
    if (!NT_SUCCESS(status))
        return status;

    // ---- 5. the manual queue: reads park here when there is no data.
    //         WDF gives us cancellation handling for free. ----
    WDF_IO_QUEUE_CONFIG_INIT(&queueConfig, WdfIoQueueDispatchManual);
    status = WdfIoQueueCreate(device, &queueConfig,
                              WDF_NO_OBJECT_ATTRIBUTES, &ctx->ManualQueue);
    if (!NT_SUCCESS(status))
        return status;

    // ---- 6. the heartbeat timer (stands in for hardware interrupts) ----
    WDF_TIMER_CONFIG_INIT_PERIODIC(&timerConfig, KrbEvtTimer, KRB_TIMER_PERIOD);
    WDF_OBJECT_ATTRIBUTES_INIT(&attrs);
    attrs.ParentObject = device;
    status = WdfTimerCreate(&timerConfig, &attrs, &ctx->Timer);
    if (!NT_SUCCESS(status))
        return status;
    WdfTimerStart(ctx->Timer, WDF_REL_TIMEOUT_IN_MS(KRB_TIMER_PERIOD));

    // ---- 7. publish a device interface so user mode can find us ----
    status = WdfDeviceCreateDeviceInterface(device,
                                            &GUID_DEVINTERFACE_KRINGBUF, NULL);
    if (!NT_SUCCESS(status))
        return status;

    // ---- 8. and a symbolic link, so \\.\KRingBuf works too ----
    DECLARE_CONST_UNICODE_STRING(link, L"\\DosDevices\\KRingBuf");
    status = WdfDeviceCreateSymbolicLink(device, &link);

    KdPrint(("KRingBuf: device added, capacity %u\n", ctx->Capacity));
    return status;
}

VOID
KrbEvtDeviceContextCleanup(_In_ WDFOBJECT Object)
{
    PDEVICE_CONTEXT ctx = DeviceGetContext((WDFDEVICE)Object);

    if (ctx->Timer)
        WdfTimerStop(ctx->Timer, TRUE);       // TRUE = wait for it to finish

    if (ctx->Buffer) {
        ExFreePoolWithTag(ctx->Buffer, KRB_POOL_TAG);
        ctx->Buffer = NULL;
    }
    KdPrint(("KRingBuf: cleaned up\n"));
}

// ---- ring helpers. The caller must hold ctx->Lock. ----

static ULONG KrbSpace(_In_ PDEVICE_CONTEXT ctx)
{
    return ctx->Capacity - ctx->Count;
}

static VOID KrbPut(_In_ PDEVICE_CONTEXT ctx, _In_reads_bytes_(Len) PUCHAR Src,
                   _In_ ULONG Len)
{
    ULONG first = min(Len, ctx->Capacity - ctx->Head);

    RtlCopyMemory(ctx->Buffer + ctx->Head, Src, first);
    if (Len > first)
        RtlCopyMemory(ctx->Buffer, Src + first, Len - first);

    ctx->Head = (ctx->Head + Len) % ctx->Capacity;
    ctx->Count += Len;
    ctx->TotalWritten += Len;
}

static VOID KrbGet(_In_ PDEVICE_CONTEXT ctx, _Out_writes_bytes_(Len) PUCHAR Dst,
                   _In_ ULONG Len)
{
    ULONG first = min(Len, ctx->Capacity - ctx->Tail);

    RtlCopyMemory(Dst, ctx->Buffer + ctx->Tail, first);
    if (Len > first)
        RtlCopyMemory(Dst + first, ctx->Buffer, Len - first);

    ctx->Tail = (ctx->Tail + Len) % ctx->Capacity;
    ctx->Count -= Len;
    ctx->TotalRead += Len;
}
driver.c — part 2: read, write, the drain helper and the timer
//
// Try to satisfy one parked read. Caller must NOT hold the lock.
// Returns TRUE if a request was completed.
//
static BOOLEAN KrbTryCompleteOnePendingRead(_In_ PDEVICE_CONTEXT ctx)
{
    WDFREQUEST request;
    NTSTATUS   status;
    PVOID      buffer;
    size_t     bufLen;
    ULONG      toCopy;

    status = WdfIoQueueRetrieveNextRequest(ctx->ManualQueue, &request);
    if (!NT_SUCCESS(status))
        return FALSE;                         // nothing parked

    status = WdfRequestRetrieveOutputBuffer(request, 1, &buffer, &bufLen);
    if (!NT_SUCCESS(status)) {
        WdfRequestComplete(request, status);
        return TRUE;
    }

    WdfSpinLockAcquire(ctx->Lock);
    if (ctx->Count == 0) {
        // Raced with another reader. Put it back and give up.
        WdfSpinLockRelease(ctx->Lock);
        WdfRequestRequeue(request);
        return FALSE;
    }
    toCopy = min((ULONG)bufLen, ctx->Count);
    KrbGet(ctx, (PUCHAR)buffer, toCopy);
    WdfSpinLockRelease(ctx->Lock);

    WdfRequestCompleteWithInformation(request, STATUS_SUCCESS, toCopy);
    return TRUE;
}

//
// EvtIoRead. If data is available, complete immediately.
// Otherwise park the request in the manual queue and return WITHOUT
// completing it - the timer or a writer will finish it later.
//
VOID
KrbEvtIoRead(_In_ WDFQUEUE Queue, _In_ WDFREQUEST Request, _In_ size_t Length)
{
    WDFDEVICE       device = WdfIoQueueGetDevice(Queue);
    PDEVICE_CONTEXT ctx    = DeviceGetContext(device);
    NTSTATUS        status;
    PVOID           buffer;
    size_t          bufLen;
    ULONG           toCopy = 0;

    if (Length == 0) {
        WdfRequestCompleteWithInformation(Request, STATUS_SUCCESS, 0);
        return;
    }

    status = WdfRequestRetrieveOutputBuffer(Request, 1, &buffer, &bufLen);
    if (!NT_SUCCESS(status)) {
        WdfRequestComplete(Request, status);
        return;
    }

    WdfSpinLockAcquire(ctx->Lock);
    if (ctx->Count > 0) {
        toCopy = min((ULONG)bufLen, ctx->Count);
        KrbGet(ctx, (PUCHAR)buffer, toCopy);
    }
    WdfSpinLockRelease(ctx->Lock);

    if (toCopy > 0) {
        WdfRequestCompleteWithInformation(Request, STATUS_SUCCESS, toCopy);
        return;
    }

    // No data. Park it. This is the Windows equivalent of a blocking read.
    status = WdfRequestForwardToIoQueue(Request, ctx->ManualQueue);
    if (!NT_SUCCESS(status))
        WdfRequestComplete(Request, status);
    // NOTE: on success we return without completing. That is correct.
}

VOID
KrbEvtIoWrite(_In_ WDFQUEUE Queue, _In_ WDFREQUEST Request, _In_ size_t Length)
{
    WDFDEVICE       device = WdfIoQueueGetDevice(Queue);
    PDEVICE_CONTEXT ctx    = DeviceGetContext(device);
    NTSTATUS        status;
    PVOID           buffer;
    size_t          bufLen;
    ULONG           toCopy;

    if (Length == 0) {
        WdfRequestCompleteWithInformation(Request, STATUS_SUCCESS, 0);
        return;
    }

    status = WdfRequestRetrieveInputBuffer(Request, 1, &buffer, &bufLen);
    if (!NT_SUCCESS(status)) {
        WdfRequestComplete(Request, status);
        return;
    }

    WdfSpinLockAcquire(ctx->Lock);
    toCopy = min((ULONG)bufLen, KrbSpace(ctx));
    if (toCopy == 0)
        ctx->Overruns++;
    else
        KrbPut(ctx, (PUCHAR)buffer, toCopy);
    WdfSpinLockRelease(ctx->Lock);

    if (toCopy == 0) {
        // Full. A real driver might park the write too; we fail fast.
        WdfRequestComplete(Request, STATUS_DEVICE_BUSY);
        return;
    }

    // New data may satisfy a parked reader.
    while (KrbTryCompleteOnePendingRead(ctx))
        ;

    WdfRequestCompleteWithInformation(Request, STATUS_SUCCESS, toCopy);
}

//
// Periodic timer. Runs at DISPATCH_LEVEL - no paging, no waiting.
// Stands in for a hardware interrupt DPC.
//
VOID
KrbEvtTimer(_In_ WDFTIMER Timer)
{
    WDFDEVICE       device = (WDFDEVICE)WdfTimerGetParentObject(Timer);
    PDEVICE_CONTEXT ctx    = DeviceGetContext(device);
    UCHAR           beat;

    ctx->TimerTicks++;
    beat = (UCHAR)('0' + (ctx->TimerTicks % 10));

    WdfSpinLockAcquire(ctx->Lock);
    if (KrbSpace(ctx) > 0)
        KrbPut(ctx, &beat, 1);
    else
        ctx->Overruns++;
    WdfSpinLockRelease(ctx->Lock);

    while (KrbTryCompleteOnePendingRead(ctx))
        ;
}
driver.c — part 3: the IOCTL handler
VOID
KrbEvtIoDeviceControl(_In_ WDFQUEUE Queue, _In_ WDFREQUEST Request,
                      _In_ size_t OutputBufferLength,
                      _In_ size_t InputBufferLength,
                      _In_ ULONG  IoControlCode)
{
    WDFDEVICE       device = WdfIoQueueGetDevice(Queue);
    PDEVICE_CONTEXT ctx    = DeviceGetContext(device);
    NTSTATUS        status = STATUS_INVALID_DEVICE_REQUEST;
    ULONG_PTR       info   = 0;
    PVOID           buffer;
    size_t          bufLen;

    UNREFERENCED_PARAMETER(OutputBufferLength);
    UNREFERENCED_PARAMETER(InputBufferLength);

    switch (IoControlCode) {

    case IOCTL_KRB_RESET:
        WdfSpinLockAcquire(ctx->Lock);
        ctx->Head = ctx->Tail = ctx->Count = 0;
        ctx->Overruns = 0;
        WdfSpinLockRelease(ctx->Lock);
        status = STATUS_SUCCESS;
        break;

    case IOCTL_KRB_GET_STATS: {
        PKRB_STATS st;

        // The minimum-size argument makes WDF reject a short buffer for us.
        status = WdfRequestRetrieveOutputBuffer(Request, sizeof(KRB_STATS),
                                                &buffer, &bufLen);
        if (!NT_SUCCESS(status))
            break;

        st = (PKRB_STATS)buffer;
        WdfSpinLockAcquire(ctx->Lock);
        st->Capacity     = ctx->Capacity;
        st->Count        = ctx->Count;
        st->TotalWritten = ctx->TotalWritten;
        st->TotalRead    = ctx->TotalRead;
        st->Overruns     = ctx->Overruns;
        st->Watermark    = ctx->Watermark;
        WdfSpinLockRelease(ctx->Lock);
        st->PendingReads = 0;

        info   = sizeof(KRB_STATS);
        status = STATUS_SUCCESS;
        break;
    }

    case IOCTL_KRB_SET_WATERMARK: {
        ULONG value;

        status = WdfRequestRetrieveInputBuffer(Request, sizeof(ULONG),
                                               &buffer, &bufLen);
        if (!NT_SUCCESS(status))
            break;

        value = *(PULONG)buffer;
        if (value > ctx->Capacity) {      // VALIDATE every user input
            status = STATUS_INVALID_PARAMETER;
            break;
        }
        WdfSpinLockAcquire(ctx->Lock);
        ctx->Watermark = value;
        WdfSpinLockRelease(ctx->Lock);
        status = STATUS_SUCCESS;
        break;
    }

    case IOCTL_KRB_GET_CAPACITY:
        status = WdfRequestRetrieveOutputBuffer(Request, sizeof(ULONG),
                                                &buffer, &bufLen);
        if (!NT_SUCCESS(status))
            break;
        *(PULONG)buffer = ctx->Capacity;
        info   = sizeof(ULONG);
        status = STATUS_SUCCESS;
        break;

    default:
        status = STATUS_INVALID_DEVICE_REQUEST;
        break;
    }

    WdfRequestCompleteWithInformation(Request, status, info);
}
Read the read path twice

KrbEvtIoRead either completes the request or forwards it — and on the forward path it returns having completed nothing. If you ever add a code path that does neither, the calling application hangs forever with no error. "Exactly one of: complete, forward, or fail" is the invariant to check on every branch of every WDF handler.

Step 6 — the user-mode application

krbtest.c — build as a separate console project
#include <windows.h>
#include <stdio.h>
#include <initguid.h>
#include "public.h"

#define DEV_PATH L"\\\\.\\KRingBuf"

static void ShowStats(HANDLE h, const char *when)
{
    KRB_STATS st;
    DWORD     ret;

    if (!DeviceIoControl(h, IOCTL_KRB_GET_STATS, NULL, 0,
                         &st, sizeof(st), &ret, NULL)) {
        printf("  GET_STATS failed: %lu\n", GetLastError());
        return;
    }
    printf("[%s] cap=%lu count=%lu written=%llu read=%llu overruns=%lu\n",
           when, st.Capacity, st.Count, st.TotalWritten, st.TotalRead,
           st.Overruns);
}

int main(void)
{
    HANDLE h;
    DWORD  ret, cap, wm;
    char   out[128], in[128];

    h = CreateFileW(DEV_PATH, GENERIC_READ | GENERIC_WRITE,
                    0, NULL, OPEN_EXISTING, 0, NULL);
    if (h == INVALID_HANDLE_VALUE) {
        printf("CreateFile failed: %lu "
               "(is the driver installed and started?)\n", GetLastError());
        return 1;
    }
    printf("opened %ws\n", DEV_PATH);

    /* --- capacity --- */
    if (DeviceIoControl(h, IOCTL_KRB_GET_CAPACITY, NULL, 0,
                        &cap, sizeof(cap), &ret, NULL))
        printf("capacity: %lu bytes\n", cap);

    /* --- start clean --- */
    DeviceIoControl(h, IOCTL_KRB_RESET, NULL, 0, NULL, 0, &ret, NULL);
    ShowStats(h, "after reset");

    /* --- write then read --- */
    strcpy_s(out, sizeof(out), "hello from user mode");
    if (!WriteFile(h, out, (DWORD)strlen(out), &ret, NULL))
        printf("WriteFile failed: %lu\n", GetLastError());
    else
        printf("wrote %lu bytes\n", ret);
    ShowStats(h, "after write");

    ZeroMemory(in, sizeof(in));
    if (ReadFile(h, in, sizeof(in) - 1, &ret, NULL))
        printf("read  %lu bytes: \"%s\"\n", ret, in);
    ShowStats(h, "after read");

    /* --- the interesting one: this read finds the buffer empty, so the
           driver parks it. It returns when the 1-second timer fires. --- */
    printf("\nblocking read (the timer should complete it within ~1 s)...\n");
    DWORD t0 = GetTickCount();
    ZeroMemory(in, sizeof(in));
    if (ReadFile(h, in, sizeof(in) - 1, &ret, NULL))
        printf("  returned %lu bytes (\"%s\") after %lu ms\n",
               ret, in, GetTickCount() - t0);
    else
        printf("  failed: %lu\n", GetLastError());

    /* --- invalid watermark must be rejected --- */
    wm = cap + 1;
    if (!DeviceIoControl(h, IOCTL_KRB_SET_WATERMARK, &wm, sizeof(wm),
                         NULL, 0, &ret, NULL))
        printf("\noversized watermark rejected: %lu "
               "(expect 87 = ERROR_INVALID_PARAMETER)\n", GetLastError());

    /* --- unknown IOCTL must be rejected --- */
    if (!DeviceIoControl(h, CTL_CODE(KRB_DEVICE_TYPE, 0x8FF,
                                     METHOD_BUFFERED, FILE_ANY_ACCESS),
                         NULL, 0, NULL, 0, &ret, NULL))
        printf("unknown ioctl rejected: %lu\n", GetLastError());

    /* --- watch the heartbeat accumulate --- */
    printf("\nwatching the heartbeat for 5 seconds:\n");
    for (int i = 0; i < 5; i++) {
        Sleep(1000);
        ShowStats(h, "tick");
    }

    CloseHandle(h);
    return 0;
}

That 638 ms is the whole lesson

The read arrived, found nothing, and was parked in the manual queue with no thread blocked in your driver. The timer DPC — running at DISPATCH_LEVEL on some arbitrary CPU, in some arbitrary thread context — retrieved and completed it. That decoupling of "who received the request" from "who completes it" is the core of the Windows I/O model, and it is what makes asynchronous I/O natural on Windows.

The design difference worth internalising

Linux drivers block a thread. Windows drivers park a request. The Linux model is simpler to write; the Windows model scales better because no kernel thread is tied up waiting, and it makes cancellation a framework concern rather than a driver concern. Neither is wrong — but knowing why each ecosystem chose its model is what makes you fluent in both.

Extensions to try

  1. Honour the watermark. Only complete a parked read once Count >= Watermark.
  2. Park writes too when the buffer is full, in a second manual queue, and complete them when a read makes room.
  3. Switch the default queue to parallel dispatch and find out what breaks. Then fix it. This teaches the WDF synchronisation scopes better than any document.
  4. Add EvtIoStop and test suspend/resume — a request parked across a power transition must be handled correctly, and this is a real-world bug source.
  5. Add WPP tracing and capture a trace with tracelog.
  6. Port it to UMDF. The logic is nearly identical, but a bug takes down a service rather than the machine — which shows exactly why UMDF is the recommended default.
