Project — a complete Linux driver

A real, buildable character driver: a kernel-resident ring buffer with blocking reads, ioctls, sysfs attributes and poll support, plus a user-space test program. Then a second driver showing device-tree binding, MMIO and interrupts. Type it, build it, break it.

What you need

Any Linux machine (a VM is fine) with kernel headers installed:

# Debian / Ubuntu
sudo apt install build-essential linux-headers-$(uname -r)
# Fedora
sudo dnf install kernel-devel kernel-headers gcc make

No hardware is required for Part 1. Part 2 needs a board or QEMU with a device tree, but reading it is valuable even if you cannot run it.

Part 1 — specification

A device /dev/kringbuf0 that behaves like a fixed-size FIFO living in the kernel:

  • write() appends bytes; if the buffer is full it blocks (or returns -EAGAIN with O_NONBLOCK).
  • read() consumes bytes; if empty it blocks until data arrives.
  • poll()/select() works, so it can be used in an event loop.
  • ioctl() exposes: reset, get statistics, set the high-water mark.
  • sysfs exposes the current fill level and total byte counters.
  • Multiple readers and writers are safe.

This exercises every mechanism from chapter 09: the char device interface, sleeping and waking, locking, user-space copies, and the ABI header.

Step 1 — the shared ABI header

This header is included by both the driver and user space. It is the contract, and once shipped you can only extend it, never change it.

kringbuf.h
#ifndef _UAPI_KRINGBUF_H
#define _UAPI_KRINGBUF_H

#include <linux/ioctl.h>
#include <linux/types.h>

/* Statistics returned by KRB_GET_STATS.
 * Fixed-width types and explicit padding so the layout is identical
 * for 32-bit and 64-bit user space.
 */
struct krb_stats {
        __u32 capacity;         /* total buffer size in bytes        */
        __u32 count;            /* bytes currently stored            */
        __u64 total_written;    /* lifetime bytes written            */
        __u64 total_read;       /* lifetime bytes read               */
        __u32 overruns;         /* writes rejected because it was full */
        __u32 watermark;        /* poll reports readable at or above this */
};

#define KRB_IOC_MAGIC       'k'

#define KRB_RESET           _IO(KRB_IOC_MAGIC,  0)
#define KRB_GET_STATS       _IOR(KRB_IOC_MAGIC, 1, struct krb_stats)
#define KRB_SET_WATERMARK   _IOW(KRB_IOC_MAGIC, 2, __u32)
#define KRB_GET_CAPACITY    _IOR(KRB_IOC_MAGIC, 3, __u32)

#define KRB_IOC_MAXNR       3

#endif /* _UAPI_KRINGBUF_H */

static int capacity = DEFAULT_CAPACITY;
module_param(capacity, int, 0444);
MODULE_PARM_DESC(capacity, "ring buffer size in bytes (default 4096)");

struct krb_dev {
        char             *buf;        /* the ring itself                   */
        size_t            capacity;
        size_t            head;       /* next write position               */
        size_t            tail;       /* next read position                */
        size_t            count;      /* bytes currently stored            */

        u64               total_written;
        u64               total_read;
        u32               overruns;
        u32               watermark;

        struct mutex      lock;       /* protects everything above         */
        wait_queue_head_t readq;      /* readers sleep here when empty     */
        wait_queue_head_t writeq;     /* writers sleep here when full      */

        dev_t             devt;
        struct cdev       cdev;
        struct device    *dev;
        atomic_t          open_count;
};

static struct krb_dev *krb;            /* single instance for this example */
static struct class   *krb_class;

/* ---- helpers. All of these require the mutex to be held. ---- */

static inline size_t krb_space(const struct krb_dev *d)
{
        return d->capacity - d->count;
}

/* Copy from user space into the ring, wrapping at the end. */
static int krb_put(struct krb_dev *d, const char __user *ubuf, size_t n)
{
        size_t first = min(n, d->capacity - d->head);   /* until the wrap */

        if (copy_from_user(d->buf + d->head, ubuf, first))
                return -EFAULT;
        if (n > first && copy_from_user(d->buf, ubuf + first, n - first))
                return -EFAULT;

        d->head   = (d->head + n) % d->capacity;
        d->count += n;
        d->total_written += n;
        return 0;
}

/* Copy from the ring out to user space, wrapping at the end. */
static int krb_get(struct krb_dev *d, char __user *ubuf, size_t n)
{
        size_t first = min(n, d->capacity - d->tail);

        if (copy_to_user(ubuf, d->buf + d->tail, first))
                return -EFAULT;
        if (n > first && copy_to_user(ubuf + first, d->buf, n - first))
                return -EFAULT;

        d->tail   = (d->tail + n) % d->capacity;
        d->count -= n;
        d->total_read += n;
        return 0;
}

/* ---- file operations ---- */

static int krb_open(struct inode *inode, struct file *file)
{
        struct krb_dev *d = container_of(inode->i_cdev, struct krb_dev, cdev);

        file->private_data = d;                  /* every later call finds it here */
        atomic_inc(&d->open_count);
        dev_dbg(d->dev, "open by %s (pid %d), now %d open\n",
                current->comm, current->pid, atomic_read(&d->open_count));
        return 0;
}

static int krb_release(struct inode *inode, struct file *file)
{
        struct krb_dev *d = file->private_data;

        atomic_dec(&d->open_count);
        return 0;
}
kringbuf.c — part 2 of 3: read, write, poll, ioctl
static ssize_t krb_read(struct file *file, char __user *ubuf,
                        size_t len, loff_t *off)
{
        struct krb_dev *d = file->private_data;
        size_t n;
        int ret;

        if (len == 0)
                return 0;

        if (mutex_lock_interruptible(&d->lock))
                return -ERESTARTSYS;

        /* Wait for data. Note the loop: after waking we must re-check,
         * because another reader may have consumed it first. */
        while (d->count == 0) {
                mutex_unlock(&d->lock);          /* never sleep holding a lock */

                if (file->f_flags & O_NONBLOCK)
                        return -EAGAIN;

                if (wait_event_interruptible(d->readq, d->count > 0))
                        return -ERESTARTSYS;     /* a signal arrived */

                if (mutex_lock_interruptible(&d->lock))
                        return -ERESTARTSYS;
        }

        n = min(len, d->count);
        ret = krb_get(d, ubuf, n);
        mutex_unlock(&d->lock);
        if (ret)
                return ret;

        wake_up_interruptible(&d->writeq);       /* space is available now */
        return n;
}

static ssize_t krb_write(struct file *file, const char __user *ubuf,
                         size_t len, loff_t *off)
{
        struct krb_dev *d = file->private_data;
        size_t n;
        int ret;

        if (len == 0)
                return 0;

        if (mutex_lock_interruptible(&d->lock))
                return -ERESTARTSYS;

        while (krb_space(d) == 0) {
                d->overruns++;
                mutex_unlock(&d->lock);

                if (file->f_flags & O_NONBLOCK)
                        return -EAGAIN;

                if (wait_event_interruptible(d->writeq, krb_space(d) > 0))
                        return -ERESTARTSYS;

                if (mutex_lock_interruptible(&d->lock))
                        return -ERESTARTSYS;
        }

        n = min(len, krb_space(d));               /* short write is legal */
        ret = krb_put(d, ubuf, n);
        mutex_unlock(&d->lock);
        if (ret)
                return ret;

        wake_up_interruptible(&d->readq);
        return n;
}

static __poll_t krb_poll(struct file *file, struct poll_table_struct *wait)
{
        struct krb_dev *d = file->private_data;
        __poll_t mask = 0;

        /* Register on both queues BEFORE checking state, or you can miss
         * a wakeup that happens between the check and the sleep. */
        poll_wait(file, &d->readq,  wait);
        poll_wait(file, &d->writeq, wait);

        mutex_lock(&d->lock);
        if (d->count >= d->watermark && d->count > 0)
                mask |= EPOLLIN | EPOLLRDNORM;
        if (krb_space(d) > 0)
                mask |= EPOLLOUT | EPOLLWRNORM;
        mutex_unlock(&d->lock);

        return mask;
}

static long krb_ioctl(struct file *file, unsigned int cmd, unsigned long arg)
{
        struct krb_dev *d = file->private_data;
        void __user *up = (void __user *)arg;
        struct krb_stats st;
        u32 val;

        /* Always validate the command before touching anything. */
        if (_IOC_TYPE(cmd) != KRB_IOC_MAGIC)
                return -ENOTTY;
        if (_IOC_NR(cmd) > KRB_IOC_MAXNR)
                return -ENOTTY;

        switch (cmd) {
        case KRB_RESET:
                mutex_lock(&d->lock);
                d->head = d->tail = d->count = 0;
                d->overruns = 0;
                mutex_unlock(&d->lock);
                wake_up_interruptible(&d->writeq);
                dev_info(d->dev, "reset by pid %d\n", current->pid);
                return 0;

        case KRB_GET_STATS:
                mutex_lock(&d->lock);
                st.capacity      = d->capacity;
                st.count         = d->count;
                st.total_written = d->total_written;
                st.total_read    = d->total_read;
                st.overruns      = d->overruns;
                st.watermark     = d->watermark;
                mutex_unlock(&d->lock);

                if (copy_to_user(up, &st, sizeof(st)))
                        return -EFAULT;
                return 0;

        case KRB_SET_WATERMARK:
                if (copy_from_user(&val, up, sizeof(val)))
                        return -EFAULT;
                if (val > d->capacity)            /* validate! */
                        return -EINVAL;
                mutex_lock(&d->lock);
                d->watermark = val;
                mutex_unlock(&d->lock);
                wake_up_interruptible(&d->readq);
                return 0;

        case KRB_GET_CAPACITY:
                val = d->capacity;
                if (copy_to_user(up, &val, sizeof(val)))
                        return -EFAULT;
                return 0;
        }
        return -ENOTTY;
}

static const struct file_operations krb_fops = {
        .owner          = THIS_MODULE,
        .open           = krb_open,
        .release        = krb_release,
        .read           = krb_read,
        .write          = krb_write,
        .poll           = krb_poll,
        .unlocked_ioctl = krb_ioctl,
        .compat_ioctl   = compat_ptr_ioctl,   /* 32-bit apps on a 64-bit kernel */
        .llseek         = no_llseek,
};

static ssize_t count_show(struct device *dev,
                          struct device_attribute *attr, char *buf)
{
        struct krb_dev *d = dev_get_drvdata(dev);
        size_t n;

        mutex_lock(&d->lock);
        n = d->count;
        mutex_unlock(&d->lock);
        return sysfs_emit(buf, "%zu\n", n);
}

        return sysfs_emit(buf, "%zu\n", d->capacity);
}
static DEVICE_ATTR_RO(capacity);

static ssize_t watermark_show(struct device *dev,
                              struct device_attribute *attr, char *buf)
{
        struct krb_dev *d = dev_get_drvdata(dev);
        return sysfs_emit(buf, "%u\n", d->watermark);
}
static ssize_t watermark_store(struct device *dev,
                               struct device_attribute *attr,
                               const char *buf, size_t len)
{
        struct krb_dev *d = dev_get_drvdata(dev);
        u32 val;
        int ret = kstrtou32(buf, 0, &val);

        if (ret)
                return ret;
        if (val > d->capacity)
                return -EINVAL;

        mutex_lock(&d->lock);
        d->watermark = val;
        mutex_unlock(&d->lock);
        wake_up_interruptible(&d->readq);
        return len;                       /* consumed all of it */
}
static DEVICE_ATTR_RW(watermark);

        &dev_attr_capacity.attr,
        &dev_attr_watermark.attr,
        NULL,
};

/* ---- module init / exit ---- */

static int __init krb_init(void)
{
        int ret;

        if (capacity < 64 || capacity > (1 << 20)) {
                pr_err("capacity must be between 64 and 1048576\n");
                return -EINVAL;
        }

        krb = kzalloc(sizeof(*krb), GFP_KERNEL);
        if (!krb)
                return -ENOMEM;

        krb->buf = kzalloc(capacity, GFP_KERNEL);
        if (!krb->buf) {
                ret = -ENOMEM;
                goto err_free_dev;
        }
        krb->capacity  = capacity;
        krb->watermark = 1;
        mutex_init(&krb->lock);
        init_waitqueue_head(&krb->readq);
        init_waitqueue_head(&krb->writeq);
        atomic_set(&krb->open_count, 0);

        ret = alloc_chrdev_region(&krb->devt, 0, 1, DRV_NAME);
        if (ret)
                goto err_free_buf;

        cdev_init(&krb->cdev, &krb_fops);
        krb->cdev.owner = THIS_MODULE;
        ret = cdev_add(&krb->cdev, krb->devt, 1);
        if (ret)
                goto err_region;

        /* class_create() lost its module argument in 6.4 */
#if LINUX_VERSION_CODE >= KERNEL_VERSION(6, 4, 0)
        krb_class = class_create(DRV_NAME);
#else
        krb_class = class_create(THIS_MODULE, DRV_NAME);
#endif
        if (IS_ERR(krb_class)) {
                ret = PTR_ERR(krb_class);
                goto err_cdev;
        }
        krb_class->dev_groups = krb_groups;

        krb->dev = device_create(krb_class, NULL, krb->devt, krb,
                                 DRV_NAME "%d", 0);
        if (IS_ERR(krb->dev)) {
                ret = PTR_ERR(krb->dev);
                goto err_class;
        }

        pr_info("loaded: /dev/%s0 major %d, capacity %zu bytes\n",
                DRV_NAME, MAJOR(krb->devt), krb->capacity);
        return 0;

err_class:
        class_destroy(krb_class);
err_cdev:
        cdev_del(&krb->cdev);
err_region:
        unregister_chrdev_region(krb->devt, 1);
err_free_buf:
        kfree(krb->buf);
err_free_dev:
        kfree(krb);
        return ret;
}

static void __exit krb_exit(void)
{
        device_destroy(krb_class, krb->devt);
        class_destroy(krb_class);
        cdev_del(&krb->cdev);
        unregister_chrdev_region(krb->devt, 1);
        kfree(krb->buf);
        kfree(krb);
        pr_info("unloaded\n");
}

module_init(krb_init);
module_exit(krb_exit);

Step 4 — the user-space test program

krbtest.c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <fcntl.h>
#include <errno.h>
#include <poll.h>
#include <sys/ioctl.h>

#include "kringbuf.h"

#define DEV "/dev/kringbuf0"

static void show_stats(int fd, const char *when)
{
        struct krb_stats st;

        if (ioctl(fd, KRB_GET_STATS, &st) < 0) {
                perror("KRB_GET_STATS");
                return;
        }
        printf("[%s] cap=%u count=%u written=%llu read=%llu "
               "overruns=%u watermark=%u\n", when,
               st.capacity, st.count,
               (unsigned long long)st.total_written,
               (unsigned long long)st.total_read,
               st.overruns, st.watermark);
}

int main(void)
{
        char out[128], in[128];
        int fd, n;
        unsigned int cap, wm = 8;

        fd = open(DEV, O_RDWR);
        if (fd < 0) { perror("open " DEV); return 1; }

        /* --- capacity --- */
        if (ioctl(fd, KRB_GET_CAPACITY, &cap) < 0) { perror("cap"); return 1; }
        printf("capacity reported by driver: %u bytes\n", cap);

        /* --- start clean --- */
        ioctl(fd, KRB_RESET);
        show_stats(fd, "after reset");

        /* --- basic write then read --- */
        strcpy(out, "hello from user space");
        n = write(fd, out, strlen(out));
        printf("wrote %d bytes\n", n);
        show_stats(fd, "after write");

        memset(in, 0, sizeof(in));
        n = read(fd, in, sizeof(in) - 1);
        printf("read  %d bytes: \"%s\"\n", n, in);
        show_stats(fd, "after read");

        /* --- non-blocking read on an empty buffer must give EAGAIN --- */
        {
                int nb = open(DEV, O_RDONLY | O_NONBLOCK);
                n = read(nb, in, sizeof(in));
                printf("non-blocking read on empty: ret=%d errno=%s (expect EAGAIN)\n",
                       n, strerror(errno));
                close(nb);
        }

        /* --- watermark + poll --- */
        ioctl(fd, KRB_SET_WATERMARK, &wm);
        printf("watermark set to %u\n", wm);

        write(fd, "1234", 4);              /* below the watermark */
        {
                struct pollfd p = { .fd = fd, .events = POLLIN };
                int r = poll(&p, 1, 100);
                printf("poll with 4 bytes (below watermark): %d (expect 0)\n", r);
        }
        write(fd, "5678", 4);              /* now at the watermark */
        {
                struct pollfd p = { .fd = fd, .events = POLLIN };
                int r = poll(&p, 1, 100);
                printf("poll with 8 bytes (at watermark): %d, revents=0x%x "
                       "(expect 1, POLLIN)\n", r, p.revents);
        }

        /* --- fill it completely and check the short write --- */
        ioctl(fd, KRB_RESET);
        {
                char *big = malloc(cap + 64);
                memset(big, 'A', cap + 64);
                n = write(fd, big, cap + 64);
                printf("write of %u into a %u buffer returned %d "
                       "(expect a short write of %u)\n", cap + 64, cap, n, cap);
                free(big);
        }
        show_stats(fd, "when full");

        /* --- invalid ioctl must give ENOTTY --- */
        if (ioctl(fd, _IO('Z', 99)) < 0)
                printf("bad ioctl rejected with %s (expect ENOTTY)\n",
                       strerror(errno));

        /* --- invalid watermark must give EINVAL --- */
        wm = cap + 1;
        if (ioctl(fd, KRB_SET_WATERMARK, &wm) < 0)
                printf("oversized watermark rejected with %s (expect EINVAL)\n",
                       strerror(errno));

        close(fd);
        return 0;
}

Step 5 — break it deliberately

This is the part that teaches the most. Each experiment produces a specific, recognisable failure.

ChangeWhat happensLesson
Replace copy_to_user() with memcpy()Oops: "Unable to handle kernel paging request" the moment PAN is enforcedUser pointers are never directly dereferenceable

#include <linux/debugfs.h>
#include <linux/iopoll.h>

#define REG_ID          0x00
#define REG_CTRL        0x04
#define   CTRL_ENABLE   BIT(0)
#define   CTRL_IRQ_EN   BIT(1)
#define   CTRL_LOOPBACK BIT(4)
#define REG_STATUS      0x08
#define   ST_READY      BIT(0)
#define   ST_IRQ        BIT(1)
#define REG_DATA        0x0C

#define EXPECTED_ID     0x57494447   /* "WIDG" */

struct widget {
        struct device    *dev;
        void __iomem     *regs;
        struct clk       *clk;
        int               irq;
        u32               fifo_depth;
        bool              loopback;
        atomic_t          irq_count;
        struct dentry    *dbg;
};

static irqreturn_t widget_isr(int irq, void *data)
{
        struct widget *w = data;
        u32 status = readl(w->regs + REG_STATUS);

        if (!(status & ST_IRQ))
                return IRQ_NONE;          /* shared line: not ours */

        writel(ST_IRQ, w->regs + REG_STATUS);   /* W1C at the source */
        atomic_inc(&w->irq_count);
        return IRQ_HANDLED;
}

static int widget_regs_show(struct seq_file *s, void *unused)
{
        struct widget *w = s->private;

        seq_printf(s, "ID      0x%08x\n", readl(w->regs + REG_ID));
        seq_printf(s, "CTRL    0x%08x\n", readl(w->regs + REG_CTRL));
        seq_printf(s, "STATUS  0x%08x\n", readl(w->regs + REG_STATUS));
        seq_printf(s, "irqs    %d\n",     atomic_read(&w->irq_count));
        return 0;
}
DEFINE_SHOW_ATTRIBUTE(widget_regs);

static int widget_probe(struct platform_device *pdev)
{
        struct device *dev = &pdev->dev;
        struct widget *w;
        u32 id, ctrl;
        int ret;

        w = devm_kzalloc(dev, sizeof(*w), GFP_KERNEL);
        if (!w)
                return -ENOMEM;
        w->dev = dev;
        atomic_set(&w->irq_count, 0);

        w->regs = devm_platform_ioremap_resource(pdev, 0);
        if (IS_ERR(w->regs))
                return dev_err_probe(dev, PTR_ERR(w->regs), "ioremap failed\n");

        w->clk = devm_clk_get_optional(dev, NULL);
        if (IS_ERR(w->clk))
                return dev_err_probe(dev, PTR_ERR(w->clk), "clock lookup\n");
        ret = clk_prepare_enable(w->clk);
        if (ret)
                return ret;

        /* Only now is the block clocked, so only now may we read a register. */
        id = readl(w->regs + REG_ID);
        if (id != EXPECTED_ID) {
                dev_err(dev, "bad ID 0x%08x, expected 0x%08x\n", id, EXPECTED_ID);
                ret = -ENODEV;
                goto err_clk;
        }

        if (device_property_read_u32(dev, "elh,fifo-depth", &w->fifo_depth))
                w->fifo_depth = 32;
        w->loopback = device_property_read_bool(dev, "elh,enable-loopback");

        w->irq = platform_get_irq(pdev, 0);
        if (w->irq < 0) {
                ret = w->irq;
                goto err_clk;
        }
        ret = devm_request_irq(dev, w->irq, widget_isr, IRQF_SHARED,
                               dev_name(dev), w);
        if (ret)
                goto err_clk;

        ctrl = CTRL_ENABLE | CTRL_IRQ_EN | (w->loopback ? CTRL_LOOPBACK : 0);
        writel(ctrl, w->regs + REG_CTRL);

        /* wait for the block to come ready, with a bounded timeout */
        ret = readl_poll_timeout(w->regs + REG_STATUS, id, id & ST_READY,
                                 100, 10000);
        if (ret) {
                dev_err(dev, "timed out waiting for READY\n");
                goto err_disable;
        }

        w->dbg = debugfs_create_dir(dev_name(dev), NULL);
        debugfs_create_file("regs", 0444, w->dbg, w, &widget_regs_fops);

        platform_set_drvdata(pdev, w);
        dev_info(dev, "ready: id 0x%08x fifo %u irq %d%s\n",
                 id, w->fifo_depth, w->irq, w->loopback ? " loopback" : "");
        return 0;

err_disable:
        writel(0, w->regs + REG_CTRL);
err_clk:
        clk_disable_unprepare(w->clk);
        return ret;
}

static void widget_remove(struct platform_device *pdev)
{
        struct widget *w = platform_get_drvdata(pdev);

        /* order matters: stop the hardware raising interrupts BEFORE the
         * managed IRQ is released, and before the mapping goes away. */
        writel(0, w->regs + REG_CTRL);
        debugfs_remove_recursive(w->dbg);
        clk_disable_unprepare(w->clk);
}

static const struct of_device_id widget_of_match[] = {
        { .compatible = "elh,widget-v1" },
        { }
};
MODULE_DEVICE_TABLE(of, widget_of_match);

static struct platform_driver widget_driver = {
        .probe  = widget_probe,
        .remove = widget_remove,
        .driver = {
                .name           = "elh-widget",
                .of_match_table = widget_of_match,
        },
};
module_platform_driver(widget_driver);

MODULE_LICENSE("GPL");
MODULE_DESCRIPTION("Example platform driver with MMIO and IRQ");
On older kernels

.remove took an int return value before 6.11 (remove_new during the transition). If your build complains about the function signature, that is why — return 0 and change the prototype to static int.

Testing it

# build and load
make
sudo insmod widget.ko
dmesg | tail
#   elh-widget 9010000.widget: ready: id 0x57494447 fifo 64 irq 42 loopback

# is it bound?
ls -l /sys/bus/platform/drivers/elh-widget/
#   9010000.widget -> ../../../../devices/platform/9010000.widget

# read the register dump
sudo cat /sys/kernel/debug/9010000.widget/regs

# are interrupts arriving?
grep widget /proc/interrupts

# exercise bind/unbind — this catches remove() bugs immediately
echo 9010000.widget | sudo tee /sys/bus/platform/drivers/elh-widget/unbind
echo 9010000.widget | sudo tee /sys/bus/platform/drivers/elh-widget/bind
The unbind/bind loop is the best driver test there is

Run it in a loop a hundred times with KASAN enabled. Nearly every resource leak, use-after-free and missing-quiesce bug in a driver shows up in that loop, and almost nowhere else.

for i in $(seq 100); do
  echo 9010000.widget | sudo tee /sys/bus/platform/drivers/elh-widget/unbind >/dev/null
  echo 9010000.widget | sudo tee /sys/bus/platform/drivers/elh-widget/bind   >/dev/null
done
dmesg | grep -iE 'kasan|use-after|leak|warn'

Extensions to try

  1. Multiple instances. Turn the single global krb into an array so /dev/kringbuf0 through /dev/kringbuf3 all work independently. This forces you to get container_of and per-device state right.
  2. Replace the mutex with a spinlock and an ISR. Add a kernel timer that injects a byte every 100 ms; now the buffer is shared with atomic context and your locking must change.
  3. Add mmap(). Let user space map the ring directly and read with no copy. You will have to think hard about coherency and about what a hostile user could do.
  4. Convert to kfifo. The kernel already has a lock-free ring buffer. Rewriting on top of it teaches you how much a well-chosen kernel primitive removes.
  5. Add a netlink or char-device event channel so user space is notified asynchronously instead of polling.
  6. Wire in DMA. Use dma_alloc_coherent() for the ring and have the platform driver's hardware fill it, using everything from chapter 05.

Project checklist

  • The module loads and unloads cleanly, repeatedly, with no dmesg warnings.
  • krbtest passes every assertion, including the error cases.
  • cat blocks on an empty buffer and unblocks on a write from another terminal.
  • sysfs shows a live, correct fill level.
  • You have caused at least three of the deliberate bugs above and recognised each failure signature.
  • checkpatch.pl --strict is clean.
  • The unbind/bind loop runs 100 times with KASAN and lockdep enabled, silently.
