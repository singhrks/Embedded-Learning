Project: an I2C sensor driver, overlay to userspace

A complete, correct Linux device driver for a real sensor on a real board: a device tree overlay that declares the hardware, a kernel module that binds to it, an IIO interface that any application can consume, and the debugging workflow that gets you there. This is the canonical embedded Linux task.

What you build

   HARDWARE: a BME280 (temperature, pressure, humidity) on
   I2C1 at address 0x76. About three pounds. The same sensor
   as the buses track's project (bus chapter 21) - which
   wrote a BARE-METAL driver for it. Doing both is the point:
   the same chip, the same registers, two completely
   different software worlds.

   WIRING (bus chapter 06 for why no pull-ups needed):
     Pi pin 1  (3V3)  -- VCC
     Pi pin 3  (GPIO2, SDA) -- SDA
     Pi pin 5  (GPIO3, SCL) -- SCL
     Pi pin 9  (GND)  -- GND

   THE STACK YOU WILL BUILD:

   +--------------------------------------------------+
   |  userspace:  cat /sys/bus/iio/devices/...         |
   |              or libiio, or your own app           |
   +--------------------------------------------------+
   |  IIO subsystem - generic sensor interface         |
   +--------------------------------------------------+
   |  YOUR DRIVER  bme280_probe(), read_raw()          |
   +--------------------------------------------------+
   |  i2c-core - bus arbitration, locking              |
   +--------------------------------------------------+
   |  i2c-bcm2835 / rp1 - the controller driver        |
   +--------------------------------------------------+
   |  the wire                                         |
   +--------------------------------------------------+

   NOTE what you do NOT write: bus arbitration, locking,
   power management, the userspace ABI, or anything about
   the controller. That is the value of the subsystem, and
   the reason a kernel driver is ~200 lines where the bare
   metal version was ~400.

   A REAL bme280 DRIVER SHIPS IN THE KERNEL. We are writing
   our own deliberately, under a different compatible
   string, so nothing binds first and you see every step.

Step 0: prove the hardware before writing code

# Never write a driver for a device you have not talked to
# by hand. Ten minutes here saves a day of debugging the

#                     ^^ present, unclaimed

/dts-v1/;
/plugin/;

/ {
    compatible = "brcm,bcm2711", "brcm,bcm2712";

    fragment@0 {
        target = <&i2c1>;
        __overlay__ {
            status = "okay";
            #address-cells = <1>;
            #size-cells = <0>;

            bme280_custom: bme280@76 {
                /* OUR OWN compatible string, so the

                compatible = "learn,bme280-custom";
                reg = <0x76>;
                status = "okay";
            };
        };
    };

    __overrides__ {
        addr = <&bme280_custom>,"reg:0";
    };
};

Step 2: the driver

// SPDX-License-Identifier: GPL-2.0
/* bme280_custom.c - an IIO driver for the BME280 over I2C.
   Compare every part of this with bus chapter 21's bare
   metal version.                                           */

#include <linux/module.h>
#include <linux/i2c.h>
#include <linux/mod_devicetable.h>
#include <linux/regmap.h>
#include <linux/iio/iio.h>
#include <linux/pm_runtime.h>

#define BME_REG_ID        0xD0
#define BME_REG_RESET     0xE0
#define BME_REG_CTRL_HUM  0xF2
#define BME_REG_STATUS    0xF3
#define BME_REG_CTRL_MEAS 0xF4
#define BME_REG_CONFIG    0xF5
#define BME_REG_PRESS_MSB 0xF7
#define BME_CHIP_ID       0x60

struct bme_calib {
	u16 t1; s16 t2, t3;
	u16 p1; s16 p2, p3, p4, p5, p6, p7, p8, p9;
	u8  h1; s16 h2; u8 h3; s16 h4, h5; s8 h6;
	s32 t_fine;              /* shared between compensations */
};

struct bme_data {
	struct regmap    *regmap;
	struct mutex      lock;   /* serialises a burst read +
				     t_fine, which is per-device
				     shared state                */
	struct bme_calib  cal;
};

/* regmap does the I2C register access for us: locking,
   caching, endianness, debugfs. Writing raw i2c_transfer
   calls by hand is what bus chapter 21 did; here it is
   three lines of configuration.                            */
static const struct regmap_config bme_regmap_config = {
	.reg_bits     = 8,
	.val_bits     = 8,
	.max_register = 0xFF,
	.cache_type   = REGCACHE_NONE,   /* data regs change */
};

/* --- compensation, straight from the datasheet --- */
static s32 bme_comp_temp(struct bme_calib *c, s32 adc_t)
{
	s32 v1 = ((((adc_t >> 3) - ((s32)c->t1 << 1))) * (s32)c->t2) >> 11;
	s32 v2 = (((((adc_t >> 4) - (s32)c->t1) *
		    ((adc_t >> 4) - (s32)c->t1)) >> 12) * (s32)c->t3) >> 14;
	c->t_fine = v1 + v2;
	return (c->t_fine * 5 + 128) >> 8;      /* milli-degC/10 */
}

/* --- the IIO channel table: what this device exposes --- */
static const struct iio_chan_spec bme_channels[] = {
	{
		.type = IIO_TEMP,
		.info_mask_separate = BIT(IIO_CHAN_INFO_PROCESSED),
	},
	{
		.type = IIO_PRESSURE,
		.info_mask_separate = BIT(IIO_CHAN_INFO_PROCESSED),
	},
	{
		.type = IIO_HUMIDITYRELATIVE,
		.info_mask_separate = BIT(IIO_CHAN_INFO_PROCESSED),
	},
};

static int bme_read_raw(struct iio_dev *indio,
			struct iio_chan_spec const *chan,
			int *val, int *val2, long mask)
{
	struct bme_data *d = iio_priv(indio);
	u8 buf[8];
	s32 adc_t, adc_p, adc_h, t;
	int ret;

	if (mask != IIO_CHAN_INFO_PROCESSED)
		return -EINVAL;

	mutex_lock(&d->lock);

	/* ONE burst read of all 8 data registers. The sensor
	   shadows them during a burst, so reading separately
	   can mix two different measurements - the same
	   correctness point as bus chapter 21.               */
	ret = regmap_bulk_read(d->regmap, BME_REG_PRESS_MSB, buf, 8);
	if (ret)
		goto out;

	adc_p = (buf[0] << 12) | (buf[1] << 4) | (buf[2] >> 4);
	adc_t = (buf[3] << 12) | (buf[4] << 4) | (buf[5] >> 4);
	adc_h = (buf[6] << 8)  |  buf[7];

	if (adc_t == 0x80000) { ret = -EAGAIN; goto out; }

	/* temperature FIRST - it computes t_fine, which the
	   others consume.                                     */
	t = bme_comp_temp(&d->cal, adc_t);

	switch (chan->type) {
	case IIO_TEMP:
		*val = t * 10;            /* IIO wants milli-degC */
		break;
	case IIO_PRESSURE:
		*val = bme_comp_press(&d->cal, adc_p);
		break;
	case IIO_HUMIDITYRELATIVE:
		*val = bme_comp_hum(&d->cal, adc_h);
		break;
	default:
		ret = -EINVAL; goto out;
	}
	ret = IIO_VAL_INT;
out:
	mutex_unlock(&d->lock);
	return ret;
}

static const struct iio_info bme_info = {
	.read_raw = bme_read_raw,
};

/* --- probe: called when a device matches our compatible --- */
static int bme_probe(struct i2c_client *client)
{
	struct device   *dev = &client->dev;
	struct iio_dev  *indio;
	struct bme_data *d;
	unsigned int     id;
	int              ret;

	/* devm_* allocations are freed automatically when the
	   device goes away. Using them removes the entire
	   class of unwind bugs that chapter 16 warned about. */
	indio = devm_iio_device_alloc(dev, sizeof(*d));
	if (!indio)
		return -ENOMEM;

	d = iio_priv(indio);
	mutex_init(&d->lock);

	d->regmap = devm_regmap_init_i2c(client, &bme_regmap_config);
	if (IS_ERR(d->regmap))
		return dev_err_probe(dev, PTR_ERR(d->regmap),
				     "regmap init failed\n");

	/* IDENTIFY THE DEVICE. A probe that does not verify it
	   is talking to what it expects will bind to anything
	   and fail confusingly later.                         */
	ret = regmap_read(d->regmap, BME_REG_ID, &id);
	if (ret)
		return dev_err_probe(dev, ret, "cannot read chip id\n");
	if (id != BME_CHIP_ID)
		return dev_err_probe(dev, -ENODEV,
				     "wrong chip id 0x%02x\n", id);

	ret = bme_read_calibration(d);
	if (ret)
		return dev_err_probe(dev, ret, "calibration read failed\n");

	/* CTRL_HUM must be written BEFORE CTRL_MEAS - a
	   datasheet footnote that costs an afternoon if missed
	   (bus chapter 21).                                   */
	regmap_write(d->regmap, BME_REG_CTRL_HUM,  0x01);
	regmap_write(d->regmap, BME_REG_CONFIG,    0xA0);
	regmap_write(d->regmap, BME_REG_CTRL_MEAS, 0x27);

	indio->name      = "bme280_custom";
	indio->info      = &bme_info;
	indio->modes     = INDIO_DIRECT_MODE;
	indio->channels  = bme_channels;
	indio->num_channels = ARRAY_SIZE(bme_channels);

	dev_info(dev, "BME280 registered\n");
	return devm_iio_device_register(dev, indio);
	/* No remove() function needed at all - devm handles it.
	   Contrast with chapter 16's manual unwind.           */
}

static const struct of_device_id bme_of_match[] = {
	{ .compatible = "learn,bme280-custom" },
	{ }
};
MODULE_DEVICE_TABLE(of, bme_of_match);

static const struct i2c_device_id bme_id[] = {
	{ "bme280-custom", 0 },
	{ }
};
MODULE_DEVICE_TABLE(i2c, bme_id);

static struct i2c_driver bme_driver = {
	.driver = {
		.name           = "bme280_custom",
		.of_match_table = bme_of_match,
	},
	.probe    = bme_probe,
	.id_table = bme_id,
};
module_i2c_driver(bme_driver);
/* module_i2c_driver expands to the init/exit boilerplate.
   The whole module is now declarative: a match table, a
   probe, and the channels it exposes.                     */

MODULE_LICENSE("GPL");
MODULE_AUTHOR("you");
MODULE_DESCRIPTION("BME280 IIO driver (teaching version)");

What the subsystem gave you

You wrote about 200 lines and got: a stable userspace ABI that predates your driver and will outlive it, automatic device/driver matching from the device tree, bus locking shared with every other I2C device, sysfs attribute plumbing, hotplug and udev integration, and lifetime management through devm_. The bare-metal version in bus chapter 21 had to invent all of it, and whatever it invented worked only with itself. That difference is the entire argument for driver frameworks, and now you have written both sides of it.

Debugging it

# PROBE NEVER CALLED?
$ ls /sys/bus/i2c/drivers/bme280_custom/
# Empty = the driver loaded but matched nothing. Check that
# the compatible string in the .dts EXACTLY matches
# bme_of_match. A single character difference gives you
# silence, not an error.

# See matching happen (chapter 17):
$ echo 'file drivers/i2c/i2c-core-base.c +p' | \
      sudo tee /sys/kernel/debug/dynamic_debug/control
$ sudo rmmod bme280_custom && sudo insmod bme280_custom.ko
$ dmesg | tail -20

# WATCH THE ACTUAL BUS TRAFFIC:
$ sudo bpftrace -e '
    kprobe:i2c_transfer { printf("%s: %d msgs\n", comm, arg2); }'

# Or trace the controller driver's transfer function and get
# durations (chapter 17):
$ cd /sys/kernel/tracing
$ echo function_graph | sudo tee current_tracer
$ echo 'bcm2835_i2c_*' | sudo tee set_ftrace_filter
$ echo 1 | sudo tee tracing_on; cat /sys/bus/iio/devices/iio:device0/in_temp_input
$ sudo cat trace | head -20

# regmap gives you a free register-level debug view:
$ ls /sys/kernel/debug/regmap/1-0076/
$ sudo cat /sys/kernel/debug/regmap/1-0076/registers

# PROBE DEFERRED? dev_err_probe logs it properly and retries
# later - which is correct behaviour when a dependency is
# not ready yet, and confusing if you do not know it exists:
$ dmesg | grep -i defer
$ cat /sys/kernel/debug/devices_deferred

Exercises

  • Compare with the in-tree driver. Read drivers/iio/pressure/bmp280-core.c in the kernel source. It handles more variants, oversampling configuration, power management and buffered reads. Identify three things it does that yours does not, and why each matters.
  • Add sysfs attributes for oversampling, exposed through IIO_CHAN_INFO_OVERSAMPLING_RATIO, so userspace can trade accuracy against time.
  • Add runtime PM so the sensor is put into sleep mode between reads and woken on demand. Measure the current difference.
  • Add a triggered buffer so IIO can sample at a fixed rate into a ring buffer that userspace reads with one syscall — the interface real data acquisition uses.
  • Break it deliberately. Disconnect SDA while a read is in flight. Confirm you get a clean -EREMOTEIO rather than a hang, and that the driver recovers when reconnected.
  • Cross-compile it (chapter 06) and deploy with the sync loop from chapter 07.
  • Package it as a .deb with the overlay and a modules-load config, so a fresh Pi is one apt install from working (chapter 07).
