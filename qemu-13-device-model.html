Writing a QEMU device model

When the silicon does not exist yet, you write the device first and the driver against it. This chapter builds a complete, working QEMU device — registers, interrupt, device tree node, migration state — and then a Linux driver for it, so you have the full loop that hardware bring-up teams actually use.

Why you would do this

SituationWhat the model buys you
Silicon is six months awayThe driver is written, reviewed and tested before tape-out. This is the normal way large SoC drivers are developed
The board is on someone else's deskEvery engineer gets a target
You need to test an error pathMake the model return the error. No fault injector on real hardware can do this as precisely
CI must run driver testsNo board farm required
You are learningWriting the device teaches you what the driver's assumptions actually are

QOM: QEMU's object model, in one page

   Every device is a QOM object. Three things to declare:

   1. A STATE struct - your instance data, beginning with the
      parent's state struct.

        struct MyDevState {
                SysBusDevice parent_obj;   <- MUST be first
                MemoryRegion iomem;
                qemu_irq     irq;
                uint32_t     regs[8];
        };

   2. A TYPE registration - name, parent, sizes, class init.

        static const TypeInfo mydev_info = {
                .name          = TYPE_MYDEV,
                .parent        = TYPE_SYS_BUS_DEVICE,
                .instance_size = sizeof(MyDevState),
                .instance_init = mydev_init,
                .class_init    = mydev_class_init,
        };

   3. LIFECYCLE callbacks:

        instance_init  object created. Set defaults only.
        realize        properties are set; NOW create memory
                       regions, allocate, connect IRQs. This is
                       where the real work goes.
        reset          system reset. Restore register defaults.
        unrealize      teardown.

   Parent classes you will use:
        TYPE_SYS_BUS_DEVICE   memory-mapped, on the system bus
        TYPE_PCI_DEVICE       a PCIe function
        TYPE_VIRTIO_DEVICE    a virtio device
        TYPE_DEVICE           anything else

A complete device

/* hw/misc/acme-widget.c
   A device with an ID register, a scratch register, a counter
   that increments on read, and an interrupt you can trigger. */

#include "qemu/osdep.h"
#include "hw/sysbus.h"
#include "hw/irq.h"
#include "hw/qdev-properties.h"
#include "migration/vmstate.h"
#include "qemu/log.h"
#include "qom/object.h"
#include "trace.h"

#define TYPE_ACME_WIDGET "acme-widget"
OBJECT_DECLARE_SIMPLE_TYPE(AcmeWidgetState, ACME_WIDGET)

/* Register map - document it here, it is the contract. */
#define R_ID        0x00   /* RO  magic 0xACE00001            */
#define R_CTRL      0x04   /* RW  bit0 enable, bit1 irq enable */
#define R_STATUS    0x08   /* RO  bit0 irq pending             */
#define R_SCRATCH   0x0c   /* RW  anything                     */
#define R_COUNTER   0x10   /* RO  increments on each read      */
#define R_IRQ_SET   0x14   /* WO  write 1 to raise the irq     */
#define R_IRQ_CLR   0x18   /* WO  write 1 to clear it          */
#define R_MAX       0x1c

#define CTRL_ENABLE   (1u << 0)
#define CTRL_IRQ_EN   (1u << 1)
#define STATUS_IRQ    (1u << 0)

struct AcmeWidgetState {
        SysBusDevice parent_obj;

        MemoryRegion iomem;
        qemu_irq     irq;

        uint32_t ctrl;
        uint32_t status;
        uint32_t scratch;
        uint32_t counter;
        uint32_t id;          /* a property, so it is settable */
};

static void acme_update_irq(AcmeWidgetState *s)
{
        int level = !!(s->status & STATUS_IRQ) &&
                    !!(s->ctrl & CTRL_IRQ_EN);
        trace_acme_widget_irq(level);
        qemu_set_irq(s->irq, level);      /* LEVEL, not pulse */
}

static uint64_t acme_read(void *opaque, hwaddr off, unsigned size)
{
        AcmeWidgetState *s = opaque;
        uint64_t v = 0;

        switch (off) {
        case R_ID:      v = s->id;                  break;
        case R_CTRL:    v = s->ctrl;                break;
        case R_STATUS:  v = s->status;              break;
        case R_SCRATCH: v = s->scratch;             break;
        case R_COUNTER: v = s->counter++;           break;
        default:
                /* NEVER fault. Log and return zero, like real
                   silicon. */
                qemu_log_mask(LOG_GUEST_ERROR,
                        "acme-widget: read of unknown offset 0x%"
                        HWADDR_PRIx "\n", off);
                break;
        }
        trace_acme_widget_read(off, v, size);
        return v;
}

static void acme_write(void *opaque, hwaddr off, uint64_t val,
                       unsigned size)
{
        AcmeWidgetState *s = opaque;

        trace_acme_widget_write(off, val, size);

        switch (off) {
        case R_CTRL:
                s->ctrl = val & (CTRL_ENABLE | CTRL_IRQ_EN);
                acme_update_irq(s);
                break;
        case R_SCRATCH:
                s->scratch = val;
                break;
        case R_IRQ_SET:
                if (val & 1) {
                        s->status |= STATUS_IRQ;
                        acme_update_irq(s);
                }
                break;
        case R_IRQ_CLR:
                if (val & 1) {
                        s->status &= ~STATUS_IRQ;
                        acme_update_irq(s);
                }
                break;
        case R_ID:
        case R_STATUS:
        case R_COUNTER:
                qemu_log_mask(LOG_GUEST_ERROR,
                        "acme-widget: write to RO offset 0x%"
                        HWADDR_PRIx "\n", off);
                break;
        default:
                qemu_log_mask(LOG_GUEST_ERROR,
                        "acme-widget: write to unknown offset 0x%"
                        HWADDR_PRIx "\n", off);
                break;
        }
}

static const MemoryRegionOps acme_ops = {
        .read       = acme_read,
        .write      = acme_write,
        .endianness = DEVICE_LITTLE_ENDIAN,
        .valid = {                 /* what the GUEST may do    */
                .min_access_size = 4,
                .max_access_size = 4,
                .unaligned       = false,
        },
        .impl = {                  /* what WE implement; QEMU
                                      splits/combines to match */
                .min_access_size = 4,
                .max_access_size = 4,
        },
};

static void acme_reset(DeviceState *dev)
{
        AcmeWidgetState *s = ACME_WIDGET(dev);

        s->ctrl    = 0;
        s->status  = 0;
        s->scratch = 0;
        s->counter = 0;
        acme_update_irq(s);
}

static void acme_realize(DeviceState *dev, Error **errp)
{
        AcmeWidgetState *s = ACME_WIDGET(dev);
        SysBusDevice   *sbd = SYS_BUS_DEVICE(dev);

        memory_region_init_io(&s->iomem, OBJECT(s), &acme_ops, s,
                              TYPE_ACME_WIDGET, 0x1000);
        sysbus_init_mmio(sbd, &s->iomem);
        sysbus_init_irq(sbd, &s->irq);
}

/* Migration state. Without this, savevm/loadvm and migration
   silently lose your device's registers.                     */
static const VMStateDescription acme_vmstate = {
        .name    = TYPE_ACME_WIDGET,
        .version_id = 1,
        .minimum_version_id = 1,
        .fields = (const VMStateField[]) {
                VMSTATE_UINT32(ctrl,    AcmeWidgetState),
                VMSTATE_UINT32(status,  AcmeWidgetState),
                VMSTATE_UINT32(scratch, AcmeWidgetState),
                VMSTATE_UINT32(counter, AcmeWidgetState),
                VMSTATE_END_OF_LIST()
        },
};

static const Property acme_properties[] = {
        DEFINE_PROP_UINT32("id", AcmeWidgetState, id, 0xACE00001),
};

static void acme_class_init(ObjectClass *klass, void *data)
{
        DeviceClass *dc = DEVICE_CLASS(klass);

        dc->realize  = acme_realize;
        device_class_set_legacy_reset(dc, acme_reset);
        dc->vmsd     = &acme_vmstate;
        dc->desc     = "Acme Widget";
        device_class_set_props(dc, acme_properties);
        set_bit(DEVICE_CATEGORY_MISC, dc->categories);
}

static const TypeInfo acme_info = {
        .name          = TYPE_ACME_WIDGET,
        .parent        = TYPE_SYS_BUS_DEVICE,
        .instance_size = sizeof(AcmeWidgetState),
        .class_init    = acme_class_init,
};

static void acme_register_types(void)
{
        type_register_static(&acme_info);
}
type_init(acme_register_types)

Test the model before writing the driver

xp from the monitor exercises every register without a single line of guest code. If the ID register does not read back, the problem is in your model; if it does, every later failure is in the driver. That separation saves hours, and it is only available because you are on QEMU.

The device tree node

/* A dynamic sysbus device gets an auto-generated node with a
   generic compatible string, which no driver will match. For a
   real device you add it to the machine.                     */

/* hw/arm/virt.c, in the machine's device creation: */

static void create_acme_widget(const VirtMachineState *vms)
{
        hwaddr base = 0x0c000000;
        int    irq  = 48;
        DeviceState *dev;
        MachineState *ms = MACHINE(vms);
        char *nodename;

        dev = qdev_new(TYPE_ACME_WIDGET);
        sysbus_realize_and_unref(SYS_BUS_DEVICE(dev), &error_fatal);
        sysbus_mmio_map(SYS_BUS_DEVICE(dev), 0, base);
        sysbus_connect_irq(SYS_BUS_DEVICE(dev), 0,
                           qdev_get_gpio_in(vms->gic,
                                            irq));

        nodename = g_strdup_printf("/acme-widget@%" PRIx64, base);
        qemu_fdt_add_subnode(ms->fdt, nodename);
        qemu_fdt_setprop_string(ms->fdt, nodename,
                                "compatible", "acme,widget-1.0");
        qemu_fdt_setprop_sized_cells(ms->fdt, nodename, "reg",
                                     2, base, 2, 0x1000);
        qemu_fdt_setprop_cells(ms->fdt, nodename, "interrupts",
                               GIC_FDT_IRQ_TYPE_SPI, irq,
                               GIC_FDT_IRQ_FLAGS_LEVEL_HI);
        g_free(nodename);
}

/* Verify it landed: */
$ qemu-system-aarch64 -M virt,dumpdtb=v.dtb ... && \
  dtc -I dtb -O dts v.dtb | grep -A5 acme

The driver, for the loop to close

// acme_widget.c - a Linux platform driver for the model above.
#include <linux/module.h>
#include <linux/platform_device.h>
#include <linux/of.h>
#include <linux/io.h>
#include <linux/interrupt.h>

#define R_ID 0x00
#define R_CTRL 0x04
#define R_STATUS 0x08
#define R_SCRATCH 0x0c
#define R_COUNTER 0x10
#define R_IRQ_SET 0x14
#define R_IRQ_CLR 0x18

struct acme {
        void __iomem *base;
        int irq;
};

static irqreturn_t acme_isr(int irq, void *dev_id)
{
        struct acme *a = dev_id;

        writel(1, a->base + R_IRQ_CLR);     /* level: must clear */
        pr_info("acme: interrupt\n");
        return IRQ_HANDLED;
}

static int acme_probe(struct platform_device *pdev)
{
        struct acme *a;
        u32 id;

        a = devm_kzalloc(&pdev->dev, sizeof(*a), GFP_KERNEL);
        if (!a)
                return -ENOMEM;

        a->base = devm_platform_ioremap_resource(pdev, 0);
        if (IS_ERR(a->base))
                return PTR_ERR(a->base);

        id = readl(a->base + R_ID);
        if (id != 0xACE00001) {
                dev_err(&pdev->dev, "bad id 0x%08x\n", id);
                return -ENODEV;
        }

        a->irq = platform_get_irq(pdev, 0);
        if (a->irq < 0)
                return a->irq;

        if (devm_request_irq(&pdev->dev, a->irq, acme_isr, 0,
                             "acme-widget", a))
                return -EBUSY;

        writel(3, a->base + R_CTRL);        /* enable + irq en   */
        platform_set_drvdata(pdev, a);

        dev_info(&pdev->dev, "probed, id 0x%08x, counter %u\n",
                 id, readl(a->base + R_COUNTER));

        writel(1, a->base + R_IRQ_SET);     /* fire one          */
        return 0;
}

static const struct of_device_id acme_of_match[] = {
        { .compatible = "acme,widget-1.0" },
        { }
};
MODULE_DEVICE_TABLE(of, acme_of_match);

static struct platform_driver acme_driver = {
        .probe  = acme_probe,
        .driver = {
                .name           = "acme-widget",
                .of_match_table = acme_of_match,
        },
};
module_platform_driver(acme_driver);
MODULE_LICENSE("GPL");

Level, not pulse

Model the interrupt as a level that you recompute from state (qemu_set_irq(irq, level)), never as a pulse. A pulse can be lost, and worse, it hides driver bugs: a driver that forgets to clear the interrupt source works fine against a pulse model and locks the machine up on real hardware. The model should be as unforgiving as the silicon.

SymptomCause
Device does not appear in info qtreeNot in meson.build, or CONFIG_ not enabled, or not rebuilt
