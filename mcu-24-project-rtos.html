Project: a preemptive RTOS from scratch

About 300 lines gives you a real preemptive kernel: fixed-priority scheduling, blocking delays, mutexes with priority inheritance, and message queues. Writing one is the fastest way to stop finding FreeRTOS mysterious, because every line of it will be something you have already decided.

Scope

   WHAT IT HAS                    WHAT IT DELIBERATELY LACKS

   fixed-priority preemption      dynamic task creation
   round-robin within a priority  task deletion
   k_sleep with a tick wheel      software timers
   counting semaphores            event groups
   mutexes + priority inheritance recursive mutexes
   fixed-size message queues      stack overflow detection
   idle task with WFI             (add the MPU guard from
                                   chapter 14 as an exercise)

   Everything is STATICALLY allocated - tasks are declared at
   compile time. That removes an entire class of failure and
   keeps the code readable.

   Target: any Cortex-M3/M4. Runs under QEMU:
     qemu-system-arm -M netduinoplus2 -cpu cortex-m4 \
                     -nographic -kernel kernel.elf

Part 1: the task control block

/* kernel.h */

#define MAX_TASKS      8
#define MAX_PRIO       8

typedef enum {
    T_READY, T_RUNNING, T_BLOCKED, T_SLEEPING
} tstate_t;

typedef struct tcb {
    uint32_t     *sp;          /* MUST be offset 0 - ch 18   */
    tstate_t      state;
    uint8_t       prio;        /* current, may be inherited  */
    uint8_t       base_prio;   /* what it was declared with  */
    uint32_t      wake_tick;
    struct tcb   *next;        /* ready or wait list link    */
    void         *wait_obj;    /* what it is blocked on      */
    uint32_t     *stack_base;
    uint32_t      stack_words;
    const char   *name;
} tcb_t;

extern tcb_t *current_tcb;     /* the assembly reads this    */

/* Ready lists: one singly-linked list per priority, plus a
   bitmap so "highest ready priority" is one instruction.   */
static tcb_t   *ready[MAX_PRIO];
static uint32_t ready_bits;

static inline int highest_ready(void)
{
    /* CLZ: count leading zeros. Bit 31 = priority 0.
       31 - clz gives the highest set bit, and one
       instruction replaces a loop. This is exactly the
       trick FreeRTOS uses in its optimised port.           */
    return ready_bits ? (31 - __builtin_clz(ready_bits)) : -1;
}

static void ready_add(tcb_t *t)
{
    t->state = T_READY;
    t->next  = ready[t->prio];
    ready[t->prio] = t;
    ready_bits |= (1u << t->prio);
}

static tcb_t *ready_pop(int prio)
{
    tcb_t *t = ready[prio];
    if (t) {
        ready[prio] = t->next;
        if (!ready[prio]) { ready_bits &= ~(1u << prio); }
    }
    return t;
}

Part 2: the scheduler

/* This is the C function chapter 18's assembly calls. All
   the POLICY is here; the assembly is pure mechanism.     */

void scheduler_next(void)
{
    /* Put the outgoing task back if it is still runnable. */
    if (current_tcb && current_tcb->state == T_RUNNING) {
        ready_add(current_tcb);
    }

    int p = highest_ready();
    /* p is never -1: the idle task is always ready at
       priority 0, which is why an idle task must exist.   */
    tcb_t *next = ready_pop(p);
    next->state = T_RUNNING;
    current_tcb = next;
}

/* Requesting a switch. One store, callable from anywhere -
   task code or any ISR (chapter 18).                      */
static inline void yield(void)
{
    SCB->ICSR = SCB_ICSR_PENDSVSET_Msk;
    __DSB(); __ISB();
}

/* The tick: accounting plus wake-ups. */
volatile uint32_t g_ticks;

void SysTick_Handler(void)
{
    g_ticks++;
    int woke = 0;

    for (int i = 0; i < MAX_TASKS; i++) {
        tcb_t *t = &g_tasks[i];
        if (t->state == T_SLEEPING &&
            (int32_t)(g_ticks - t->wake_tick) >= 0) {  /* wrap-safe */
            ready_add(t);
            woke = 1;
        }
    }

    /* Switch if someone woke, or to round-robin the current
       priority level.                                      */
    if (woke || ready[current_tcb->prio]) {
        yield();
    }
}

/* The O(MAX_TASKS) scan is fine for 8 tasks and wrong for
   200. A real kernel keeps a delta list sorted by wake time
   so the tick is O(1). Worth implementing as an exercise -
   it is the difference between a toy and a kernel.        */

Part 3: the switch

/* Chapter 18's handler, unchanged. This is the only
   assembly in the project.                                */

__attribute__((naked)) void PendSV_Handler(void)
{
    __asm volatile (
    "   mrs   r0, psp             \n"
    "   cbz   r0, 1f              \n"
    "   subs  r0, r0, #32         \n"
    "   stm   r0, {r4-r11}        \n"
    "   ldr   r1, =current_tcb    \n"
    "   ldr   r2, [r1]            \n"
    "   str   r0, [r2]            \n"   /* sp at offset 0    */
    "1:                           \n"
    "   push  {lr}                \n"
    "   bl    scheduler_next      \n"
    "   pop   {lr}                \n"
    "   ldr   r1, =current_tcb    \n"
    "   ldr   r2, [r1]            \n"
    "   ldr   r0, [r2]            \n"
    "   ldm   r0, {r4-r11}        \n"
    "   adds  r0, r0, #32         \n"
    "   msr   psp, r0             \n"
    "   isb                       \n"
    "   bx    lr                  \n"
    );
}

/* Task creation: fabricate a frame (chapter 18). */
void task_create(tcb_t *t, const char *name,
                 void (*entry)(void *), void *arg,
                 uint8_t prio, uint32_t *stack, uint32_t words)
{
    for (uint32_t i = 0; i < words; i++) {
        stack[i] = 0xA5A5A5A5u;              /* watermark    */
    }

    uint32_t *sp = (uint32_t *)(((uint32_t)(stack + words)) & ~7u);
    *(--sp) = 0x01000000u;                   /* xPSR: T BIT  */
    *(--sp) = (uint32_t) entry & ~1u;        /* PC           */
    *(--sp) = (uint32_t) task_exit;          /* LR           */
    *(--sp) = 0; *(--sp) = 0; *(--sp) = 0; *(--sp) = 0;
    *(--sp) = (uint32_t) arg;                /* R0           */
    for (int i = 0; i < 8; i++) { *(--sp) = 0; }  /* R4-R11  */

    t->sp = sp;  t->name = name;
    t->prio = t->base_prio = prio;
    t->stack_base = stack;  t->stack_words = words;
    ready_add(t);
}

Part 4: blocking, and a mutex that inherits

/* Critical sections use BASEPRI, so high-priority interrupts
   keep running through them (chapters 03, 08).            */

#define KERNEL_PRIO  (5u << (8 - __NVIC_PRIO_BITS))

static inline uint32_t enter_critical(void)
{
    uint32_t old = __get_BASEPRI();
    __set_BASEPRI(KERNEL_PRIO);
    __DSB(); __ISB();
    return old;
}
static inline void exit_critical(uint32_t old)
{
    __set_BASEPRI(old);
}

void k_sleep(uint32_t ms)
{
    uint32_t s = enter_critical();
    current_tcb->wake_tick = g_ticks + ms;
    current_tcb->state     = T_SLEEPING;
    exit_critical(s);
    yield();                      /* switches away here     */
}

/* MUTEX WITH PRIORITY INHERITANCE - the feature that
   distinguishes a mutex from a semaphore (chapter 19).    */

typedef struct {
    tcb_t *owner;
    tcb_t *waiters;
    int    locked;
} mutex_t;

void mutex_lock(mutex_t *m)
{
    uint32_t s = enter_critical();

    if (!m->locked) {
        m->locked = 1;
        m->owner  = current_tcb;
        exit_critical(s);
        return;
    }

    /* THE INHERITANCE. If we outrank the owner, lend it our
       priority so it cannot be preempted by anything in
       between - which is exactly the unbounded inversion of
       the concurrency track, chapter 15.                   */
    if (current_tcb->prio > m->owner->prio) {
        /* remove, re-prioritise, re-insert: the owner may be
           sitting in a ready list under its old priority.   */
        ready_remove(m->owner);
        m->owner->prio = current_tcb->prio;
        if (m->owner->state == T_READY) { ready_add(m->owner); }
    }

    current_tcb->state    = T_BLOCKED;
    current_tcb->wait_obj = m;
    current_tcb->next     = m->waiters;
    m->waiters            = current_tcb;

    exit_critical(s);
    yield();
}

void mutex_unlock(mutex_t *m)
{
    uint32_t s = enter_critical();

    /* Drop any inherited priority first. */
    current_tcb->prio = current_tcb->base_prio;

    tcb_t *best = NULL, **link = &m->waiters;
    for (tcb_t **p = &m->waiters; *p; p = &(*p)->next) {
        if (!best || (*p)->prio > best->prio) { best = *p; link = p; }
    }

    if (best) {
        *link     = best->next;
        m->owner  = best;
        ready_add(best);
    } else {
        m->locked = 0;
        m->owner  = NULL;
    }

    exit_critical(s);
    yield();
}
Where this kernel is honestly simplified

Inheritance here is one level deep: if the owner is itself blocked on a second mutex, the boost does not propagate down the chain. Real kernels handle that transitively. The kernel also has no deadlock detection, no timeout on mutex_lock, and an O(n) tick. Each is a deliberate omission you can now see the shape of — which is the point of writing one. Read FreeRTOS's xTaskPriorityInherit afterwards and it will read as a careful version of code you have already written.

Part 5: it running

static uint32_t s_hi[256], s_lo[256], s_idle[64];
static tcb_t    t_hi, t_lo, t_idle;
static mutex_t  bus;

static void hi_task(void *a)
{
    for (;;) {
        k_sleep(100);
        uint32_t t0 = DWT->CYCCNT;
        mutex_lock(&bus);
        use_shared_bus();
        mutex_unlock(&bus);
        record_max(DWT->CYCCNT - t0);   /* blocking time     */
    }
}

static void lo_task(void *a)
{
    for (;;) {
        mutex_lock(&bus);
        busy_work(50);           /* hold it a long time      */
        mutex_unlock(&bus);
        k_sleep(10);
    }
}

static void idle_task(void *a) { for (;;) { __WFI(); } }

int main(void)
{
    clock_init();  uart_init(115200);  cycle_counter_init();

    task_create(&t_hi,   "hi",   hi_task,   NULL, 5, s_hi,   256);
    task_create(&t_lo,   "lo",   lo_task,   NULL, 2, s_lo,   256);
    task_create(&t_idle, "idle", idle_task, NULL, 0, s_idle,  64);

    NVIC_SetPriority(PendSV_IRQn,  0xFF);   /* LOWEST - ch 18 */
    NVIC_SetPriority(SysTick_IRQn, 0xFE);

    current_tcb = ready_pop(highest_ready());
    current_tcb->state = T_RUNNING;
    __set_PSP((uint32_t) current_tcb->sp + 32);
    __set_CONTROL(0x02);
    __ISB();                                /* MANDATORY      */

    SysTick->LOAD = (SystemCoreClock / 1000u) - 1u;
    SysTick->VAL  = 0;
    SysTick->CTRL = 7u;

    ((void (*)(void *)) hi_task)(NULL);      /* never returns  */
    for (;;) { }
}

/* THE EXPERIMENT WORTH RUNNING: add a medium-priority task
   at priority 3 that does nothing but busy_work. Then
   disable the inheritance block in mutex_lock and compare
   the recorded worst-case blocking time of hi_task.

   Without inheritance it grows without bound as the medium
   task takes more CPU - that is unbounded priority
   inversion, reproduced on your own desk in about twenty
   lines. Re-enable inheritance and it collapses to the
   length of lo_task's critical section. Seeing that number
   change is worth more than any amount of reading about
   Mars Pathfinder.                                         */

Exercises

  • Reproduce unbounded inversion as described above, and plot worst-case blocking against the medium task's CPU share.
  • Add a message queue with blocking send and receive, and the wake-on-send that makes a waiting receiver runnable immediately rather than at the next tick.
  • Replace the O(n) tick with a delta list sorted by wake time.
  • Add the MPU stack guard from chapter 14, one region per task, reloaded in scheduler_next. Then recurse until it fires and confirm the fault names the right task.
  • Add FP context saving and verify with two tasks doing float maths that corruption appears without it and vanishes with it.
  • Make inheritance transitive across a chain of two mutexes.
  • Measure the switch. Toggle a pin in PendSV and count cycles with DWT. Compare against FreeRTOS on the same board.
