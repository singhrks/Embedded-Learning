Designing an ioctl interface

read() and write() move bytes. An ioctl is everything else — configure, query, control — and because it becomes part of the user-facing ABI the moment it ships, it deserves more care than any other part of a driver.

Encoding a request number

An ioctl request is not just a number you pick. It packs together the direction of the transfer, the size of the argument and an identifier for your driver, so that the kernel can catch a program calling the wrong driver's ioctl instead of quietly corrupting memory.

#include <linux/ioctl.h>

/* NEVER a bare integer. The encoding macros pack DIRECTION,
   SIZE and TYPE into the number, which is what lets the
   kernel validate calls generically and what lets tools like
   strace decode them without your driver's headers.        */

#define MYDEV_MAGIC  'k'                /* pick something
                                           unused - see
                                           Documentation/
                                           userspace-api/
                                           ioctl/ioctl-number.rst */

struct mydev_config {
        __u32  sample_rate_hz;
        __u32  channel_mask;
        __u8   gain;
        __u8   _pad[3];                 /* EXPLICIT padding -
                                           see below           */
};

#define MYDEV_GET_CONFIG   _IOR(MYDEV_MAGIC, 1, struct mydev_config)
#define MYDEV_SET_CONFIG   _IOW(MYDEV_MAGIC, 2, struct mydev_config)
#define MYDEV_RESET        _IO (MYDEV_MAGIC, 3)
#define MYDEV_XFER         _IOWR(MYDEV_MAGIC, 4, struct mydev_xfer)

/* _IO    no data transferred
   _IOR   kernel -> user  (a "get")
   _IOW   user -> kernel  (a "set")
   _IOWR  both

   These are NAMES for humans; nothing enforces that _IOR
   actually only reads. But get it right anyway - strace and
   generic ioctl tracers use it to know which direction to
   dump.                                                     */
The struct is a permanent ABI

Once a binary in the field calls MYDEV_SET_CONFIG with this struct layout, that layout is frozen forever. Pad explicitly — never rely on the compiler's implicit padding, which can differ between 32-bit and 64-bit builds of the same struct. Use fixed-width types (__u32, not int or unsigned long) so the size does not change with the architecture. And leave room: reserved fields cost nothing today and save you from ever needing MYDEV_SET_CONFIG_V2.

The handler

The handler is a switch statement, and the interesting part is everything around it: validating the command, copying the argument safely, and returning the right error for a request you do not recognise.

static long mydev_ioctl(struct file *file, unsigned int cmd,
                        unsigned long arg)
{
        struct mydev_session *s = file->private_data;
        struct mydev *d = s->dev;
        void __user *uarg = (void __user *)arg;
        struct mydev_config cfg;
        int ret;

        switch (cmd) {
        case MYDEV_GET_CONFIG:
                mutex_lock(&d->lock);
                cfg = d->config;
                mutex_unlock(&d->lock);

                if (copy_to_user(uarg, &cfg, sizeof(cfg)))
                        return -EFAULT;
                return 0;

        case MYDEV_SET_CONFIG:
                if (!(file->f_mode & FMODE_WRITE))
                        return -EBADF;    /* a read-only fd may
                                             not configure it   */

                if (copy_from_user(&cfg, uarg, sizeof(cfg)))
                        return -EFAULT;

                /* VALIDATE EVERYTHING. This struct came from
                   userspace and userspace is the attacker.   */
                if (cfg.sample_rate_hz == 0 ||
                    cfg.sample_rate_hz > MYDEV_MAX_RATE)
                        return -EINVAL;
                if (cfg.channel_mask & ~MYDEV_VALID_CHANNELS)
                        return -EINVAL;
                if (cfg.gain > MYDEV_MAX_GAIN)
                        return -EINVAL;

                mutex_lock(&d->lock);
                ret = mydev_apply_config(d, &cfg);
                if (!ret)
                        d->config = cfg;
                mutex_unlock(&d->lock);
                return ret;

        case MYDEV_RESET:
                if (!(file->f_mode & FMODE_WRITE))
                        return -EBADF;
                return mydev_hw_reset(d);

        default:
                return -ENOTTY;          /* THE conventional
                                             "no such ioctl",
                                             not -EINVAL       */
        }
}
Five rules, in order of how often they are violated
  1. Validate every field. Range, mask, size — the struct crossed a trust boundary.
  2. Check f_mode for anything that changes state.
  3. Use copy_to_user/copy_from_user, never a direct dereference of a user pointer (chapter 09).
  4. Return -ENOTTY for an unknown command, so a program can probe for support.
  5. Never trust a size or count field to bound a copy without an upper limit check first — that is a remotely-triggered kmalloc(attacker_controlled).

Variable-length transfers

Fixed-size structures cover most cases, but sometimes the caller needs to pass an array whose length it chooses. Doing that safely requires care, because the size now comes from user space and is therefore untrusted.

/* A common and commonly wrong pattern: a struct that carries a
   pointer and a count, for a buffer of user-chosen size.    */

struct mydev_xfer {
        __u64  buf;              /* NOT a pointer type - see
                                    below */
        __u32  len;
        __u32  flags;
};

static long mydev_xfer_ioctl(struct mydev *d, unsigned long arg)
{
        struct mydev_xfer x;
        void __user *ubuf;
        void *kbuf;
        int ret;

        if (copy_from_user(&x, (void __user *)arg, sizeof(x)))
                return -EFAULT;

        /* BOUND IT before allocating. Without this, a
           malicious len of 0xFFFFFFFF either fails an
           allocation harmlessly, or - on a kernel without
           kmalloc's internal limits - is a resource
           exhaustion vector. Always cap explicitly.        */
        if (x.len == 0 || x.len > MYDEV_MAX_XFER)
                return -EINVAL;

        kbuf = kmalloc(x.len, GFP_KERNEL);
        if (!kbuf)
                return -ENOMEM;

        ubuf = u64_to_user_ptr(x.buf);   /* the CORRECT cast -
                                            see below          */

        if (x.flags & MYDEV_XFER_WRITE) {
                if (copy_from_user(kbuf, ubuf, x.len)) {
                        ret = -EFAULT;
                        goto out;
                }
                ret = mydev_hw_write(d, kbuf, x.len);
        } else {
                ret = mydev_hw_read(d, kbuf, x.len);
                if (!ret && copy_to_user(ubuf, kbuf, x.len))
                        ret = -EFAULT;
        }

out:
        kfree(kbuf);
        return ret;
}

/* WHY __u64 buf, not a pointer field:

   A struct with a real pointer member has a DIFFERENT SIZE
   and LAYOUT on a 32-bit process running on a 64-bit kernel
   (compat mode) versus a native 64-bit process. __u64 is the
   same size everywhere, and u64_to_user_ptr() does the
   sign-correct, warning-free cast:

        #define u64_to_user_ptr(x) ( \
                (void __user *)(uintptr_t)(x) )

   This single choice is what saves you from writing a
   .compat_ioctl handler at all (see below).                */

32-bit compatibility

If a 32-bit program calls your ioctl on a 64-bit kernel, the structure it passes is laid out differently from the one your driver expects. There is a straightforward way to avoid the problem entirely, and a much less pleasant way to fix it after the fact.

/* If a 32-bit binary can ever run on this kernel - and on
   arm64 that is common - a struct containing a real pointer
   or a `long` has a DIFFERENT layout when called from that
   32-bit process. Without a compat handler, SET_CONFIG from a
   32-bit program corrupts the kernel's read of the struct.  */

static const struct file_operations mydev_fops = {
        .unlocked_ioctl = mydev_ioctl,
        .compat_ioctl   = compat_ptr_ioctl,   /* the escape
                                                 hatch, below */
};

/* compat_ptr_ioctl() is a generic helper that works AS LONG
   AS your structs contain no raw pointers or `long` fields -
   exactly the discipline the __u64/__u32-only design above
   already gives you. It converts the compat pointer and calls
   your normal handler. No second implementation needed.

/* If you cannot avoid a pointer-shaped field (rare, and
   usually a sign to redesign), you need a REAL compat
   handler with a parallel struct and explicit translation.
   Avoid that path; it doubles your maintenance burden and is
   a frequent source of security bugs in real drivers.       */

Consider not using ioctl at all

A bespoke ioctl is the most flexible interface available and, for that same reason, usually the wrong default: Linux already has a standard, subsystem-backed ABI for most needs a driver author reaches for ioctl to solve, and reusing it means generic tools already know how to use your device without reading your header file first.

NeedBetter than a custom ioctl
A simple scalar settingA sysfs attribute (chapter 25) — discoverable, self-documenting, one cat/echo
Streaming samplesread(), or a subsystem with its own buffering (IIO, V4L2)
Asynchronous eventspoll() plus read(), or uevent
A sensor, ADC, DACThe IIO subsystem — a whole ABI already exists
Temperature, voltage, fanhwmon
A structured, extensible protocolNetlink — versioned, introspectable, used by real subsystems
Something genuinely device-specific and complexioctl — the right tool, sometimes

An ioctl interface is invisible to generic tools, undiscoverable without your header, and — once shipped — permanent. Every one of the alternatives above is inspectable with tools that already exist. Reach for ioctl last, after checking whether a subsystem already solves your problem (chapter 19).

Shipping the header

Your ioctl definitions have to be identical in the kernel and in every program that calls them. The kernel has a specific place for headers that user space is meant to include, and putting them anywhere else guarantees the two copies drift apart.

/* include/uapi/linux/mydev.h - the UAPI header, shared
   VERBATIM between the kernel and any userspace program.   */

#ifndef _UAPI_LINUX_MYDEV_H
#define _UAPI_LINUX_MYDEV_H

#include <linux/types.h>
#include <linux/ioctl.h>

struct mydev_config {
        __u32 sample_rate_hz;
        __u32 channel_mask;
        __u8  gain;
        __u8  reserved[3];       /* MUST be zeroed by callers;
                                    reserved for future use   */
};

#define MYDEV_MAGIC        'k'
#define MYDEV_GET_CONFIG   _IOR(MYDEV_MAGIC, 1, struct mydev_config)
#define MYDEV_SET_CONFIG   _IOW(MYDEV_MAGIC, 2, struct mydev_config)
#define MYDEV_RESET        _IO (MYDEV_MAGIC, 3)

#endif

/* RULES for this file specifically:
     · __u8/__u16/__u32/__u64 only - never int, long, size_t
     · no kernel-only types (struct list_head, spinlock_t...)
     · no #ifdef __KERNEL__ tricks to hide fields - one struct,
       one meaning, for both sides
     · never remove or renumber an ioctl or a struct field
     · a NEW, incompatible need gets a NEW number - never
       redefine an existing one

   This header is literally installed into
   /usr/include/linux/mydev.h on end systems and copied into
   the musl/glibc UAPI headers. Programs compile against it
   for years after you have forgotten it exists.             */

Testing an ioctl

An ioctl cannot be exercised from the shell the way read and write can, so testing needs a small program. Writing it early is worthwhile — it doubles as the example other people will copy.

#include <fcntl.h>
#include <stdio.h>
#include <string.h>
#include <sys/ioctl.h>
#include <unistd.h>
#include <linux/mydev.h>

int main(void)
{
        int fd = open("/dev/mydev", O_RDWR);
        struct mydev_config cfg;

        if (ioctl(fd, MYDEV_GET_CONFIG, &cfg) < 0) {
                perror("GET_CONFIG");
                return 1;
        }
        printf("rate=%u mask=0x%x gain=%u\n",
               cfg.sample_rate_hz, cfg.channel_mask, cfg.gain);

        cfg.sample_rate_hz = 48000;
        if (ioctl(fd, MYDEV_SET_CONFIG, &cfg) < 0)
                perror("SET_CONFIG");        /* expect -EINVAL
                                                cases to work
                                                too - test them*/

        close(fd);
        return 0;
}
# strace decodes the direction/size/type automatically once it
# has your header or a matching ioctl.rst entry:
    strace -e ioctl ./mytest
      ioctl(3, MYDEV_GET_CONFIG or _IOC(_IOC_READ, 0x6b, 1, 8), ...) = 0

# NEGATIVE TESTS - the ones that actually find bugs:
    · call with a NULL pointer                 -> -EFAULT
    · call with an unmapped pointer            -> -EFAULT
    · call an unknown command number           -> -ENOTTY
    · SET_CONFIG on an O_RDONLY fd              -> -EBADF
    · out-of-range fields                      -> -EINVAL
    · a huge `len` in a variable transfer      -> -EINVAL,
                                                   not a hang
    · call from a 32-bit binary, if supported  -> matches
                                                   64-bit result

# Fuzz it - genuinely effective on ioctl handlers:
    syzkaller, or a simple harness with
    CONFIG_KCOV + CONFIG_KASAN, feeding random cmd/arg pairs
    at your fd. This finds the missing validation checks that
    manual testing never will.                              */
