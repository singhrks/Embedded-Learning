Project: a capture-to-encode pipeline

A complete program: capture NV12 from a V4L2 camera, encode it with a hardware codec, and mux it to a file — with the frames never once copied by the CPU. Everything in this track appears here at once, and the zero-copy path is the part that makes it worth building rather than downloading.

What we are building

   /dev/video0            /dev/video11           out.mp4
   CAMERA                 HARDWARE ENCODER
   V4L2 capture           V4L2 M2M
        |                      |
        |  VIDIOC_EXPBUF       |
        |  -> dmabuf fd        |
        +--------------------->|  OUTPUT queue
                               |  V4L2_MEMORY_DMABUF
                               |
                               |  the encoder reads the
                               |  CAMERA'S buffer directly.
                               |  ZERO COPIES.
                               |
                               |  CAPTURE queue
                               |  -> H.264 bitstream
                               v
                          userspace: mux to MP4

   THE PROPERTY WE ARE ENFORCING
     the CPU never touches a pixel. It moves file descriptors
     and a few hundred bytes of bitstream per frame.

   THE MEASUREMENT THAT PROVES IT
     at 1080p30 NV12, one copy is 93 MB/s. A naive pipeline
     does three or four (chapter 16). We will measure memory
     bandwidth and CPU and show both are near zero.

   TARGET: a Raspberry Pi 4/5, a Rockchip or i.MX board, or
   any x86 machine with a UVC camera and VA-API. The V4L2 code
   is identical; only the device node changes.

The shared header

/* pipeline.h */

#ifndef PIPELINE_H
#define PIPELINE_H

#include <linux/videodev2.h>
#include <stdint.h>
#include <stdbool.h>

#define MAX_BUFS 8

struct buf {
        void    *start;      /* mmap'd, only for MMAP bufs   */
        size_t   length;
        int      dmabuf_fd;  /* -1 if not exported           */
};

struct stream {
        int      fd;
        enum v4l2_buf_type type;
        enum v4l2_memory   memory;
        struct buf bufs[MAX_BUFS];
        unsigned n;
        uint32_t fourcc;
        int      width, height;
        int      stride;      /* NOT width (chapter 1)        */
};

/* Every ioctl retries on EINTR (chapter 17). */
int xioctl(int fd, unsigned long req, void *arg);

#endif

The capture side

/* capture.c - open the camera, export its buffers as dmabufs.
   The EXPBUF call is what makes zero-copy possible. */

#include "pipeline.h"
#include <sys/ioctl.h>
#include <sys/mman.h>
#include <fcntl.h>
#include <unistd.h>
#include <errno.h>
#include <string.h>
#include <stdio.h>

int xioctl(int fd, unsigned long req, void *arg)
{
        int r;
        do { r = ioctl(fd, req, arg); }
        while (r == -1 && errno == EINTR);
        return r;
}

int capture_open(struct stream *s, const char *dev,
                 int w, int h, unsigned nbuf)
{
        s->fd = open(dev, O_RDWR | O_NONBLOCK | O_CLOEXEC);
        if (s->fd < 0) { perror(dev); return -1; }

        s->type   = V4L2_BUF_TYPE_VIDEO_CAPTURE;
        s->memory = V4L2_MEMORY_MMAP;   /* the DRIVER
                                           allocates, and we
                                           export what it gave
                                           us                 */

        struct v4l2_capability cap = {0};
        if (xioctl(s->fd, VIDIOC_QUERYCAP, &cap) < 0) return -1;
        if (!(cap.capabilities & V4L2_CAP_STREAMING)) {
                fprintf(stderr, "no streaming support\n");
                return -1;
        }

        struct v4l2_format fmt = {0};
        fmt.type = s->type;
        fmt.fmt.pix.width       = w;
        fmt.fmt.pix.height      = h;
        fmt.fmt.pix.pixelformat = V4L2_PIX_FMT_NV12;
        fmt.fmt.pix.field       = V4L2_FIELD_NONE;
        if (xioctl(s->fd, VIDIOC_S_FMT, &fmt) < 0) return -1;

        /* READ BACK WHAT YOU GOT. The driver may have chosen
           something else entirely (chapter 17). */
        s->width  = fmt.fmt.pix.width;
        s->height = fmt.fmt.pix.height;
        s->stride = fmt.fmt.pix.bytesperline;
        s->fourcc = fmt.fmt.pix.pixelformat;

        if (s->fourcc != V4L2_PIX_FMT_NV12) {
                fprintf(stderr, "camera gave %.4s, not NV12 - "
                        "a conversion will be needed\n",
                        (char *)&s->fourcc);
                /* Do not silently continue. Either accept the
                   conversion cost deliberately, or pick a
                   camera mode that gives you NV12. */
        }

        struct v4l2_requestbuffers req = {0};
        req.count  = nbuf;
        req.type   = s->type;
        req.memory = s->memory;
        if (xioctl(s->fd, VIDIOC_REQBUFS, &req) < 0) return -1;
        s->n = req.count;         /* may be fewer than asked  */

        for (unsigned i = 0; i < s->n; i++) {
                struct v4l2_buffer b = {0};
                b.type = s->type; b.memory = s->memory; b.index = i;
                if (xioctl(s->fd, VIDIOC_QUERYBUF, &b) < 0) return -1;

                s->bufs[i].length = b.length;
                s->bufs[i].start  = mmap(NULL, b.length,
                                         PROT_READ | PROT_WRITE,
                                         MAP_SHARED, s->fd, b.m.offset);
                /* We mmap only so we CAN look at pixels while
                   debugging. The fast path never dereferences
                   this pointer. */

                /* THE KEY CALL: hand me a file descriptor
                   representing this buffer, so another driver
                   can import the same memory. */
                struct v4l2_exportbuffer exp = {0};
                exp.type  = s->type;
                exp.index = i;
                exp.flags = O_CLOEXEC | O_RDWR;
                if (xioctl(s->fd, VIDIOC_EXPBUF, &exp) < 0) {
                        perror("EXPBUF - no dmabuf export, so "
                               "no zero-copy on this driver");
                        return -1;
                }
                s->bufs[i].dmabuf_fd = exp.fd;
        }
        return 0;
}

int capture_start(struct stream *s)
{
        for (unsigned i = 0; i < s->n; i++) {
                struct v4l2_buffer b = {0};
                b.type = s->type; b.memory = s->memory; b.index = i;
                if (xioctl(s->fd, VIDIOC_QBUF, &b) < 0) return -1;
        }
        return xioctl(s->fd, VIDIOC_STREAMON, &s->type);
}
VIDIOC_EXPBUF is the whole project

It turns a driver-allocated buffer into a file descriptor that any other driver can import. Without it you must mmap the camera buffer, memcpy into an encoder buffer, and pay 93 MB/s at 1080p30 — plus a second copy if the encoder cannot take your memory either. A driver that does not implement EXPBUF cannot participate in a zero-copy pipeline at all, and that is worth discovering at design time rather than after the code is written.

The encoder side

/* encoder.c - a V4L2 M2M encoder whose OUTPUT queue imports
   the camera's dmabufs. */

int encoder_open(struct stream *out, struct stream *cap_out,
                 const struct stream *cam, int bitrate)
{
        int fd = open("/dev/video11", O_RDWR | O_NONBLOCK | O_CLOEXEC);
        if (fd < 0) { perror("encoder"); return -1; }
        out->fd = cap_out->fd = fd;   /* one device, two queues */

        /* --- OUTPUT queue: raw frames IN.
               "OUTPUT" is from the APPLICATION's point of view
               (chapter 17). --- */
        out->type   = V4L2_BUF_TYPE_VIDEO_OUTPUT_MPLANE;
        out->memory = V4L2_MEMORY_DMABUF;      /* <-- import   */

        struct v4l2_format f = {0};
        f.type = out->type;
        f.fmt.pix_mp.width        = cam->width;
        f.fmt.pix_mp.height       = cam->height;
        f.fmt.pix_mp.pixelformat  = V4L2_PIX_FMT_NV12;
        f.fmt.pix_mp.field        = V4L2_FIELD_NONE;
        f.fmt.pix_mp.num_planes   = 1;   /* NV12 is often ONE
                                            plane with the UV
                                            after the Y - not
                                            two. Check
                                            num_planes after
                                            S_FMT.            */
        f.fmt.pix_mp.plane_fmt[0].bytesperline = cam->stride;
        if (xioctl(fd, VIDIOC_S_FMT, &f) < 0) return -1;

        /* --- CAPTURE queue: compressed bitstream OUT --- */
        cap_out->type   = V4L2_BUF_TYPE_VIDEO_CAPTURE_MPLANE;
        cap_out->memory = V4L2_MEMORY_MMAP;

        struct v4l2_format g = {0};
        g.type = cap_out->type;
        g.fmt.pix_mp.width       = cam->width;
        g.fmt.pix_mp.height      = cam->height;
        g.fmt.pix_mp.pixelformat = V4L2_PIX_FMT_H264;
        g.fmt.pix_mp.num_planes  = 1;
        /* YOU size the bitstream buffer. Too small and frames
           are truncated - a keyframe is 10-20x a P-frame
           (chapter 10), so size for the keyframe. */
        g.fmt.pix_mp.plane_fmt[0].sizeimage = 2 * 1024 * 1024;
        if (xioctl(fd, VIDIOC_S_FMT, &g) < 0) return -1;

        /* --- controls: rate control and GOP (chapters 10, 13) */
        struct v4l2_control c;
        c.id = V4L2_CID_MPEG_VIDEO_BITRATE;        c.value = bitrate;
        xioctl(fd, VIDIOC_S_CTRL, &c);
        c.id = V4L2_CID_MPEG_VIDEO_BITRATE_MODE;
        c.value = V4L2_MPEG_VIDEO_BITRATE_MODE_CBR;
        xioctl(fd, VIDIOC_S_CTRL, &c);
        c.id = V4L2_CID_MPEG_VIDEO_GOP_SIZE;       c.value = 30;
        xioctl(fd, VIDIOC_S_CTRL, &c);
        c.id = V4L2_CID_MPEG_VIDEO_H264_PROFILE;
        c.value = V4L2_MPEG_VIDEO_H264_PROFILE_HIGH;
        xioctl(fd, VIDIOC_S_CTRL, &c);
        /* REPEAT THE HEADERS before every keyframe, or a
           player tuning in mid-stream has no SPS/PPS
           (chapter 10). */
        c.id = V4L2_CID_MPEG_VIDEO_REPEAT_SEQ_HEADER; c.value = 1;
        xioctl(fd, VIDIOC_S_CTRL, &c);
        /* Not every driver implements every control. Check the
           return, and do not assume silence means success. */

        /* --- buffers ---
           OUTPUT: DMABUF, so REQBUFS allocates only the
           bookkeeping, not memory. */
        struct v4l2_requestbuffers r = {0};
        r.count = cam->n; r.type = out->type; r.memory = out->memory;
        if (xioctl(fd, VIDIOC_REQBUFS, &r) < 0) return -1;
        out->n = r.count;

        /* CAPTURE: MMAP, and we DO read these - they are the
           bitstream, a few hundred KB per second, not pixels. */
        struct v4l2_requestbuffers r2 = {0};
        r2.count = 4; r2.type = cap_out->type; r2.memory = cap_out->memory;
        if (xioctl(fd, VIDIOC_REQBUFS, &r2) < 0) return -1;
        cap_out->n = r2.count;

        for (unsigned i = 0; i < cap_out->n; i++) {
                struct v4l2_plane p = {0};
                struct v4l2_buffer b = {0};
                b.type = cap_out->type; b.memory = cap_out->memory;
                b.index = i; b.m.planes = &p; b.length = 1;
                if (xioctl(fd, VIDIOC_QUERYBUF, &b) < 0) return -1;
                cap_out->bufs[i].length = p.length;
                cap_out->bufs[i].start  = mmap(NULL, p.length,
                        PROT_READ | PROT_WRITE, MAP_SHARED,
                        fd, p.m.mem_offset);
                if (xioctl(fd, VIDIOC_QBUF, &b) < 0) return -1;
        }

        if (xioctl(fd, VIDIOC_STREAMON, &out->type) < 0) return -1;
        if (xioctl(fd, VIDIOC_STREAMON, &cap_out->type) < 0) return -1;
        return 0;
}

The loop

/* main.c - three file descriptors, one poll, no pixel copies. */

int run(struct stream *cam, struct stream *enc_in,
        struct stream *enc_out, FILE *sink)
{
        /* Map camera buffer index -> encoder buffer index.
           They are separate index spaces; the DMABUF fd is
           what actually links them. */
        int cam_of_enc[MAX_BUFS];
        for (unsigned i = 0; i < MAX_BUFS; i++) cam_of_enc[i] = -1;

        unsigned free_enc = enc_in->n;   /* encoder input slots */
        uint64_t frames = 0;

        for (;;) {
                struct pollfd p[2] = {
                        { .fd = cam->fd,     .events = POLLIN },
                        { .fd = enc_out->fd, .events = POLLIN },
                };
                if (poll(p, 2, 1000) <= 0) continue;

                /* ---- a camera frame is ready ---- */
                if ((p[0].revents & POLLIN) && free_enc > 0) {
                        struct v4l2_buffer cb = {0};
                        cb.type = cam->type; cb.memory = cam->memory;
                        if (xioctl(cam->fd, VIDIOC_DQBUF, &cb) < 0)
                                continue;

                        /* Hand the SAME MEMORY to the encoder,
                           by file descriptor. No copy. */
                        unsigned slot = cb.index;   /* 1:1 here */
                        struct v4l2_plane pl = {0};
                        struct v4l2_buffer eb = {0};
                        eb.type      = enc_in->type;
                        eb.memory    = V4L2_MEMORY_DMABUF;
                        eb.index     = slot;
                        eb.m.planes  = &pl;
                        eb.length    = 1;
                        pl.m.fd      = cam->bufs[cb.index].dmabuf_fd;
                        pl.bytesused = cb.bytesused;
                        pl.length    = cam->bufs[cb.index].length;

                        /* CARRY THE CAPTURE TIMESTAMP THROUGH.
                           Do not stamp frames when your code
                           happens to run (chapter 15). */
                        eb.timestamp = cb.timestamp;
                        eb.flags     = V4L2_BUF_FLAG_TIMESTAMP_COPY;

                        if (xioctl(enc_in->fd, VIDIOC_QBUF, &eb) < 0) {
                                /* give the camera buffer back
                                   even on failure, or we leak
                                   it and the camera starves */
                                xioctl(cam->fd, VIDIOC_QBUF, &cb);
                                continue;
                        }
                        cam_of_enc[slot] = cb.index;
                        free_enc--;
                }

                /* ---- an encoded frame is ready ---- */
                if (p[1].revents & POLLIN) {
                        struct v4l2_plane pl = {0};
                        struct v4l2_buffer ob = {0};
                        ob.type     = enc_out->type;
                        ob.memory   = enc_out->memory;
                        ob.m.planes = &pl;
                        ob.length   = 1;
                        if (xioctl(enc_out->fd, VIDIOC_DQBUF, &ob) == 0) {
                                fwrite(enc_out->bufs[ob.index].start,
                                       1, pl.bytesused, sink);
                                /* pl.bytesused only - the buffer
                                   is 2 MB and the frame is not */
                                xioctl(enc_out->fd, VIDIOC_QBUF, &ob);
                                frames++;
                        }

                        /* Reclaim the OUTPUT buffer the encoder
                           has finished reading, and return the
                           camera buffer it was pointing at. */
                        struct v4l2_plane ipl = {0};
                        struct v4l2_buffer ib = {0};
                        ib.type     = enc_in->type;
                        ib.memory   = V4L2_MEMORY_DMABUF;
                        ib.m.planes = &ipl;
                        ib.length   = 1;
                        if (xioctl(enc_in->fd, VIDIOC_DQBUF, &ib) == 0) {
                                int ci = cam_of_enc[ib.index];
                                if (ci >= 0) {
                                        struct v4l2_buffer cb = {0};
                                        cb.type   = cam->type;
                                        cb.memory = cam->memory;
                                        cb.index  = ci;
                                        xioctl(cam->fd, VIDIOC_QBUF, &cb);
                                        cam_of_enc[ib.index] = -1;
                                }
                                free_enc++;
                        }
                }
        }
}

/* THE BUFFER DISCIPLINE, which is the part that goes wrong:

   A camera buffer is IN USE from the moment the encoder
   accepts it until the encoder returns the OUTPUT buffer.
   Requeue it to the camera before that and the encoder reads
   a frame the sensor is overwriting - producing tearing that
   looks like a decoder bug.

   And if you never requeue, the camera runs out of buffers
   and starts dropping frames silently. Watch
   v4l2_buffer.sequence for gaps (chapter 17).            */
The elementary stream is not a file anyone can play

out.h264 is raw Annex B — no container, no timestamps, no duration. It plays in ffmpeg because ffmpeg guesses a frame rate, and it will not play in most players or seek at all. Mux it. The -r below is a claim about frame rate, not a measurement; for variable-rate capture you must carry the real timestamps through and mux from those (chapter 15).
