IOCTLs and DeviceIoControl

The control interface between your driver and user mode. The control code encodes the device type, the access required and the buffering method — and because it is a permanent ABI reachable by whoever can open your device, it is where driver security is won or lost.

The control code

An IOCTL code is not an arbitrary number. It packs the device type, the access required, the buffering method and your function number into one value — which is what lets the I/O manager validate a request before your driver ever sees it.

#define CTL_CODE(DeviceType, Function, Method, Access) ( \
        ((DeviceType) << 16) | ((Access) << 14) |         \
        ((Function) << 2)    | (Method) )

/* 32 bits, four fields:

     31..16  DeviceType   FILE_DEVICE_UNKNOWN (0x22) for a
                          custom device
     15..14  Access       FILE_ANY_ACCESS
                          FILE_READ_ACCESS
                          FILE_WRITE_ACCESS
                          FILE_READ_ACCESS|FILE_WRITE_ACCESS
     13..2   Function     YOURS. 0x800-0xFFF is the range
                          reserved for third parties.
                          Below 0x800 is Microsoft's.
      1..0   Method       METHOD_BUFFERED     0
                          METHOD_IN_DIRECT    1
                          METHOD_OUT_DIRECT   2
                          METHOD_NEITHER      3   */

/* A realistic set: */

#define MY_DEVICE_TYPE  0x8000    /* >= 0x8000 for vendors */

#define IOCTL_MY_GET_VERSION \
    CTL_CODE(MY_DEVICE_TYPE, 0x800, METHOD_BUFFERED, \
             FILE_ANY_ACCESS)

#define IOCTL_MY_GET_CONFIG \
    CTL_CODE(MY_DEVICE_TYPE, 0x801, METHOD_BUFFERED, \
             FILE_READ_ACCESS)

#define IOCTL_MY_SET_CONFIG \
    CTL_CODE(MY_DEVICE_TYPE, 0x802, METHOD_BUFFERED, \
             FILE_WRITE_ACCESS)

#define IOCTL_MY_READ_BULK \
    CTL_CODE(MY_DEVICE_TYPE, 0x803, METHOD_OUT_DIRECT, \
             FILE_READ_ACCESS)

#define IOCTL_MY_RESET \
    CTL_CODE(MY_DEVICE_TYPE, 0x804, METHOD_BUFFERED, \
             FILE_WRITE_ACCESS)
The Access field is enforced by the I/O manager

An IOCTL declared FILE_WRITE_ACCESS is rejected before your driver sees it if the handle was not opened with write access. This is free, automatic, correct access control — and it is why FILE_ANY_ACCESS should be reserved for genuinely harmless queries. Marking a "reconfigure the device" IOCTL as FILE_ANY_ACCESS means any process that can open the device read-only can reconfigure it, which is precisely the bug Linux chapter 23 warns about — except here the platform would have prevented it for free.

The handler

The handler is a switch statement, and everything interesting is around it: validating lengths, retrieving buffers safely, and setting the number of bytes returned so the application knows how much is valid.

VOID
MyEvtIoDeviceControl(_In_ WDFQUEUE Queue,
                     _In_ WDFREQUEST Request,
                     _In_ size_t OutputBufferLength,
                     _In_ size_t InputBufferLength,
                     _In_ ULONG IoControlCode)
{
        WDFDEVICE device = WdfIoQueueGetDevice(Queue);
        PMY_CONTEXT ctx = MyGetContext(device);
        NTSTATUS status = STATUS_INVALID_DEVICE_REQUEST;
        size_t info = 0;
        PVOID inBuf, outBuf;
        size_t inLen, outLen;

        switch (IoControlCode) {

        case IOCTL_MY_GET_VERSION: {
                PMY_VERSION ver;

                status = WdfRequestRetrieveOutputBuffer(Request,
                                sizeof(MY_VERSION),
                                (PVOID *)&ver, &outLen);
                if (!NT_SUCCESS(status))
                        break;

                ver->Major = MY_VERSION_MAJOR;
                ver->Minor = MY_VERSION_MINOR;
                info = sizeof(MY_VERSION);
                status = STATUS_SUCCESS;
                break;
        }

        case IOCTL_MY_SET_CONFIG: {
                PMY_CONFIG cfg;
                MY_CONFIG local;

                status = WdfRequestRetrieveInputBuffer(Request,
                                sizeof(MY_CONFIG),
                                (PVOID *)&cfg, &inLen);
                if (!NT_SUCCESS(status))
                        break;

                /* COPY IN before validating. With
                   METHOD_BUFFERED this is already a kernel
                   copy, but copying to a local also protects
                   against the shared input/output buffer
                   overwriting your input (chapter 08).    */
                RtlCopyMemory(&local, cfg, sizeof(local));

                /* VALIDATE EVERY FIELD */
                if (local.SampleRate == 0 ||
                    local.SampleRate > MY_MAX_RATE) {
                        status = STATUS_INVALID_PARAMETER;
                        break;
                }
                if (local.ChannelMask & ~MY_VALID_CHANNELS) {
                        status = STATUS_INVALID_PARAMETER;
                        break;
                }
                if (local.Gain > MY_MAX_GAIN) {
                        status = STATUS_INVALID_PARAMETER;
                        break;
                }

                status = MyApplyConfig(ctx, &local);
                break;
        }

        case IOCTL_MY_READ_BULK: {
                PMDL mdl;

                status = WdfRequestRetrieveOutputWdmMdl(Request,
                                                        &mdl);
                if (!NT_SUCCESS(status))
                        break;

                /* start a DMA transfer into it (chapter 19),
                   and complete the request in the DPC.
                   Forward to the manual queue and return. */
                status = WdfRequestForwardToIoQueue(Request,
                                ctx->PendingQueue);
                if (NT_SUCCESS(status))
                        return;      /* do NOT complete here */
                break;
        }

        default:
                status = STATUS_INVALID_DEVICE_REQUEST;
                break;
        }

        WdfRequestCompleteWithInformation(Request, status, info);
}

/* NOTE the early `return` in the bulk case: a request that
   has been forwarded is no longer yours to complete
   (chapter 16). Falling through to the completion call would
   be a double completion and a bugcheck.               */

Internal IOCTLs

Some interfaces are meant for other drivers rather than for applications. Internal IOCTLs are exactly that — unreachable from user mode by design, which changes what you must validate and what you may trust.

/* IRP_MJ_INTERNAL_DEVICE_CONTROL is reachable ONLY from
   kernel mode. Use it for driver-to-driver interfaces.    */

        queueConfig.EvtIoInternalDeviceControl =
                        MyEvtIoInternalDeviceControl;

/* WHY IT MATTERS FOR SECURITY: an internal IOCTL cannot be
   invoked by a user-mode program at all - the I/O manager
   refuses. So a privileged operation exposed only as an
   internal IOCTL has no user-mode attack surface.

   USE IT for anything a peer driver needs but userspace must
   never reach: raw register access, firmware update entry
   points, diagnostic modes.

/* AND THE TRAP: an internal IOCTL's buffers come from a
   kernel caller, so RequestorMode is KernelMode and probing
   is skipped. That is correct - but it means you are
   trusting the calling driver completely. Do not expose an
   internal IOCTL that a user-mode-reachable driver simply
   forwards, because you have then re-opened the surface you
   were closing.                                          */

The user-mode side

The application side is short, and the important part is the shared header: the same definitions must be used by both sides, because a mismatch produces a silent misinterpretation rather than a compile error.

#include <windows.h>
#include <setupapi.h>
#include <cfgmgr32.h>
#include "mydevice.h"      /* the shared header */

/* --- find the device by interface GUID (chapter 06) ------ */
static HANDLE OpenMyDevice(void)
{
        HDEVINFO info;
        SP_DEVICE_INTERFACE_DATA ifData = { sizeof(ifData) };
        PSP_DEVICE_INTERFACE_DETAIL_DATA_W detail = NULL;
        DWORD needed = 0;
        HANDLE h = INVALID_HANDLE_VALUE;

        info = SetupDiGetClassDevsW(&GUID_DEVINTERFACE_MYDEVICE,
                        NULL, NULL,
                        DIGCF_PRESENT | DIGCF_DEVICEINTERFACE);
        if (info == INVALID_HANDLE_VALUE)
                return INVALID_HANDLE_VALUE;

        if (SetupDiEnumDeviceInterfaces(info, NULL,
                        &GUID_DEVINTERFACE_MYDEVICE, 0,
                        &ifData)) {

                SetupDiGetDeviceInterfaceDetailW(info, &ifData,
                                NULL, 0, &needed, NULL);
                detail = malloc(needed);
                if (detail) {
                        detail->cbSize = sizeof(*detail);
                        if (SetupDiGetDeviceInterfaceDetailW(
                                info, &ifData, detail, needed,
                                &needed, NULL)) {
                                h = CreateFileW(detail->DevicePath,
                                        GENERIC_READ | GENERIC_WRITE,
                                        0, NULL, OPEN_EXISTING,
                                        FILE_ATTRIBUTE_NORMAL, NULL);
                        }
                        free(detail);
                }
        }

        SetupDiDestroyDeviceInfoList(info);
        return h;
}

/* --- a synchronous IOCTL -------------------------------- */
        MY_CONFIG cfg = { .SampleRate = 48000,
                          .ChannelMask = 0x0F, .Gain = 3 };
        DWORD returned;

        if (!DeviceIoControl(h, IOCTL_MY_SET_CONFIG,
                             &cfg, sizeof(cfg),     /* in  */
                             NULL, 0,               /* out */
                             &returned, NULL)) {
                printf("SET_CONFIG failed: %lu\n",
                       GetLastError());
        }

/* --- an asynchronous one -------------------------------- */
        HANDLE h = CreateFileW(path, GENERIC_READ | GENERIC_WRITE,
                        0, NULL, OPEN_EXISTING,
                        FILE_FLAG_OVERLAPPED,    /* <-- async */
                        NULL);

        OVERLAPPED ov = {0};
        ov.hEvent = CreateEvent(NULL, TRUE, FALSE, NULL);
        BYTE buffer[4096];

        if (!DeviceIoControl(h, IOCTL_MY_READ_BULK,
                             NULL, 0,
                             buffer, sizeof(buffer),
                             NULL, &ov)) {
                if (GetLastError() != ERROR_IO_PENDING) {
                        /* a real error */
                }
        }

        /* do other work, then: */
        WaitForSingleObject(ov.hEvent, INFINITE);

        DWORD transferred;
        if (GetOverlappedResult(h, &ov, &transferred, FALSE))
                printf("read %lu bytes\n", transferred);

        CloseHandle(ov.hEvent);

/* THE PENDING CONVENTION: DeviceIoControl returns FALSE with
   GetLastError() == ERROR_IO_PENDING when the driver returned
   STATUS_PENDING. That is SUCCESS, not failure - a mistake
   almost every first async client makes.                */

Designing the interface

Rules, in order of how much damage breaking them causes
  1. Validate every field of every input. The caller is untrusted, always, even if today only your own application calls it.
  2. Use the Access field. Free enforcement; use it rather than checking by hand.
  3. Fixed-width types, explicit padding. UINT32, not ULONG or int; pad so the layout is identical for 32- and 64-bit callers.
  4. Version the interface from day one, with a GET_VERSION IOCTL that never changes.
  5. Reserved fields, required to be zero. They cost nothing and are the only way to extend a struct later.
  6. Never embed a pointer in a buffer. It breaks 32-on-64 and it is a confused-deputy invitation.
  7. Bound every length. A size field that drives an allocation is a denial-of-service vector.
  8. Return STATUS_INVALID_DEVICE_REQUEST for unknown codes, so callers can probe for support.
  9. Keep the surface small. Every IOCTL is an entry point into kernel mode from whoever can open the device.
/* THE 32-ON-64 PROBLEM, and why fixed-width types matter: */

/* WRONG - different layout for a 32-bit caller: */
typedef struct {
        ULONG   Count;
        PVOID   Buffer;     /* 4 bytes on Win32, 8 on Win64 */
        SIZE_T  Length;     /* same problem                 */
} BAD_REQUEST;

/* RIGHT: */
typedef struct {
        UINT32  Count;
        UINT32  Reserved;   /* explicit padding */
        UINT64  BufferOffset;   /* an OFFSET, not a pointer */
        UINT32  Length;
        UINT32  Reserved2[3];
} GOOD_REQUEST;

/* AND IF YOU MUST support both layouts, detect the caller: */
        if (IoIs32bitProcess(NULL)) {
                /* the request came from a 32-bit process -
                   parse the 32-bit layout                  */
        }

/* THE BETTER ANSWER: design so there is only one layout.
   Fixed-width types and no pointers achieves it, exactly as
   in Linux chapter 23.                                     */

Testing

An IOCTL interface is worth exercising deliberately — including with the wrong sizes and invalid values — because those are precisely the paths an attacker will take and the ones your normal client never will.

# NEGATIVE TESTS - these find the bugs:
    · an unknown control code       -> STATUS_INVALID_DEVICE_REQUEST
    · a zero-length input buffer     -> STATUS_BUFFER_TOO_SMALL
    · an input buffer one byte short -> same
    · a NULL output buffer           -> handled, not a crash
    · out-of-range field values      -> STATUS_INVALID_PARAMETER
    · a huge length field            -> refused, not a hang
    · SET_CONFIG on a read-only handle -> refused by the I/O
                                          manager
    · concurrent calls from many threads -> no corruption
    · a 32-bit client, if you support one

# TOOLS:
    devcon.exe                  install, enable, disable

                                interface. REQUIRED for
                                certification anyway, so run
                                them early (chapter 22).
