Writing a platform driver

Everything from the last two chapters, assembled into one complete, correct driver. This is the skeleton you will start from every time — read it once carefully and you have the shape of most of drivers/.

What a platform device is

A platform device is the kernel's category for hardware that cannot announce itself — memory-mapped blocks inside an SoC, described by the device tree rather than discovered. It is the most common kind of driver in embedded work, and the template every other bus driver follows.

/* A device that is NOT on a discoverable bus. It exists
   because the device tree says so.

     · memory-mapped registers at a fixed address
     · one or more interrupts
     · clocks, resets, regulators, GPIOs, DMA channels
     · no vendor/product ID to enumerate

   That describes most of an SoC: UARTs, timers, watchdogs,
   crypto engines, DMA controllers, display pipelines, custom
   accelerators.

   IF YOUR DEVICE IS ON I2C, SPI OR PCI, use that bus's driver
   type instead - i2c_driver, spi_driver, pci_driver. The
   shape is identical; only the resource acquisition and the
   register access differ.                                  */

The complete driver

This is a complete, working driver with nothing omitted. Read it once end to end to see the shape, then use the sections after it to understand each piece — every driver you write later will be a variation on exactly this.

// SPDX-License-Identifier: GPL-2.0
/*
 * Vendor MyDevice driver
 *
 * Locking:
 *      d->lock protects d->state and the register block.
 *      Never held across a sleep other than the regmap I/O
 *      it wraps.
 */

#include <linux/clk.h>
#include <linux/interrupt.h>
#include <linux/io.h>
#include <linux/mod_devicetable.h>
#include <linux/module.h>
#include <linux/platform_device.h>
#include <linux/pm_runtime.h>
#include <linux/property.h>
#include <linux/regulator/consumer.h>
#include <linux/reset.h>

#define MYDEV_REG_ID            0x00
#define MYDEV_REG_CTRL          0x04
#define  MYDEV_CTRL_ENABLE      BIT(0)
#define  MYDEV_CTRL_RESET       BIT(1)
#define MYDEV_REG_STATUS        0x08
#define  MYDEV_STATUS_IRQ       BIT(0)
#define  MYDEV_STATUS_BUSY      BIT(1)
#define MYDEV_REG_DATA          0x0c
#define MYDEV_REG_INT_EN        0x10

#define MYDEV_EXPECTED_ID       0x4d594445

/* Per-variant differences, selected by compatible string */
struct mydev_config {
        unsigned int nr_channels;
        unsigned int fifo_depth;
        bool         has_dma;
};

struct mydev {
        struct device           *dev;
        void __iomem            *base;
        struct clk              *clk;
        struct reset_control    *rst;
        struct regulator        *vdd;
        int                      irq;

        const struct mydev_config *cfg;
        u32                      fifo_depth;

        struct mutex             lock;      /* see header */
        wait_queue_head_t        wq;
        bool                     done;
        u32                      last_status;
};

/* ---- register helpers ---------------------------------- */

static inline u32 mydev_rd(struct mydev *d, u32 reg)
{
        return readl(d->base + reg);
}

static inline void mydev_wr(struct mydev *d, u32 reg, u32 val)
{
        writel(val, d->base + reg);
}

/* ---- interrupt ----------------------------------------- */

static irqreturn_t mydev_isr(int irq, void *dev_id)
{
        struct mydev *d = dev_id;
        u32 status;

        status = mydev_rd(d, MYDEV_REG_STATUS);
        if (!(status & MYDEV_STATUS_IRQ))
                return IRQ_NONE;        /* shared line */

        mydev_wr(d, MYDEV_REG_STATUS, MYDEV_STATUS_IRQ);  /* ack */

        d->last_status = status;
        d->done = true;
        wake_up_interruptible(&d->wq);

        return IRQ_HANDLED;
}

/* ---- hardware ------------------------------------------ */

static int mydev_hw_init(struct mydev *d)
{
        u32 id;

        /* IDENTITY CHECK - catches a wrong address, an
           unclocked block, or an access-control denial
           (isolation ch. 20) immediately, at probe, instead
           of as mysterious behaviour later.               */
        id = mydev_rd(d, MYDEV_REG_ID);
        if (id != MYDEV_EXPECTED_ID) {
                dev_err(d->dev, "bad ID 0x%08x (expected 0x%08x)\n",
                        id, MYDEV_EXPECTED_ID);
                return -ENODEV;
        }

        /* reset, then wait for it to clear */
        mydev_wr(d, MYDEV_REG_CTRL, MYDEV_CTRL_RESET);
        udelay(10);
        mydev_wr(d, MYDEV_REG_CTRL, 0);

        mydev_wr(d, MYDEV_REG_INT_EN, MYDEV_STATUS_IRQ);
        mydev_wr(d, MYDEV_REG_CTRL, MYDEV_CTRL_ENABLE);

        return 0;
}

static void mydev_hw_shutdown(struct mydev *d)
{
        mydev_wr(d, MYDEV_REG_INT_EN, 0);
        mydev_wr(d, MYDEV_REG_CTRL, 0);
}

/* ---- probe --------------------------------------------- */

static int mydev_probe(struct platform_device *pdev)
{
        struct device *dev = &pdev->dev;
        struct mydev *d;
        int ret;

        d = devm_kzalloc(dev, sizeof(*d), GFP_KERNEL);
        if (!d)
                return -ENOMEM;

        d->dev = dev;
        mutex_init(&d->lock);
        init_waitqueue_head(&d->wq);

        d->cfg = of_device_get_match_data(dev);
        if (!d->cfg)
                return -EINVAL;

        /* --- resources --- */

        d->base = devm_platform_ioremap_resource(pdev, 0);
        if (IS_ERR(d->base))
                return PTR_ERR(d->base);

        d->vdd = devm_regulator_get(dev, "vdd");
        if (IS_ERR(d->vdd))
                return dev_err_probe(dev, PTR_ERR(d->vdd),
                                     "no vdd supply\n");

        d->clk = devm_clk_get(dev, NULL);
        if (IS_ERR(d->clk))
                return dev_err_probe(dev, PTR_ERR(d->clk),
                                     "no clock\n");

        d->rst = devm_reset_control_get_optional_exclusive(dev,
                                                           NULL);
        if (IS_ERR(d->rst))
                return dev_err_probe(dev, PTR_ERR(d->rst),
                                     "no reset control\n");

        d->irq = platform_get_irq(pdev, 0);
        if (d->irq < 0)
                return d->irq;

        /* --- optional DT properties --- */

        ret = device_property_read_u32(dev, "vendor,fifo-depth",
                                       &d->fifo_depth);
        if (ret)
                d->fifo_depth = d->cfg->fifo_depth;

        /* --- power on, in order --- */

        ret = regulator_enable(d->vdd);
        if (ret)
                return dev_err_probe(dev, ret,
                                     "cannot enable vdd\n");

        ret = clk_prepare_enable(d->clk);
        if (ret) {
                dev_err(dev, "cannot enable clock: %d\n", ret);
                goto err_reg;
        }

        ret = reset_control_deassert(d->rst);
        if (ret) {
                dev_err(dev, "cannot deassert reset: %d\n", ret);
                goto err_clk;
        }

        ret = mydev_hw_init(d);
        if (ret)
                goto err_rst;

        /* --- interrupt: after the hardware is quiet --- */

        ret = devm_request_irq(dev, d->irq, mydev_isr,
                               IRQF_SHARED, dev_name(dev), d);
        if (ret) {
                dev_err(dev, "cannot request irq %d: %d\n",
                        d->irq, ret);
                goto err_hw;
        }

        platform_set_drvdata(pdev, d);

        dev_info(dev, "ready: %u channels, fifo %u\n",
                 d->cfg->nr_channels, d->fifo_depth);
        return 0;

err_hw:
        mydev_hw_shutdown(d);
err_rst:
        reset_control_assert(d->rst);
err_clk:
        clk_disable_unprepare(d->clk);
err_reg:
        regulator_disable(d->vdd);
        return ret;
}

/* ---- remove: the exact mirror -------------------------- */

static void mydev_remove(struct platform_device *pdev)
{
        struct mydev *d = platform_get_drvdata(pdev);

        mydev_hw_shutdown(d);           /* stop interrupts
                                           BEFORE devm frees
                                           the handler       */
        reset_control_assert(d->rst);
        clk_disable_unprepare(d->clk);
        regulator_disable(d->vdd);
}

/* ---- match --------------------------------------------- */

static const struct mydev_config mydev_v1_cfg = {
        .nr_channels = 4,
        .fifo_depth  = 16,
        .has_dma     = false,
};

static const struct mydev_config mydev_v2_cfg = {
        .nr_channels = 8,
        .fifo_depth  = 64,
        .has_dma     = true,
};

static const struct of_device_id mydev_of_match[] = {
        { .compatible = "vendor,my-device-v2",
          .data = &mydev_v2_cfg },
        { .compatible = "vendor,my-device",
          .data = &mydev_v1_cfg },
        { }
};
MODULE_DEVICE_TABLE(of, mydev_of_match);

static struct platform_driver mydev_driver = {
        .driver = {
                .name           = "my-device",
                .of_match_table = mydev_of_match,
                .pm             = pm_ptr(&mydev_pm_ops),
        },
        .probe  = mydev_probe,
        .remove = mydev_remove,
};
module_platform_driver(mydev_driver);

MODULE_DESCRIPTION("Vendor MyDevice driver");
MODULE_AUTHOR("Your Name <you@example.com>");
MODULE_LICENSE("GPL");
Read the error path

The goto ladder unwinds in exactly the reverse order of setup, each label undoing one step. This is idiomatic kernel C, not a code smell — it is the only structure that stays correct when you add a step in the middle. Chapter 33 shows how devm_ removes most of it, and why some of it always remains.

Acquiring the common resources

Almost every device needs the same handful of things before it can work: registers, a clock, power, a reset line, an interrupt. Each has a standard accessor that reads the device tree for you, and each can fail in the same way if the provider has not probed yet.

/* --- MMIO ------------------------------------------------ */
        d->base = devm_platform_ioremap_resource(pdev, 0);
        /* by name, better when there are several: */
        d->base = devm_platform_ioremap_resource_byname(pdev,
                                                        "ctrl");
        /* the long form, if you need the resource itself: */
        struct resource *res;
        res = platform_get_resource(pdev, IORESOURCE_MEM, 0);
        d->base = devm_ioremap_resource(dev, res);
        d->phys = res->start;

/* --- CLOCKS ---------------------------------------------- */
        d->clk = devm_clk_get(dev, "core");     /* by name */
        d->clk = devm_clk_get(dev, NULL);       /* the only one */
        clk_prepare_enable(d->clk);
        clk_disable_unprepare(d->clk);

        /* the modern one-liner - gets AND enables, and
           disables automatically on unbind: */
        d->clk = devm_clk_get_enabled(dev, "core");

        /* several at once: */
        struct clk_bulk_data clks[] = {
                { .id = "core" }, { .id = "bus" },
        };
        devm_clk_bulk_get(dev, ARRAY_SIZE(clks), clks);
        clk_bulk_prepare_enable(ARRAY_SIZE(clks), clks);

        rate = clk_get_rate(d->clk);
        clk_set_rate(d->clk, 100000000);

/* --- OPTIONAL resources: use the _optional variants ------ */
        d->rst = devm_reset_control_get_optional_exclusive(dev,
                                                           NULL);
        /* returns NULL, not an error, if absent - and the
           reset_control_* calls accept NULL harmlessly. So
           optional resources need no `if` at every use.    */

        d->gpio = devm_gpiod_get_optional(dev, "reset",
                                          GPIOD_OUT_HIGH);
        gpiod_set_value_cansleep(d->gpio, 0);

/* --- REGULATORS ------------------------------------------ */
        d->vdd = devm_regulator_get(dev, "vdd");
        regulator_enable(d->vdd);
        /* several: */
        devm_regulator_bulk_get(dev, n, supplies);

/* --- DMA channels (chapter 30) --------------------------- */
        d->rxchan = dma_request_chan(dev, "rx");
        if (IS_ERR(d->rxchan))
                return dev_err_probe(dev, PTR_ERR(d->rxchan),
                                     "no rx dma\n");

/* --- FIRMWARE -------------------------------------------- */
        const struct firmware *fw;

        ret = request_firmware(&fw, "vendor/mydev-v2.bin", dev);
        if (ret)
                return dev_err_probe(dev, ret,
                                     "firmware not found\n");
        /* fw->data / fw->size - push it to the device, then: */
        release_firmware(fw);           /* ALWAYS - the buffer is
                                           not devm-managed      */

/* The file is looked up under /lib/firmware. The blob is NOT
   part of your module: it is loaded from the filesystem, which
   means it may not exist yet when a built-in driver probes at
   boot. Two ways out:

     request_firmware_nowait()   asynchronous - probe returns
                                 immediately, a callback runs
                                 when the file is available.
                                 THE RIGHT DEFAULT for anything
                                 that might probe before the
                                 root filesystem is mounted.
     firmware in the initramfs   if the device must work early

   Firmware also has licensing implications: the blob usually
   is not GPL, lives in the separate linux-firmware repository,
   and cannot simply be committed alongside your driver.     */

Power-on ordering

Hardware usually has to be brought up in a specific order — power before clock, clock before reset release — and the datasheet will say so. Getting this wrong produces a device that works on one board and not another, or works only after a warm reboot.

Hardware has a sequence, and the datasheet's timing diagram is
the specification. The usual shape:

   1. REGULATORS on          (the device has power)
   2. wait the specified settling time
   3. CLOCKS enabled         (it has a clock to run on)
   4. RESET deasserted       (it starts)
   5. wait for it to be ready
   6. verify the ID register
   7. configure
   8. enable interrupts LAST

  AND EXACTLY THE REVERSE ON THE WAY DOWN.

  GETTING IT WRONG produces some of the most confusing
  symptoms in embedded work:

    reset released before the clock runs
        -> the block latches garbage; often works at room
           temperature and fails cold
    register written before power is stable
        -> the write is lost, silently
    clock enabled before the regulator
        -> in-rush current, a brownout, or a latch-up
    interrupts enabled before the handler is registered
        -> "nobody cared", or a crash in an unregistered
           handler
    no settling delay
        -> intermittent, temperature- and part-dependent

  READ THE POWER-UP SEQUENCE in the datasheet and implement it
  literally, including the delays. This is one of the few
  places in software where "it works on my board" is
  genuinely not evidence of correctness.

Testing it

Once it builds, you need to confirm the kernel actually bound it to your device rather than silently ignoring it. These checks take a few seconds and distinguish "my driver is broken" from "my driver never ran."

# 1. Does it bind?
    dmesg | grep my-device
    ls -l /sys/bus/platform/drivers/my-device/

# 2. Are the registers reachable?
    dmesg      # the ID check tells you immediately

# 3. Does probe fail cleanly?
#    Force each failure and confirm no leak and no crash:
#      - remove the clock from the DT
#      - point reg at a wrong address
#      - make the ID check fail
#    Each should log one clear error and unwind fully.

# 4. Unbind and rebind, repeatedly, under KASAN:
    for i in $(seq 50); do
      echo 1c000000.my-device > .../my-device/unbind
      echo 1c000000.my-device > .../my-device/bind
    done
    dmesg | grep -i -e kasan -e warn -e bug

# 5. Load and unload as a module, 100 times (chapter 17).

# 6. Check for leaked resources after unbind:
    cat /proc/iomem | grep 1c000000     # should be gone
    cat /proc/interrupts | grep my-device
    cat /proc/vmallocinfo | grep my_probe

# ALL FIVE of these find real bugs in code that "works".
